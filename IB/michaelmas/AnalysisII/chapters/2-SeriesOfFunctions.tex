\chapter{Series of Functions}
\section{Definitions}
In the previous chapter, we extended our theory of sequences of numbers to sequences of functions.
We would also like to extend the idea of series of numbers to series of functions.
\begin{definition}[$n$-th Partial Sum]
  Let $g_n: E \to \R$ be  a sequence of functions.
  We define its \textit{$n$-th partial sum} to be the sequence of functions $f_n: E \to \R$ given by:
  \[
    f_n(x) = \sum_{j = 1}^{n} g_j(x)
  \]
\end{definition}
\begin{definition}[Convergence of a series of functions at a point]
  We say that the series of functions $\sum_{j = 1}^{\infty} g_j$ converges at a point $x \in E$ if the sequence $(f_n(x))$ converges.
\end{definition}
\begin{definition}[Uniform convergece for series of functions]
  We say that the series of functions $\sum_{j = 1}^{\infty} g_j$ \textit{converges uniformly} on $E$ if $(f_n)$ converges uniformly on $E$.
\end{definition}
\section{Absolute Convergence}
We can also extend the concept of absolute convergence.
\begin{definition}[Absolute convergence for series of functions at a point]
  We say that the series of functions $\sum_{j = 1}^{\infty} g_j$ \textit{converges absolutely} at a point $x \in E$ if the series $\sum_{j = 1}^{\infty} |g_j(x)|$ converges.
\end{definition}
\begin{definition}[Absolute uniform convergence for series of functions]
  We say that the series of functions $\sum_{j = 1}^{\infty} g_j$ \textit{converges absolutely uniformly} on $E$ if the series of functions $\sum_{j = 1}^{\infty} |g_j|$ converges uniformly.
\end{definition}
\begin{remark}[Recap]
  Recall from Analysis I that for series of numbers in $\C$, absolute convergence of the series implies convergence.
\end{remark}
\begin{proposition}
  If $g_j: E \to \R$ is a sequence of functions and $\sum_{j = 1}^{\infty} g_j$ converges uniformly absolutely on $E$, then $\sum_{j = 1}^{\infty} g_j$ converges uniformly on $E$.
\end{proposition}
\begin{proof}
  Define the partial sums $f_n(x) = \sum_{j = 1}^{n} g_j(x)$ and $h_n(x) = \sum_{j = 1}^{n} |g_j(x)|$.
  We do not know what the limit function of $f_n$ should be and so we want to appeal to \cref{cauchyCritereon}.
  Considering $n > m$, we have:
  \begin{align*}
    |f_n(x) - f_m(x)| &= \abs{\sum_{j = m + 1}^{n} g_j(x)} \\
                      &\leq \sum_{j = m + 1}^{n} |g_j(x)| \\
                      &= h_n(x) - h_m(x) \\
                      &= |h_n(x) - h_m(x)|
  \end{align*}
  Therefore:
  \[
    \sup_{x \in E} |f_n(x) - f_m(x)| \leq \sup_{x \in E} |h_n(x) - h_m(x)|
  \]
  Since $\sum_{j = 1}^{\infty} g_j$ converges absolutely uniformly, $h_n$ converges uniformly and so by \cref{cauchyCritereon} it is uniformly Cauchy.
  Thus as $m \to \infty$ $\sup_{x \in E} |f_n(x) - f_m(x)| \to 0$ and so $(f_n)$ converges uniformly by \cref{cauchyCritereon}.
\end{proof}
\begin{remark}
  Uniform convergence of $\sum_{j = 1}^{\infty} g_j$ and absolute convergence of $\sum_{j = 1}^{\infty} g_j(x)$ at every $x \in E$ is \textbf{not} enough to conclude that a series is absolutely uniformly convergent.

  For example, consider the series of functions defined on $E = (0, 1)$:
  \[
    \sum_{j = 1}^{\infty} \frac{(-1)^j}{j} x^j
  \]
  Using the ratio test for power series from IA Analysis I, $\sum_{j = 1}^{\infty} \frac{x^j}{j}$ has radius of convergence $R = 1$.
  Thus $\sum_{j = 1}^{\infty} \frac{(-1)^j}{j} x^j$ converges absolutely as a series of numbers at every $x \in (0, 1)$.

  We will see on Example Sheet 1 Q7 that the series converges uniformly on $(0, 1)$.
  You will see that we are able to do this because the series $\sum_{j = 1}^{\infty} \frac{(-1)^j}{j}$ converges (by the alternating series test) which is the behaviour at the endpoint of $(0, 1)$.

  However, when we consider the absolute version, we now have that $\sum_{j = 1}^{\infty} \frac{1}{j}$ at the endpoint and so we cannot use the strategy above.
  In fact, this does not converge absolutely uniformly on $(0, 1)$.
  We can demonstrate this by showing that it violates the uniform Cauchy criterion.
  For $n > m$ we have:
  \begin{align*}
    \sum_{j = 1}^{n} \abs{\frac{(-1)^j}{j} x^j} - \sum_{j = 1}^{m} \abs{\frac{(-1)^j}{j} x^j} &= \sum_{j = n + 1}^{n} \frac{x^j}{j} \\
                                                                                              &\geq \sum_{j = m + 1}^{n}\frac{x^n}{j}
  \end{align*}
  Taking the supremum of both sides over $(0, 1)$:
  \[
    \sup_{x \in (0, 1)} \left[\sum_{j = 1}^{n} \abs{\frac{(-1)^j}{j} x^j} - \sum_{j = 1}^{m} \abs{\frac{(-1)^j}{j} x^j}\right] \geq \sum_{j = m + 1}^{n} \frac{1}{j}
  \]
  But for any fixed $m$, as $n \to \infty$, $\sum_{j = m + 1}^{n} \frac{x^{n}}{j} \to \infty$ and so $\sum_{j = 1}^{\infty} \abs{\frac{(-1)^j}{j} x^j}$ does not converge uniformly on $(0, 1)$.
\end{remark}
We have the following simple but useful criterion for absolute uniform convergence.
\begin{theorem}[Weierstrass $M$-test]
  Let $g_j: E \to \R$ be a series of functions.
  If there exists $M_j \geq 0$ such that: \label{MTest}
  \[
    \sup_{x \in E} |g_j(x)| \leq M_j\ \forall j \in \N
  \]
  and $\sum_{j = 1}^{\infty} M_j$ converges, then $\sum_{j = 1}^{\infty} g_j$ converges absolutely uniformly on $E$.
\end{theorem}
\begin{proof}
  Define the partial sums $h_n(x) = \sum_{j = 1}^{n} |g_j(x)|$ and $s_n = \sum_{j = 1}^{n} M_j$.

  We again wish to appeal to the Cauchy critereon and so considering $n > m$:
  \begin{align*}
    h_n(x) - h_m(x) &= \sum_{j = m + 1}^{n} |g_j(x)| \\
                    &\leq \sum_{j = m + 1}^{n} M_j \\
                    &= s_n - s_m
  \end{align*}
  Thus for all $n, m$:
  \[
    \sup_{x \in E} \abs{h_n(x) - h_m(x)} \leq |s_n - s_m|
  \]
  Since $\sum_{j = 1}^{\infty} M_j$ converges, its partial sums $(s_n)$ are Cauchy and so $h_n$ is uniformly Cauchy on $E$.
  Thus by \cref{cauchyCritereon}, $h_n$ converges uniformly on $E$, i.e. $\sum_{j }^{\infty} g_j$ converges uniformly absolutely on $E$.
\end{proof}
\section{Power Series}
\begin{remark}[Recap]
  Recall from IA Analaysis 1 that if $\sum_{n = 0}^{\infty} c_n(x - a)^n$ is a real power series then there exists a unique $R \in [0, \infty]$ called the \textit{radius of convergence}, such that:
  \begin{itemize}
    \item If $|x - a| < R$, then the series converges absolutely.
    \item If $|x - a| > R$, then the series diverges.
  \end{itemize}
  Further, we can obtain $R$ using the following:
  \[
    R^{-1} = \limsup_{n \to \infty} |c_n|^{1/n}
  \]
\end{remark}
\begin{theorem}[Absolute uniform convergence of power series]
  Let $R$ be the radius of convergence of the power series $\sum_{n = 0}^{\infty} c_n (x - a)^n$.
  If $R > 0$ and $r$ is such that $0 < r < R$, then the power series $\sum_{n = 0}^{\infty} c_n (x - a)^n$ converges absolutely uniformly on $[a - r, a + r]$. \label{powerConv}
\end{theorem}
\begin{remark}
  Since the power series converges uniformly on $[a - r, a + r]$ for any $r$ with $0 < r < R$ and each of the partial sums is a finite polynomial, by \cref{unifContFunc}, $\sum_{n = 0}^{\infty} c_n(x - a)^n$ is continuous on $[a - r, a + r]$.

  We can then define the function:
  \[
    f: (a - R, a + R) \to \R,\ x \mapsto \sum_{n = 0}^{\infty} c_n(x - a)^n
  \]
  which is therefore continuous on its whole domain.
\end{remark}
\begin{proof}
  For any $r$ with $0 < r < R$, from IA Analysis I, the power series converges absolutely at $x = a + r$, i.e. $\sum_{n = 0}^{\infty} |c_n|r^n$ is convergent.

  We then see that for any $x \in [a - r, a + r]$, we have $|c_n(x - a)^n| \leq |c_n| |x - a|^n \leq |c_n| r^n$, thus:
  \[
    \sup_{x \in [a - r, a + r]} |c_n(x - a)^n| \leq |c_n| r^n
  \]
  So we can apply the $M$-test (\cref{MTest}) with $M_n = |c_n| r^n$ and so the series converges absolutely uniformly on $[a - r,  a + r]$.
\end{proof}
\begin{definition}[Locally Uniform Convergence]
  Let $I = (a, b) \subseteq \R$ and $g_n, g: I \to \R$.
  We say that $g_n \to g$ \textit{locally uniformly} on $I$ if for all $y \in I$, there exists an $r > 0$ such that $g_n \to g$ uniformly on $[y - r, y + r]$.
\end{definition}
\begin{remark}[Intuition]
  This is not the same as uniform convergence on $(a, b)$.
  Because our choice of $N(\varepsilon)$ for uniform convergence has to work for the entire interval, we are saying that for any point, we can find an non-empty interval around it on which we have uniform convergence.
\end{remark}
So \cref{powerConv} tells us that that the power series converges locally absolutely uniformly on its \textit{interval of convergence} $(a - R, a + R)$.
\begin{theorem}[Differentiability of power series]
  Let $\sum_{n = 0}^{\infty} c_n (x - a)^n$ be a real power series with radius of convergence $R > 0$ and define: \label{diffPowerSeries}
  \[
    f: (a - R,  a + R) \to \R,\ x \mapsto \sum_{n = 0}^{\infty} c_n (x - a)^n
  \]
  We then have the following:
  \begin{enumerate}
    \item The term-wise differentiation $\sum_{n = 1}^{\infty} n c_n(x - a)^{n - 1}$ also has radius of convergence $R$.
    \item $f$ is differentiable on $(a - R, a + R)$ with derivative $f'(x) = \sum_{n = 1}^{\infty} n c_n(x - a)^{n - 1}$.
  \end{enumerate}
\end{theorem}
\begin{proof}[\textbf{i}]
  To find the radius of convergence of the term-wise differentiated series:
  \[
    \limsup_{n \to \infty} (n |c_n|)^{1/n} = \limsup_{n \to \infty} n^{1/n} \cdot \limsup_{n \to \infty} |c_n|^{1/n}
  \]
  Since $\lim_{n \to \infty} n^{1/n} = 1$, $\limsup_{n \to \infty} n^{1/n} = 1$ and so $\limsup_{n \to \infty} (nc_n)^{1/n} = R^{-1}$.
  Thus the radius of convergence of $\sum_{n = 1}^{\infty} n c_n (x - a)^{n - 1}$ is also $R$.
\end{proof}
\begin{proof}[\textbf{ii}]
  Define the partial sums:
  \[
    f_n(x) = \sum_{j = 0}^{n} c_j (x - a)^j
  \]
  We see that $f_n$ is differentiable with:
  \[
    f_n'(x) = \sum_{j = 1}^{n} j c_j (x - a)^{j - 1}
  \]
  Since the original power series has $R > 0$, by \textbf{(i)}, the term-wise differentiation also has $R > 0$ so by \cref{powerConv}, $f_n'$ converges locally uniformly on $(a - R, a + R)$.
  Fix some $r$ such that $0 < r < R$, and so $f_n'$ converges uniformly on $[a - r, a + r]$ with limit:
  \[
    g(x) = \sum_{n = 1}^{\infty} nc_n(x - a)^{n - 1}
  \]
  We also have that $(f_n(c))$ converges at some $c \in [a - r, a + r]$, since it converges at every point in $(a - R, a + R)$.

  Thus by we have that $f$ is differentiable with derivative
  \[
    f'(x) = g(x) =  \sum_{n = 1}^{\infty} n c_n (x - a)^{n - 1}
  \]
  on any $[a - r, a + r]$.

  Since $r$ was arbitrary in $0 < r < R$, $f$ is differentiable on $(a - R, a + R)$ with derivative as given.
\end{proof}
\begin{corollary}
  If the power series $\sum_{n = 0}^{\infty} c_n(x - a)^n$ has radius of convergence $R > 0$, then the function:
  \[
    f: (a - R, a + R) \to \R,\ x \mapsto \sum_{n = 0}^{\infty} c_n(x - a)^n
  \]
  is smooth, i.e. infinitely differentiable, on $(a - R, a + R)$.
  Further, its derivatives are given by:
  \[
    f^{(k)}: (a - R, a + R) \to \R,\ x \mapsto \sum_{n = k}^{\infty} n(n - 1) \cdots (n - k + 1) \cdot c_n (x - a)^{n - k}
  \]
\end{corollary}
\begin{proof}
  Repeatedly apply \cref{diffPowerSeries}, noting that the radius of convergence is unchanged after each step.
\end{proof}
Taking $x = a$, we see that:
\[
  f^{(k)}(a) = k(k - 1) \cdots (k - k + 1) c_k = k! \cdot c_k \implies c_k = \frac{f^{(k)}(a)}{k!}
\]
for any $k \in \N_0$.
So the whole function on $(a - R, a + R)$ is determined by its behaviour in an arbitrarily small interval of positive length around $a$.

Suppose we have another power series $g(x) = \sum_{n = 0}^{\infty} d_n (x - a)^n$ with radius of convergence $R_1 > 0$.
If $g \equiv f$ on any arbitrary small interval of positive length around $a$, then their derivatives at $a$ must agree.
Thus $d_n = c_n$ for all $n$, so $R_1 = R$ and $g \equiv f$ everywhere in $(a - R, a + R)$.
This is known as the \textit{unique continuation} property.
\begin{definition}
  We call functions which can be defined by a power series centered at $a$ with radii of convergence $R_a > 0$ at every point $a$ in their domain \textit{analytic functions}.
\end{definition}
\begin{remark}[Warning]
  It is \textbf{not} true that every smooth function is analytic.
  For example, on a IA Analysis I Sheet 4, we saw that the function:
  \[
    f(x) = \begin{cases}
    e^{-\frac{1}{x^2}} & \text{ if } x \neq 0\\
    0 & \text{ if } x =0
    \end{cases}
  \]
  is smooth with $f^{(k)}(0) = 0$ for all $k \in \N_0$.
  So if it had a power series about 0, all the coefficients would have to be 0 and so it could not be identical to $f$ on any interval of positive length around $0$.
\end{remark}
