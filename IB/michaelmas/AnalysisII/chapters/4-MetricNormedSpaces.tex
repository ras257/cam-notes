\chapter{Metric and Normed Spaces}
\section{Motivation}
Thus far in IA Analysis I and IB Analysis II, we have been doing analysis on $\R$.
To do this, we relied on the structure provided by $|\argdot|$ on $\R$ that allowed us to define convergence with unique limits.
We then looked at Cauchy sequences and we saw that all such sequences also converge due to the \textit{completeness} of $\R$.
We also saw that closed bounded intervals are ``special'' because of the Bolzano Weierstrass theorem.

We then moved onto functions $f: \R \to \R$, or more generally $f: E \subseteq \R \to \R$.
We then defined continuity and showed that closed bounded intervals are also ``special'' in this case because of the extreme value theorem.
Finally, in this course, we extended this to notions of uniform continuity.

These definitions and properties are quite useful and so we would like to extend these notions to functions $f: X \to \R$ or $f: X \to Y$.
But what such sets $X$ and $Y$ can we do this on, and how can we generalise the structures from $\R$?
\section{Metric Spaces}
\subsection{Definitions}
\begin{remark}[Note]
  In this course, we will assume that all vector spaces are over $\R$.
\end{remark}
To impose some structure on a set $X$, we can introduce a \textit{metric}.
More specifically, when $X$ is a vector space, we do this through the notion of a \textit{norm}.
\begin{definition}[Metric]
  A \textit{metric} on a set $X$ is a ``distance'' function $d: X \times X \to \R$ satisfying the following for any $x, y, z \in X$: \label{metricDef}
  \begin{enumerate}
    \item $d(x, y) \geq 0$ (i.e. distance is non-negative) with $d(x, y) = 0 \iff x = y$ (i.e. two points are the same if and only if the distance between them is 0)
    \item $d(x, y) = d(y, x)$ (i.e. the distance from $x$ to $y$ is the same as from $y$ to $x$)
    \item $d(x, y) \leq d(x, z) + d(z, y)$ (i.e. the triangle inequality)
  \end{enumerate}
  We then call $(X, d)$ a \textit{metric space}.
\end{definition}
\begin{definition}[Norm]
  A \textit{norm} on a (real) vector space $V$ is a function $\norm{\argdot}: V \to \R$ satisfying the following for any $x, y \in V$ and $\lambda \in \R$: \label{normDef}
  \begin{enumerate}
    \item $\norm{x} \geq 0$ with equality if and only if $x = \vec{0}$.
    \item $\norm{\lambda x} = |\lambda| \norm{x}$
    \item $\norm{x + y} \leq \norm{x} + \norm{y}$
  \end{enumerate}
  We then call $(V, \norm{\argdot})$ a \textit{normed vector space} or \textit{normed space}.
\end{definition}
\begin{proposition}
  If $(V, \norm{\argdot})$ is a normed space and we define: \label{normToMetric}
  \[
    d: V \times V \to \R,\ (x, y) \mapsto \norm{x - y}
  \]
  then $(V, d)$ is a metric space.
\end{proposition}
\begin{proof}
  Checking the conditions of \cref{metricDef}, we have:
  \begin{enumerate}
    \item By \cref{normDef} \textbf{(i)}, $d(x, y) = \norm{x - y} \geq 0$ with equality if and only if $x - y = \vec{0} \iff x = y$.
    \item By \cref{normDef} \textbf{(ii)}:
      \[
        d(x, y) = \norm{x - y} = \norm{-(y - x)} = |-1| \norm{y - x} = \norm{y - x} = d(x, y)
      \]
    \item By \cref{normDef} \textbf{(iii)}:
      \[
        d(x, y) = \norm{x - y} = \norm{x - z + z - y} \leq \norm{x - z} + \norm{z - y} = d(x, z) + d(z, y)
      \]
  \end{enumerate}
  So $d$ is indeed a metric on $V$.
\end{proof}
\subsection{Examples of Norms} \label{normExamples}
\begin{example}[Finite dimensional vector space]
  The prototypical example of a finite dimensional normed space is $\R^n$ equipped with various norms.

  Recall that:
  \[
    \R^n = \{(x_1, \ldots, x_n): x_j \in \R\ \forall j \in \{1, \ldots, n\}\}
  \]
  is a vector space under the usual component-wise addition and scalar multiplication.

  We can introduce the following norms:
  \begin{itemize}
    \item The \textit{Euclidean} or $\ell^2$-norm defined as:
      \[
        \norm{x}_{E} = \left[\sum_{i = 1}^{n} (x_i)^2\right]^{\frac{1}{2}}
      \]
      This is sometimes denoted $\norm{\argdot}_2$ or $\norm{\argdot}_{\ell^2}$.

      This is a norm on $\R^n$, \textbf{(i)} and \textbf{(ii)} are trivial to check, for \textbf{(iii)}, consider $x = (x_1, \ldots, x_n)$ and $y = (y_1, \ldots, y_n)$:
      \begin{align*}
        \norm{x + y}^2_E &= \sum_{i = 1}^{n} (x_i + y_i)^2 \\
                         &= \norm{x}^2_E + 2 x \cdot y+ \norm{y}^2_E \\
                         &\leq \norm{x}^2_E + 2 \norm{x}^2_E \norm{y}^2_E + \norm{y}^2_E \text{ by Cauchy Schwarz} \\
                         &= (\norm{x}^2_E + \norm{y}^2_E)^2 \implies \norm{x + y}_E \leq \norm{x}_E + \norm{y}_E
      \end{align*}
    \item The $\ell^1$-norm on $\R^n$ defined as:
      \[
        \norm{x}_{\ell^1} = \sum_{i = 1}^{n} |x_i|
      \]
      This is sometimes denoted $\norm{\argdot}_1$.
      Conditions \textbf{(i)} and \textbf{(ii)} are trivial and \textbf{(iii)} follows from the triangle inequality on $|\argdot|$.
    \item The $\ell^{\infty}$-norm on $\R^n$ defined as:
      \[
        \norm{x}_{\ell^{\infty}} = \sup \{|x_i| : i \in \{1, \ldots, n\}\}
      \]
      This is sometimes denoted $\norm{\argdot}_{\infty}$.
      Conditions \textbf{(i)} and \textbf{(ii)} follow from the properties of $|\argdot|$ on $\R$.
      For \textbf{(iii)}, we have:
      \begin{align*}
        \norm{x + y}_{\infty} &= \sup_{i \in \{1, \ldots, n\}} [|x_i + y_i|] \\
                              &\leq \sup_{i \in \{1, \ldots, n\}} [|x_i| + |y_i|] \text{ using the triangle inequality} \\
                              &\leq \sup_{i \in \{1, \ldots, n\}} |x_i| + \sup_{i \in \{1, \ldots, n\}} |y_i| \\
                              &= \norm{x}_{\infty} + \norm{y}_{\infty}
      \end{align*}
  \end{itemize}
\end{example}
\begin{remark}[Recap]
  Suppose that $V$ is an $\F$-vector space with subset $W \subseteq V$.
  Recall from IB Linear Algebra that the subspace tells us that if $\vec{0} \in W$ and for all $w_1, w_2 \in W$ and $\lambda \in \F$ we have $w_1 + \lambda w_2 \in W$, then $W$ is a subspace of $V$, denoted $W \leq V$.
\end{remark}
\begin{example}[Infinite dimensional vector space] \label{infNorms}
  Consider the vector space of real sequences, denoted $\R^{\N}$:
  \[
    \R^{\N} = \{\text{sequence $(x_k)_{k \in \N}$ with $x_k \in \R\ \forall k \in \N$}\}
  \]
  This forms a vector space with element-wise addition and scalar multiplication.
  In this case, we will define subspaces of $\R^\N$ that have a particular property and then use this property to define a norm on the subspace.
  \begin{itemize}
    \item Define the subset $\ell^1$ of $\R^{\N}$ as:
      \[
        \ell^1 = \left\{(x_k) \in \R^{\N}: \sum_{k = 1}^{\infty} |x_k| < \infty\right\}
      \]
      Clearly $\vec{0} \in \ell^1$, and since $|x_i + \lambda y_i| \leq |x_i| + |\lambda| |y_i|$, if $x, y \in \ell_1$ and $\lambda \in \R$, we have $x + \lambda y \in \ell^1$.
      Thus by the subspace test, $\ell^1 \leq \R^{\N}$.

      We then define the $\ell^1$-norm on the $\ell^1$ subspace as:
      \[
        \norm{x}_{1} = \sum_{i = 1}^{\infty} |x_i| \text{ for $x \in \ell^1$}
      \]
      The conditions for $\norm{\argdot}_1$ to be a norm are all inherited from $|\argdot|$ on $\R$.
    \item We can similarly define the subset $\ell^2$ as:
      \[
        \ell^2 = \left\{(x_k) \in \R^{\N}: \sum_{k = 1}^{\infty} x_k^2 < \infty\right\}
      \]
      Again, clearly $\vec{0} \in \ell^2$.
      For any $x, y \in \ell^2$ and $\lambda \in \R$, we see that:
      \[
        0 \leq (x_i - \lambda y_i)^2 = x^2_i + \lambda^2 y^2_i - 2\lambda x_i y_i \implies 2 \lambda x_i y_i \leq x^2_i + \lambda^2 y^2_i
      \]
      and so:
      \[
        (x_i + \lambda y_i)^2 = x^2_i + \lambda^2 y^2_i + 2\lambda x_i y_i \leq 2(x^2_i + \lambda^2 y^2_i)
      \]
      Thus $x + \lambda y \in \ell^2$ and so by the subspace test, we have $\ell^2 \leq \R^{\N}$.

      We then define the $\ell^2$-norm on the $\ell^2$ subspace as:
      \[
        \norm{x}_{2} = \left[\sum_{i = 1}^{\infty} x^2_i\right]^{\frac{1}{2}} \text{ for $x \in \ell^2$}
      \]
      Clearly conditions \textbf{(i)} and \textbf{(ii)} hold.
      For \textbf{(iii)}, we see that:
      \[
        \norm{x}_2 = \lim_{n \to \infty} \norm{(x_1, \ldots, x_n)}_{\ell^2}
      \]
      and so the triangle inequality for $\norm{\argdot}_2$ follows from the triangle inequality of $\norm{\argdot}_{\ell^2}$ on $\R^n$.
    \item We also define the subset $\ell^{\infty}$ as:
      \[
        \ell_{\infty} = \left\{(x_k) \in \R^{\N}: \sup_{k \in \N} |x_k| < \infty\right\}
      \]
      Again, clearly $\vec{0} \in \ell^{\infty}$.
      For any $x, y \in \ell^{\infty}$ and $\lambda \in \R$:
      \[
        \sup_{k \in \N} |x_k + \lambda y_k| \leq \sup_{k \in \N} |x_k| + |\lambda| \sup_{k \in \N} |y_k| < \infty
      \]
      so $x + \lambda y \in \ell^{\infty}$.
      Thus by the subspace test, we have $\ell^{\infty} \leq \R^{\N}$.

      We then define the $\ell^{\infty}$-norm on the $\ell^{\infty}$ space as:
      \[
        \norm{x}_{\infty} = \sup_{k \in \N} |x_k|
      \]
      Conditions \textbf{(i)} and \textbf{(ii)} are inherited from $|\argdot|$ on $\R$.
      For condition \textbf{(iii)}, we have:
      \[
        \norm{x}_{\infty} = \lim_{n \to \infty} \norm{(x_1, \ldots, x_n)}_{\ell^{\infty}}
      \]
      and so the triangle inequality for $\norm{\argdot}_{\infty}$ follows from the triangle inequality of $\norm{\argdot}_{\ell^{\infty}}$ on $\R^n$.
  \end{itemize}
\end{example}
\begin{remark}[Remarks]
  \begin{enumerate}
    \item We can analogously define the $\ell^p$ subspace of $\R^\N$ for $p \geq 1$ as:
      \[
        \ell^p = \left\{x \in \R^\N: \sum_{k = 1}^{\infty} |x_k|^p < \infty\right\}
      \]
      This is a subspace of $\R^\N$ and $(\ell^p, \norm{\argdot}_p)$ is a normed space where the $\ell^p$-norm is defined as:
      \[
        \norm{x}_{p}  = \left[\sum_{k = 1}^{\infty} |x_k|^p\right]^\frac{1}{p}
      \]
      We can also similarly define an $\ell^p$ norm on $\R^n$.
    \item In fact, $\lim_{p \to \infty} \norm{x}_{p} = \norm{x}_{\infty}$.
    \item Further $\ell^p \subset \ell^q$ for $p < q$.
  \end{enumerate}
\end{remark}
\begin{example}[On continuous functions]
  Consider the vector space of continuous functions:
  \[
    V = C([a, b]) = \{f: [a, b] \to \R: f \text{ is continuous}\}
  \]
  We can define the following norms on this space:
  \begin{enumerate}
    \item The $L^1$-norm on $C([a, b])$ defined by:
      \[
        \norm{f}_{L^1([a, b])} = \int_{a}^{b} |f(x)| \d{x}
      \]
      This is also denoted $\norm{\argdot}_{L^1}$ or $\norm{f}_1$ if the domain is obvious.

      Since $f \in C([a, b])$, we get $\int_{a}^{b} |f(x)| \d{x} = 0 \iff f \equiv 0$ so property \textbf{(i)} is satisfied.
      Properties \textbf{(ii)} and \textbf{(iii)} follow from the triangle inequality of $|\argdot|$ on $\R$ and properties of integrals from IA Analysis 1.
    \item The $L^2$-norm on $C([a, b])$ defined by:
      \[
        \norm{f}_{L^2([a, b])} = \left[\int_{a}^{b} f(x)^2 \d{x}\right]^{\frac{1}{2}}
      \]
      This is also denoted $\norm{\argdot}_{L^2}$ or $\norm{f}_2$ if the domain is obvious.
      We saw in IA Analysis I that if $f$ is integrable, so is $f^2$ so the above is indeed well defined.

      For this norm, the triangle inequality follows from the integral version of Cauchy-Schwarz:
      \[
        \int_{a}^{b} |fg| \d{x} \leq \left(\int_{a}^{b} |f|^2 \d{x}\right)^{\frac{1}{2}} \left(\int_{a}^{b} |g|^2 \d{x}\right)^{\frac{1}{2}}
      \]
      which follows from applying Cauchy Schwarz to the inner product on $C([a, b])$ defined as:
      \[
        \langle f, g \rangle = \int_{a}^{b} f(x)g(x) \d{x}
      \]
    \item The $L^\infty$-norm on $C([a, b])$, or \textit{uniform norm}, defined by:
      \[
        \norm{f}_{L^\infty([a, b])} = \sup_{x \in [a, b]} |f(x)|
      \]
      This is also denoted $\norm{\argdot}_{L^\infty}$ or $\norm{f}_{\infty}$ if the domain is obvious.
  \end{enumerate}
\end{example}
\begin{remark}[Aside]
  \nonexaminable
  The ``natural'' space for the integral norms $L^1, L^2, \ldots$ to be defined on is $\mathscr{R}([a, b])$, the set of Riemann integrable $f: [a, b] \to \R$.
  Using properties of integrability IA Analysis I, we see that $\mathscr{R}([a, b])$ this forms a vector space under addition and scalar multiplications of functions.

  However, there is an immediate concern if we do this:
  \[
    f \in \mathscr{R}([a, b]) \text{ and } \norm{f}_1 = \int_{a}^{b} |f(x)| \d{x} = 0 \centernot\implies f \equiv 0 \text{ on } [a. b]
  \]
  For example, Thomae's function has a zero integral but is not identically zero.

  We can however rectify this issue without needing to restrict to just $C([a, b])$.
  This is because:
  \[
    \int_{a}^{b} |f| \d{x} = 0 \iff f = 0 \text{ almost every where in $[a, b]$}
  \]
  By \textit{almost everywhere} (abbreviated \textit{a.e.}), we mean that $f = 0$ except on a null set.
  This follows from Lebesgue's theorem (\cref{lebesgueInt}).

  We know that since if $f$ is integrable, so is $|f|$, thus the discontinuities of $|f|$ have measure 0.
  So if $\int_{a}^{b} |f| \d{x} = 0$, then we must have that $f = 0$ at every continuity point of $f$ as if this was not the case, then there would be an interval on which $|f| > 0$, contradicting $I(|f|) = 0$.
  Thus the set of points on which $f \neq 0$ is null.
  Conversely, if $f = 0$ almost everywhere, then $L(\mathcal{P}, |f|) = 0$ for any partition $\mathcal{P}$ and so $I(|f|) = 0$.

  We can now define an equivalence relation $f \sim g$ on $\mathscr{R}([a, b])$ by:
  \[
    f \sim g \iff f = g \text{ almost everywhere on } [a, b]
  \]
  If we then quotient out by this relation, we obtain the set of equivalence classes $\widetilde{\mathscr{R}}([a, b]) = \mathscr{R}([a, b]) / \sim$.
  Suppose we define the operations on this space as:
  \[
    [f] + [g] = [f + g] \text{ and } \lambda [f] = [\lambda f]
  \]
  where $[f]$ is the equivalence class of $f$ under $\sim$,
  These are well defined as $[f_1] = [f_2]$ and $[g_1] = [g_2]$, we have $f_1 = f_2$ a.e. and $g_1 = g_2$ a.e. so $f_1 + g_1 = f_2 + g_2$ a.e., thus $[f_1 + g_1] = [f_2 + g_2]$ so this operation is well defined.
  Similarly, if $[f_1] = [f_2]$, we have $f_1 = f_2$ a.e. and so $\lambda f_1 = \lambda f_2$ a.e., thus $[\lambda f_1] = [\lambda f_2]$.

  Further, if $f \sim g$, then $|f| - |g| = 0$ a.e. so by previous discussion, $\int_{a}^{b} |f| \d{x} - \int_{a}^{b} |g| \d{x} = 0$, thus the norm defined as:
  \[
    \norm{[f]}_{L^1([a, b])} = \int_{a}^{b} |f| \d{x}
  \]
  is a well defined norm on $\widetilde{\mathscr{R}}([a, b])$.
\end{remark}
\subsection{Examples of Metrics}
\begin{remark}
  By \cref{normToMetric}, all the examples from \cref{normExamples} are metric spaces with respect to the natural metric induced by the norm.

  For example, the \textit{Euclidean metric} on $\R^n$ is:
  \[
    d_{E}(x, y) = \norm{x - y}_{E} = \left[\sum_{i = 1}^{n} (x_i - y_i)^2\right]^{\frac{1}{2}}
  \]
  or the \textit{uniform metric} on $C([a, b])$:
  \begin{equation}
    d_{\infty}(f, g) = \norm{f - g}_{\infty} = \sup_{x \in [a, b]} |f(x) - g(x)| \label{unifMetric}
  \end{equation}
  This is the notion of ``distance'' between functions we alluded to when introducing uniform convergence.
\end{remark}
\begin{example}[Discrete metric]
  For any set $X$ the \textit{discrete metric} is defined as:
  \[
    f: X \times X \to \R,\ (x, y) \mapsto \begin{cases}
    0 & x = y \\
    1 & x \neq y
    \end{cases}
  \]
  This is a metric since it satisfies all of the properties from \cref{metricDef}:
  \begin{enumerate}
    \item $d(x, y) = 0 \iff x = y$
    \item $d(x, y) = d(y, x)$ by definition
    \item If $x = y$, then $d(x, y) = 0$ so we must have $d(x, y) \leq d(x, z) + d(z, y)$ vacuously.
      If $x \neq y$, then $d(x, y) = 1$, we cannot have both $x = z$ and $y = z$ so $d(x, z) + d(z, y) \geq 1$.
  \end{enumerate}
\end{example}
\begin{example}[British/French Railways metric] \label{railwayMetric}
  On $\R^2$, the \textit{British railways} or \textit{French railways} metric is defined as:
  \[
    d(x, y) = \begin{cases}
    \norm{x - y}_2 & \text{ if } x = ty \text{ for some } t \in \R\\
    \norm{x}_2 + \norm{y}_2 & \text{ otherwise }
    \end{cases}
  \]
  where $\norm{\argdot}_2$ is the Euclidean norm on $\R^2$.
\end{example}
\begin{example}[New metrics from a given metric]
  Suppose that we have a metric space $(X, d)$, we can define the following new metrics from this metric:
  \begin{enumerate}
    \item $g: X \times X \to \R,\ (x, y) \mapsto \min\{1, d(x, y)\}$
    \item $h: X \times X \to \R,\ (x, y) \mapsto \frac{d(x, y)}{1 + d(x, y)}$
  \end{enumerate}
  These metrics are both bounded, more specifically $g(x, y) \leq 1$ and $h(x, y) < 1$ for any $x, y \in X$.
\end{example}
\section{Open and Closed Subspaces}
\begin{definition}[Metric subpaces]
  If $(X, d)$ is a metric space and $Y \subseteq X$, then the restriction:
  \[
    \at{d}{Y\times Y}: Y \times Y \to \R
  \]
  is a metric on $Y$.
  This is called the \textit{subspace metric} or \textit{induced metric} on $Y$.
\end{definition}
\begin{definition}[Open ball]
  For a metric space $(X, d)$ and a point $a \in X$, the \textit{open ball with radius $r$ centered at $a$} is defined to be the set:
  \[
    B_r(a) = \{x \in X: d(x, a) < r\}
  \]
\end{definition}
\begin{definition}[Open subset]
  We say that a subset $U \subseteq X$  is \textit{open} if for every point $a \in U$ there exists $r > 0$  such that $B_r(a) \subseteq U$.
\end{definition}
\begin{definition}[Closed subset]
  We say that a subset $F \subseteq X$  is \textit{closed} if the subset $X \setminus F$ is open
\end{definition}
\begin{remark}[Warning]
  \begin{enumerate}
    \item The definition of open and closed subsets are \textbf{not} negations of one another.
      For example both $X$ and $\emptyset$ are open and closed.
    \item The notion of a set being open or closed depends on the choice of metric, some sets will be open under some metrics but not under others.
  \end{enumerate}
\end{remark}
\begin{lemma}
  If $(X, d)$ is a metric space, then $X$ and $\emptyset$ are both open and closed. \label{emptyClosed}
\end{lemma}
\begin{proof}
  $\emptyset$ is vacuously open as there are no point in it to check and the set $X$ is vacuously open as $B_r(x) \subseteq X$ for any $x \in X$.

  We then see that $X \setminus X = \emptyset$, which is open so $X$ is closed.
  Similarly, $X \setminus \emptyset = X$, which is open so $\emptyset$ is closed.
\end{proof}
When considering metrics which are \textit{norm-induced}, shifting the centre of the ball translates the ball.
Suppose we have the open balls $B_r(a)$ and $B_r(b)$, since the metric is norm induced it acts on a vector space and so:
\[
  x \in B_r(a) \iff \norm{x - a} < r \iff \norm{[x + (b - a)] - b} < r \iff x + (b - a) \in B_r(b)
\]
so the ball $B_r(b)$ is a translation of the ball $B_r(a)$ by the vector $b - a$.

However metrics which are not norm induced will need not behave as nicely and can look vastly different depending on the centre point.
\begin{example}[Open balls under the railway metric]
  Consider the metric defined in \cref{railwayMetric} on $\R^2$:
  \[
    d(x, y) = \begin{cases}
    \norm{x - y}_2 & \text{ if } x = ty \text{ for some } t \in \R\\
    \norm{x}_2 + \norm{y}_2 & \text{ otherwise }
    \end{cases}
  \]
  The shape of the open ball $B_{\frac{1}{2}}(a)$ varies depending on where the point $a$ is:
  \begin{center}
  \begin{tikzpicture}[>=stealth]
    \draw[dashed, thick] (0, 0) circle (1);
    \fill[gray!40] (0, 0) circle (1);
    \fill (0, 0) circle (1pt);
    \node[below left] at (2, 2) {$B_{\frac{1}{2}}(0, 0)$};

    \draw[->] (-2, 0) -- (2, 0);
    \draw[->] (0, -2) -- (0, 2);

    \begin{scope}[xshift=5cm]
      \draw (0.5, 0.5) -- (1.5, 1.5);
      \fill (1, 1) circle (1pt);

      \draw (0.5, 0.5) circle (1pt);
      \draw (1.5, 1.5) circle (1pt);
      \node[below left] at (2, 2) {$B_{\frac{1}{2}}(1, 1)$};

      \draw[->] (-2, 0) -- (2, 0);
      \draw[->] (0, -2) -- (0, 2);
    \end{scope}
  \end{tikzpicture}
  \end{center}
  In the case of $a = (0, 0)$, every point lies on a ray through the origin and so we just get the ball $\norm{x}_2 < \frac{1}{2}$.
  However, in the case of $a = (1, 1)$, we have $\norm{a}_2 > \frac{1}{2}$ and so if $x$ does not lie on a ray through $a$ and the origin, $d(x, a) > \frac{1}{2}$ and so all points in the ball must be on the ray connecting $a$ and the origin.
\end{example}
It is also important to be aware that sets may be open when considering a metric on a whole set but not open  when considering an induced metric on a subset.
\begin{example}[Subspace metric] \label{subspaceMetric}
  Consider $X = \R$ with the Euclidean metric and let $Y = [0, 1) \cup \{2\}$ be a subset of $X$ equipped with the subspace metric.
  That is, we consider the metric spaces $(X, d_2)$ and $(X, \at{d_2}{Y \times Y})$
  \begin{enumerate}
    \item $[0, 1)$ is both open and closed in $Y$.\par
      \textbf{Open - } At any point $x \in [0, 1)$, $B^{Y}_{\frac{1}{2}}(x) \subseteq [0, 1)$ as it is not possible for this ball to capture the only other point $2$ in $Y$.

      \textbf{Closed - } The complement of $[0, 1)$ in $Y$ is $\{2\}$.
        This is open due to the ball $B^{Y}_{\frac{1}{2}}(2) = \{2\}$.
    \item $\{2\}$ is both open and closed in $Y$.
      We have already showed that $[0, 1)$ is both open and closed in $Y$.
      This is the complement of $\{2\}$ in $Y$ and so $\{2\}$ is both open and closed also.
    \item $[0, 1)$ is neither open nor closed in $X$.\par
      \textbf{Not open -} The ball $B_r(0)$ for $r > 0$ always contains a point in $X$ that is not in $[0, 1)$.

      \textbf{Not closed -} The complement of $[0, 1)$ in $X$ is $(-\infty, 0) \cup [1, \infty)$.
        The ball $B_r(0)$ always contains a point in $X$ that is not in either of these sets.
    \item $\{2\}$ is closed in $X$ but is not open.\par
      \textbf{Not open - } The ball $B_r(2)$ for $r > 0$ always contains a point in $X$ that is not $2$.

      \textbf{Closed - } By \cref{singletonClosed}.
  \end{enumerate}
\end{example}
\begin{proposition}
  If $(X, d)$ is a metric space, then: \label{singletonClosed}
  \begin{enumerate}
    \item For any $a \in X$ and $r > 0$, the open ball $B_r(a)$ is an open set in $X$.
    \item Any singleton $\{x\}$ is a closed set in $X$.
  \end{enumerate}
\end{proposition}
\begin{proof}[\textbf{i}]
  Suppose $y \in B_r(a)$, then $d(y, a) < r$ and so we let $r_1 = r - d(y, a) > 0$.
  Consider the ball of radius $r_1$ centered at $y$, if $x \in B_{r_1}(y)$, then:
  \[
    d(x, a) \leq d(x, y) + d(y, a) < r_1 + d(y, a) = r
  \]
  So $x \in B_r(a)$, thus $B_{r_1}(y) \subseteq B_r(a)$.
\end{proof}
\begin{proof}[\textbf{ii}]
  For any $y \in X \setminus \{x\}$, we must have $y \neq x$ and so we let $r = d(y, x) > 0$.
  Then $B_r(y) \subseteq X \setminus \{x\}$ as it cannot contain $x$.
  Thus $X \setminus \{x\}$ is open, so by definition $\{x\}$ is closed.
\end{proof}
\begin{remark}[Warning]
  Open balls can be closed sets an singletons can be open.
  We already saw an example of an open singleton in \cref{subspaceMetric}.

  Consider the discrete metric on some set $X$:
  \[
    d(x, y) = \begin{cases}
    0 & \text{ if } x = y \\
    1 & \text{ if } x \neq y
    \end{cases}
  \]
  Then $B_{\frac{1}{2}}(x) = \{x\}$ which is closed by \cref{singletonClosed}, so we have an example of an open ball which is an empty set.
\end{remark}
\begin{theorem}[Unions and intersections of open sets]
  If $(X, d)$ is a metric space, then: \label{unionProp}
  \begin{enumerate}
    \item The union of any (possibly uncountable) collection of open sets is an open set.
    \item The intersection of a finite collection of open sets is an open set.
  \end{enumerate}
\end{theorem}
\begin{proof}[\textbf{i}]
  Suppose $(U_i)_{i \in I}$ is a collection of open sets indexed by the possibly uncountable set $I$.
  Define their union as the set $U = \bigcup_{i \in I} U_i$.

  Take any $x \in U$, then $x \in U_i$ for some $i \in I$ so there exists $r > 0$ such that $B_r(x) \subseteq U_i$.
  By construction $U_i \subseteq U$ and so $B_r(x) \subseteq U$.
  Thus $U$ is an open set.
\end{proof}
\begin{proof}[\textbf{ii}]
  Suppose the sets $U$ and $V$ are open and consider $W = U \cap V$.
  If $x \in W$, then $x \in U$ and $x \in V$ so there exists $r_1, r_2 > 0$ such that $B_{r_1}(x) \subseteq U$ and $B_{r_2}(x) \subseteq V$.
  Let $r = \min\{r_1, r_2\} > 0$, then:
  \[
    y \in B_r(x) \implies d(y, x) < r \leq r_1, r_2 \implies y \in B_{r_1}(x) \cap B_{r_2}(x) \subseteq U \cap V = W
  \]
  So $B_r(x) \subseteq W$, thus $W$ is also open.

  The empty set is open by \cref{emptyClosed} and so by induction on the number of sets, any finite intersection of open sets is an open set.
\end{proof}
\begin{remark}
  The finiteness in \textbf{(ii)} is important.
  For example with $X = \R$ and the Euclidean metric:
  \[
    \bigcap_{n = 1}^{\infty} \left(-\frac{1}{n}, \frac{1}{n}\right) = \{0\}
  \]
  which is not open in $X$.

  The above proof breaks down as given the point $0$, the infimum (instead of the minimum) of the radii $r_n$ from each of the open sets $(-\frac{1}{n}, \frac{1}{n})$ is 0.
\end{remark}
\begin{theorem}[Unions and intersections of closed sets]
  If $(X, d)$ is a metric space, then: \label{interProp}
  \begin{enumerate}
    \item The intersection of any (possibly uncountable) collection of closed sets is a closed set.
    \item The union of a finite collection of closed sets is a closed set.
  \end{enumerate}
\end{theorem}
\begin{proof}[\textbf{i}]
  Suppose $(U_i)_{i \in I}$ is a collection of closed sets indexed by the possibly uncountable set $I$.
  Define their intersection as the set $U = \bigcap_{i \in I} U_i$.
  The complement of this set in $X$ is:
  \[
    U \setminus X = U \setminus \left[\bigcap_{i \in I} U_i\right] = \bigcup_{i \in I} (X \setminus U_i)
  \]
  Since $U_i$ is closed, its complement $X \setminus U_i$ is open, so $U \setminus X$ is a finite intersection of open sets.
  Thus by \cref{unionProp} \textbf{(ii)}, $U \setminus X$ is open, i.e. $U$ is closed.
\end{proof}
\begin{proof}[\textbf{ii}]
  Suppose $(U_i)_{i \in I}$ is a collection of closed sets indexed by the finite set $I$.
  Define their union as $U = \bigcup_{i \in I} U_i$.
  The complement of this set in $X$ is:
  \[
    U \setminus X = U \setminus \left[\bigcup_{i \in I} U_i\right] = \bigcap_{i \in I} (X \setminus U_i)
  \]
  Since $U_i$ is closed, its complement $X \setminus U_i$ is open, so $U \setminus X$ is a union of a collection of open sets.
  Thus by \cref{unionProp}, $U \setminus X$ is open, i.e. $U$ is closed.
\end{proof}
\begin{corollary}[Closure of finite sets]
  If $(X, d)$ is a metric space, then any finite subset of $X$ is closed. \label{finiteClosed}
\end{corollary}
\begin{proof}
  From \cref{singletonClosed}, for any $x \in X$ singleton $\{x\}$ is closed.
  Thus if $U$ is a finite subset of $X$, then we can write it as a finite union of singletons, so by \cref{interProp}, $U$ is closed.
\end{proof}
\section{Convergence in Metric Spaces}
\begin{definition}[Convergence in a metric space]
  Let $(X, d)$ be a metric space.
  We say that a sequence $(x_k)$ in $X$ \textit{converges} to a point $x \in X$ if:
  \[
    d(x_k, x) \to 0 \text{ as $k \to \infty$}
  \]
  as a limit on $\R$.
\end{definition}
\begin{remark}
  We can unpack the above using the definition of a limit:
  \[
    \forall \varepsilon > 0\ \exists N \in \N \text{ s.t. } d(x_k, x) < \varepsilon\ \forall k \geq N
  \]
  and can also define it using open balls:
  \[
    \forall \varepsilon > 0\ \exists N \in \N \text{ s.t. } x_k \in B_{\varepsilon}(x)\ \forall k \geq N
  \]
\end{remark}
\begin{proposition}[Uniqueness of limits]
  If $(X, d)$ is a metric space and $(x_k)$ is a sequence on $X$ converging to both $x$ and $y$, then $x = y$.
\end{proposition}
\begin{proof}
  For any $\varepsilon > 0$, since $d(x, x_k), d(y, x_k) \to 0$, for sufficiently large $k$:
  \[
    d(x, y) \leq d(x, x_k) + d(x_k, y) < \varepsilon
  \]
  Since $\varepsilon > 0$ was arbitrary, $d(x, y) = 0$ and so $x = y$.
\end{proof}
\begin{proposition}[Convergence in $\R^n$ with respect to the Euclidean metric]
  Convergence in $(\R^n, \norm{\argdot}_2)$ is equivalent to coordinate-wise convergence.
  That is, if $(x^{(k)})_{k \in \N} = (x^{(k)}_1, \ldots, x^{(k)}_n) \in \R^{n}$, then: \label{vectorConv}
  \[
    x^{(k)} \xrightarrow{\norm{\argdot}_2} x = (x_1, \ldots, x_n) \iff x^{(k)}_j \to x_j\ \forall j \in \{1, \ldots, n\}
  \]
\end{proposition}
\begin{proof}
\begin{proofdirection}{Assume $x^{(k)}$ converges}
  So if we fix $\varepsilon > 0$, then there exists $K$ such that $\norm{x^{(k)} - x}_2 < \varepsilon$ for all $k \geq K$.
  Thus, for any $j \in \{1, \ldots, n\}$:
  \begin{align*}
    \abs{x^{(k)}_j - x_j} &= \left[(x^{(k)}_j - x_j)^2\right]^{\frac{1}{2}} \\
                          &\leq \left[(x^{(k)}_1 - x_1)^2 + \cdots + (x^{(k)}_n - x_n)^2\right]^{\frac{1}{2}} \\
                          &= \norm{x^{(k)} - x}_{2} \\
                          &< \varepsilon\ \forall k \geq K
  \end{align*}
  and so $x^{(k)}_j \to x_j$ as $k \to \infty$ for each $j \in \{1, \ldots, n\}$.
\end{proofdirection}
\begin{proofdirection}{Assume each $x^{(k)}_j$ converges}
  So for each $j \in \{1, \ldots, n\}$, there exists $K_j$ such that $|x^{(k)}_j - x_j| < \frac{\varepsilon}{\sqrt{n}}$ for all $k \geq K_j$.
  Since there are finitely many of these, we can take $K = \max\{K_1, \ldots, K_n\}$, so that:
  \begin{align*}
    \norm{x^{(k)} - x}_2 &= \left[\sum_{i = 1}^{n} (x^{(k)}_i - x_i)^2\right]^{\frac{1}{2}} \\
                         &< \left[\sum_{i = 1}^{n} \left(\frac{\varepsilon}{\sqrt{n}}\right)^2\right] \\
                         &= \varepsilon
  \end{align*}
  and so $x^{(k)} \to x$ under $\norm{\argdot}_2$ as $k \to \infty$.
\end{proofdirection}
\end{proof}
\begin{remark}
  We will later show that all norms on $\R^{n}$ (or indeed any other finite dimensional vector space) are, in  sense, equivalent to each other.
  So the choice of $\norm{\argdot}_2$ is not special in any way, i.e. the result extends to any finite dimensional vector space equipped with any norm induced metric.
\end{remark}
\begin{definition}[Bounded set]
  Let $(X, d)$ be a metric space.
  We say that a subset $E \subseteq X$ is \textit{bounded} if there exists $R > 0$ and $a \in X$ such that $E \subseteq B_R(a)$.
\end{definition}
\begin{remark}[Intuition]
  A subset $E$ is closed if there exists a ball in which $E$ is entirely contained in.
\end{remark}
\begin{remark}
  Let $E \subseteq X$ is bounded and let $b \in X$ be arbitrary.
  By definition, there exists $a \in X$ and $R > 0$ such that $E \subseteq B_R(a)$.
  We see that $E \subseteq B_{R + d(a, b)}(b)$ and so a ball centered at $b$ with radius $R + d(a, b) > 0$ also contains $E$.
  Thus if $E$ is bounded, then for any point $b \in X$, we can find a ball of radius $R > 0$ such that $E \subseteq B_{R}(b)$.
\end{remark}
\begin{corollary}[Subsequences of a convergent sequence on $\R^n$]
  If $x^{(k)} \in \R^{n}$ is convergent with $x^{(k)} \to x \in \R^{n}$ under $\norm{\argdot}_2$ as $k \to \infty$, then any subsequence $x^{(k_\ell)}$ also has $x^{(k_{\ell})} \to x$ under $\norm{\argdot}_2$ as $\ell \to \infty$. \label{subseqConv}
\end{corollary}
\begin{proof}
  By \cref{vectorConv}, each component has $x^{(k)}_j \to x_j$ as $k \to \infty$.
In IA Analysis I, we saw that any subsequence of a convergent sequence in $\R$ also converges to the same limit as the original sequence, thus $x^{(k_{\ell})}_j \to x_j$ as $\ell \to \infty$.
So applying the other \cref{vectorConv}, we have $x^{(k_{\ell})} \to x$ as $\ell \to \infty$ under $\norm{\argdot}_2$.
\end{proof}
\begin{definition}[Bolzano-Weierstrass in $\R^n$]
  Any bounded sequence in $\R^n$ has a convergent subsequence.
\end{definition}
We will assume the metric is induced by the Euclidean norm but as in the previous remark, the result also holds with respect to any norm induced metric on a finite dimensional vector space.
\begin{proof}
  Suppose $(x^{(k)})_{k \in \N}$ is bounded in $\R^n$, by the above remark, we can assume the ball is centered at the origin and so there exists $R > 0$ such that $\norm{x^{k}}_2 < R$ for all $k \in \N$.

  We proceed by induction on the dimension of the space $n$.
  \quietinduction
  {$n = 1$}{
    This is just the usual Bolzano-Weierstrass on $\R$ from IA Analysis I.
  }
  {$n - 1$}{}
  {$n$}{
    Define $y^{(k)} = (x^{(k)}_1, \ldots, x^{(k)}_{n - 1}) \in \R^{n - 1}$ so that $x^{(k)} = (y^{(k)}, x^{(k)}_n) \in \R^{n}$.
    By definition of $\norm{\argdot}_2$, we have:
    \[
      \norm{y^{(k)}}^2_2 + \abs{x^{(k)}_n}^2 = \norm{x^{(k)}}^2_2 < R^2 \implies \norm{y^{(k)}}_2 < R \text{ and } \abs{x^{(k)}_n} < R
    \]
    So $(y^{(k)})_{k \in \N}$ is bounded in $\R^{n - 1}$ and $(x^{(k)}_n)_{k \in \N}$ is bounded in $\R$.

    Thus by the induction hypothesis, there exists a subsequence $(y^{(k_j)})_{j \in \N}$ of $(y^{(k)})$ and a point $y \in \R^{n - 1}$ such that $y^{(k_j)} \to y$ under $\norm{\argdot}_2$ on $\R^{n - 1}$ as $j \to \infty$.
    Clearly the subsequence $x^{(k_j)}_n$ is still bounded, so the Bolzano-Weierstrass theorem on $\R$ tells us that there exists a subsequence $(x^{(k_{j_{\ell}})}_n)_{\ell \in \N}$ with $x^{(k_{j_{\ell}})}_n \to x_n$ for some $x_n \in \R$ as $\ell \to \infty$.
    Since $(y^{(k_j)})_{j \in \N}$ converges to $y$, by \cref{subseqConv}, we also have $y^{k_{j_\ell}} \to y$ under $\norm{\argdot}_2$ as $\ell \to \infty$.

    Thus since each component of $x^{(k_{j_\ell})}$ converges as $\ell \to \infty$, appealing to \cref{vectorConv}, we have $x^{(k_{j_{\ell}})} \to x = (y, x_n) \in \R^{n}$ under $\norm{\argdot}_2$ on $\R^n$ as $\ell \to \infty$, i.e. the subsequence $(x^{(k_{j_\ell})})_{\ell \in \N}$ is convergent under $\norm{\argdot}_2$ on $\R$.
  }
\end{proof}
\begin{remark}
  The finite dimensionality of $\R^n$ is necessary for the above.
  In fact, it can be shown that this property holds on an arbitrary normed space if and only if it the underlying vector space is finite dimensional.

  For a counter example, consider the normed space $(\ell^{\infty}, \norm{\argdot}_{\infty})$ and let $(e^{(k)})_{k \in \N} \in \ell^{\infty}$ be defined by $e^{(k)}_j = \delta_{k j}$.
  This is clearly bounded as $\norm{e^{k}}_{\infty} = 1$ for each $k$.
  Suppose this has a convergent subsequence $e^{k_{j}} \to e = (e_1, e_2, \ldots) \in \ell^{\infty}$ under $\norm{\argdot}_{\infty}$ as $j\to \infty$, i.e. $\norm{e^{(k_j)} - e}_{\infty} \to 0$ as $j \to \infty$.

  Using the definition of $\norm{\argdot}_{\infty}$ (from \cref{infNorms}), for each $\ell \in \N$:
  \begin{align*}
    \abs{e^{(k_j)}_{\ell} - e_{\ell}} \leq \sup_{\ell \in \N} \abs{e^{(k_j)}_{\ell} - e_{\ell}} &= \norm{e^{(k_j)} - e}_{\infty} \to 0 \text{ as } j \to \infty \\
    \implies \abs{e^{(k_j)}_{\ell} - e_{\ell}} &\to 0 \text{ as } j \to \infty
  \end{align*}
  But by how we constructed $e^{(k_j)}$, for any $\ell$, $e^{(k_{j})}_{\ell} = 0$ for all sufficiently large $j$, hence $e_{\ell} = 0$ for all $\ell$, i.e. $\norm{e^{k_j}}_{\infty} \to 0$ as $j \to \infty$.
  This is a contradiction as $\norm{e^{(k_j)}}_{\infty} = 1$ for each $j$ by construction.
\end{remark}
For the following definitions, let $(X, d)$ be a metric space and $E \subseteq X$ a subset.
\begin{definition}[Limit point]
  We say that a point $x \in X$ is a \textit{limit point of $E$} if there exists a sequence $(x_k)$ in $E \setminus \{x\}$ such that $x_k \to x$.
\end{definition}
\begin{remark}
  The exclusion of the point itself from $E$ is important as otherwise every point of $E$ is a limit point of $E$ by considering a constant sequence.
\end{remark}
\begin{definition}[Isolated point]
  We say that a point $x \in E$ is isolated if it is not a limit point of $E$.
\end{definition}
\begin{definition}[Closure]
  The \textit{closure} of the set $E$ is defined to be the set:
  \begin{align*}
    \overline{E} &= \left\{x \in X: x  = \lim_{k \to \infty} x_k \text{ for some sequence $(x_k)$ in $E$}\right\} \\
                 &= E \cup \{x \in X: x \text{ is a limit point of $E$}\}
  \end{align*}
\end{definition}
\begin{definition}[Interior]
  We say that a point $x \in X$ is an \textit{interior point} of $E$ if there exists $r > 0$ such that $B_r(x) \subseteq E$.

  The \textit{interior} of $E$, denoted $\dot{E}$ is then defined to be the set of all interior points of $E$.
\end{definition}
\begin{remark}
  For any $r > 0$ we have $x \in B_r(x) \subseteq E$ and so $x \in E$, i.e. all interior points of $E$ must be in $E$, or equivalently, $\dot{E} \subseteq E$.
\end{remark}
\begin{example}
  Consider the set $X = \R$ with the Euclidean metric and the subset $E = [0, 1) \cup \{2\}$.
  We have the following:
  \begin{itemize}
    \item The limit points of $E$ are $[0, 1]$.

      For $x \in [0, 1)$, we can set $r = 1 - x > 0$ and so we can construct the sequence $(x + \frac{r}{n})_{n \in \N}$ on $E \setminus \{x\}$ which has $x + \frac{r}{n} \to x$, i.e. $x$ is a limit point of $E$.
      For $x = 1$ we instead use sequence $1 - 1/n$ on $E \{1\}$ which converges to 1.
    \item The only isolated point of $E$ is $\{2\}$.

      This is because all points in $x \in E \setminus \{2\}$ have $|x - 2| > 1$ and so there cannot be a sequence on $E \setminus \{2\}$ converging to 2.
    \item The closure of $E$ is $\overline{E} = [0, 1] \cup \{2\}$.
    \item The interior of $E$ is $\dot{E} = (0, 1)$.

      By the remark, we need only check points in $E$ itself.
      For any $x \in (0, 1)$, we take $r = \min\{x, 1 - x\} > 0$ so that $B_r(x) \subseteq E$.
      Further, for any $r > 0$, both $B_r(0)$ and $B_r(2)$ will always contain a point not in $E$, i.e. $0, 2 \notin \dot{E}$
  \end{itemize}
\end{example}
\begin{proposition}
  If $E \subseteq X$, then $\overline{E}$ is a closed subset of $X$.
  More specifically, we can write: \label{closureInter}
  \[
    \overline{E} = \bigcap_{\substack{F \subseteq X \text{ is closed }\\ E \subseteq F}} F
  \]
  i.e. $\overline{E}$ is the smallest closed set in $X$ (by \cref{interProp}) which contains $E$.
\end{proposition}
\begin{proof}
  See Example Sheet 2 Q8.
\end{proof}
\begin{proposition}
  If $E \subseteq X$, then the following are equivalent:
  \begin{enumerate}
    \item If $(x_k)$ is a sequence in $E$ such that $x_k \to x \in X$, then $x \in E$
    \item $E = \overline{E}$
    \item $E$ is a closed subset of $X$
  \end{enumerate}
\end{proposition}
\begin{proof}
  We have:
  \[
    \text{\textbf{(i)}} \iff \{x \in X: x \text{ is a limit point of E}\} \subseteq E \iff E = \overline{E} \iff \text{\textbf{(ii)}}
  \]

  If we have \textbf{(ii)}, then by \cref{closureInter}, $\overline{E}$ is closed, thus so is $E$.
  If we have \textbf{(iii)}, then $E \subseteq X$ is closed and $E \subseteq E$, thus $E$ is a set in the intersection from \cref{closureInter} and so $\overline{E} = E$.
  Therefore \textbf{(i)} $\iff$ \textbf{(ii)} $\iff$ \textbf{(iii)}, as required.
\end{proof}
\section{Cauchy Sequences and Completeness}
\begin{definition}[Cauchy sequence]
  Let $(X, d)$ be a metric space.
  We say that a sequence $(x_k)_{k \in \N}$ in $X$ is \textit{Cauchy} if:
  \[
    \forall \varepsilon > 0\ \exists N \in \N \text{ s.t. } d(x_n, x_m) < \varepsilon\ \forall n, m \geq N
  \]
\end{definition}
\begin{proposition}
  For a metric space $(X, d)$ and sequence $(x_n)_{n \in \N}$ on $X$ we have the following:
  \begin{enumerate}
    \item If $(x_n)$ is convergent, then $(x_n)$ is Cauchy
    \item If $(x_n)$ is Cauchy, then it is bounded
    \item If $(x_n)$ is Cauchy and has a subsequence $x_{n_k} \to x \in X$, then $x_n \to X$
  \end{enumerate}
\end{proposition}
\begin{remark}
  When we say that a sequence $(x_n)$ on $X$ is bounded, we mean that the set $\{x_n: n \in \N\}$ is a bounded subset of $X$.
\end{remark}
\begin{proof}[\textbf{i}]
  We can use the triangle inequality to write:
  \[
    d(x_n, x_m) \leq d(x_n, x) + d(x_m, x)
  \]
  Since $x_n \to x$ as $n \to \infty$, we can find $N$ such that $d(x_n, x) < \varepsilon/2$ for all $n \geq N$ and so $d(x_n, x_m) < \varepsilon$ for all $n, m \geq N$, i.e. $(x_n)$ is Cauchy.
\end{proof}
\begin{proof}[\textbf{ii}]
  Taking $\varepsilon = 1$, there exists $N \in \N$ such that $d(x_n, x_m) < 1$ for all $n, m \geq N$.
  If we take $m = N$, then $d(x_n, x_N) < 1$ for all $n \geq N$.
  So we know all sequence terms $n \geq N$ are within a ball of radius $1$ centered at $x_N$.
  As for the terms before $x_N$, since there are finitely many such terms, we can define:
  \[
    R = \max\{d(x_1, x_N), \ldots, d(x_{N - 1}, x_N), 1\} \geq 1
  \]
  With this $R$, we have that $x_n \in B_{R}(x_N)$ for all $n \in \N$, i.e. $(x_n)$ is bounded.
\end{proof}
\begin{proof}[\textbf{iii}]
  Since $(x_n)$ is Cauchy, given $\varepsilon > 0$, there exists $N$ such that $d(x_n, x_m) < \varepsilon$ for all $n, m \geq N$.
  We can use the triangle inequality to write:
  \[
    d(x_n, x) \leq d(x_{n_k}, x) + d(x_{n_k}, x_n)
  \]
  Since $x_{n_k} \to x$, there exists $K$ such that $d(x_{n_k}, x) < \varepsilon$ for all $k \geq K$.
  By definition of subsequences, the sequence $n_k$ is a strictly increasing sequence on $\N$ and so $n_k \geq k$ for any $k$, thus if we take $k = \max\{N, K\}$ then $n_k \geq N, K$ so that $d(x_{n_k}, x) < \varepsilon$ and $d(x_{n_k}, x_n) < \varepsilon$ for $n \geq N$.

  Thus for $n \geq N$, we have $d(x_n, x) < 2\varepsilon$, i.e. $x_{n} \to x$.
\end{proof}
\begin{definition}[Completeness]
  We say that a metric space $(X, d)$ is \textit{complete} if every Cauchy sequence in $X$ converges to some element in $X$.

  Moreover, we say a normed space $(V, \norm{\argdot})$ is \textit{complete} if it is complete as a metric space when equipped with the norm induced metric.
\end{definition}
\begin{theorem}[Completeness of $\R^n$]
  $\R^n$ with the Euclidean metric $\norm{\argdot}_2$ is complete.
\end{theorem}
\begin{remark}
  As discussed previously, this result will also holds with respect to any norm induced metric on a finite dimensional vector space.
\end{remark}
\begin{proof}
  Suppose $(x^{(k)})_{k \in \N}$ is a Cauchy sequence under $\norm{\argdot}_2$ in $\R^n$, thus there exists $N$ such that $\norm{x^{(k)} - x^{(m)}}_2 < \varepsilon$ for all $k, m \geq N$

  Considering each of the components of $x^{(k)} - x^{(m)}$, we see that for each $j \in \{1, \ldots, n\}$:
  \begin{align*}
    \abs{x^{(k)}_j - x^{(m)}_j} &= \left[\left(x^{(k)}_j - x^{(m)}_j\right)^2\right]^{\frac{1}{2}} \\
                                &\leq \left[(x^{(k)}_1 - x^{(m)}_1)^2 + \cdots + (x^{(k)}_n - x^{(m)}_n)^2\right]^{\frac{1}{2}} \\
                                &= \norm{x^{(k)} - x^{(m)}}_{2} \\
                                &< \varepsilon\ \forall k, m \geq N
  \end{align*}
  and so the sequences $(x^{(k)}_j)_{k \in \N}$ are Cauchy for each $j \in \{1, \ldots, n\}$.
  We saw in IA Analysis I, that $\R$ is complete under $|\argdot|$ and so for each $j \in \{1, \ldots, n\}$ we must have $x^{(k)}_j \to x_j$ for some $x_j \in \R$.
  Thus using \cref{vectorConv}, $x^{(k)} \to x = (x_1, \ldots, x_n) \in \R^{n}$ under $\norm{\argdot}_2$ as $k \to \infty$.
\end{proof}
\begin{theorem}
  The metric space $(C([a, b]), \norm{\argdot}_{\infty})$ is complete.
\end{theorem}
\begin{proof}
  Looking at the uniform metric from \cref{unifMetric}, we see that the sequence of functions $f_n: [a, b] \to \R$ is uniformly if and only if it is a Cauchy sequence on $C([a, b])$ under $\norm{\argdot}_{\infty}$.

  Suppose $(f_n)$ is a Cauchy sequence on $C([a, b])$ under $\norm{\argdot}_{\infty}$, then by the Cauchy criterion (\cref{cauchyCritereon}), $(f_n)$ converges under $\norm{\argdot}_{\infty}$ to some $f: [a, b] \to \R$.
  Further, in \cref{cabClosed}, we showed that the uniform limit of continuous functions is also continuous and so $f \in C([a, b])$, i.e. the space is complete.
\end{proof}
\begin{theorem}[Completeness of $\ell^1$, $\ell^2$, and $\ell^\infty$]
  The metric spaces $(\ell^1, \norm{\argdot}_1)$, $(\ell^2, \norm{\argdot}_2)$, and $(\ell^{\infty}, \norm{\argdot}_{\infty})$ are all complete normed spaces.
\end{theorem}
\begin{proof}[For $\ell^1$ and $\ell^2$]
  See Example Sheet 2.
\end{proof}
\begin{proof}[For $\ell^\infty$]
  By definition of $\ell^{\infty}$, each sequence in $\ell^{\infty}$ must be bounded.
  Further, since sequences are just functions $\N \to \R$, $\ell^{\infty}$ can be thought of as the set of bounded functions $\N \to \R$ .

  Thus a Cauchy sequence in $\ell^{\infty}$ under $\norm{\argdot}_{\infty}$ is a Cauchy sequence of bounded functions $\N \to \R$ under the uniform norm for functions.

  The Cauchy criterion only requires the domain of the functions to be a subset of $\R$, thus by the Cauchy criterion, every uniformly Cauchy sequence of functions $f_n: \N \to \R$ converges uniformly to some $f: \N \to \R$.
  That is, every Cauchy sequence in $\ell^{\infty}$ converges under $\norm{\argdot}_{\infty}$.
  Moreover, we saw in Example Sheet 1, that since $f_n$ is bounded for each $n$, the limit $f$ is also bounded and so $f \in \ell^{\infty}$.
  Thus $\ell^{\infty}$ is closed.
\end{proof}
