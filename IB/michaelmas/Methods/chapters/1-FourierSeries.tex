\chapter{Fourier Series}
\section{Motivation}
In 1807, J. Fourier was studying heat conduction along a metal rod.
This lead him to investigate $2\pi$ periodic functions, i.e. functions for which $f(\theta + 2\pi) = f(\theta)\ \forall \theta$.

Fourier boldly claimed that such functions could be written in the following form:
\[
  f(\theta) = \sum_{n \in \Z} \hat{f}_n e^{i n \theta}
\]
where the coefficients $\hat{f}_{n}$ can be obtained from $f$ using:
\[
  \hat{f}_{n} = \frac{1}{2\pi} \int_{0}^{2\pi} f(\theta) e^{-in \theta} \d{\theta}
\]
He thought that every $2\pi$ periodic function could be expressed in this way, which is in fact false, and we will discuss this later.
\section{Modern Approach}
Consider a vector space:
\[
  V = \{\text{``nice'' functions $f: \R \to \C$ such that} f(\theta + L) = f(\theta)\ \forall \theta\}
\]
It can be easily check that the vector space axioms hold provided that the sum of two ``nice'' functions is also ``nice'', and scalar multiples of ``nice'' functions are also ``nice''.
\begin{remark}[Remarks]
  \begin{itemize}
    \item We will discuss later what we mean by ``nice'' functions.
    \item If $f \in V$, then we need only specify the values of $f$ on some interval of length $L$, for example, $[0, L)$ or $[-L/2, L/2)$.
  \end{itemize}
\end{remark}
We equip $V$ with the following inner product:
\begin{equation}
  \langle f, g \rangle = \int_{0}^{L} f(\theta)\overline{g(\theta)} \d{\theta} \label{innerProdFourier}
\end{equation}
where $\overline{g}$ is the complex conjugate of $g$.
Since $f$ and $g$ are $L$-periodic, we can take the integral bounds to be any interval of length $L$.
We can now define the norm in $V$ the standard way:
\begin{equation}
  \norm{f} = \sqrt{\langle f, f \rangle} = \sqrt{\int_{0}^{L} |f|^2 \d{\theta}} \label{fourierNorm}
\end{equation}

Consider the following family of functions:
\[
  e_{n}(\theta) = e^{2 \pi i n \theta/ L}
\]
for $n \in \Z$.
Certainly $e_n \in V$ for each $n$ and:
\[
  \langle e_n, e_m \rangle = \int_{0}^{L} e^{2\pi i (n - m) \theta/L} \d{\theta} = L \delta_{n m}
\]
Thus the functions $\{e_n\}_{n \in \Z}$ are orthogonal in $V$ and all have $\norm{e_n}^2 = L$.
\begin{remark}[Othogonal Basis]
  Recall from IA Vectors and Matrices, that if $V_N$ is an $N$ dimensional vector space and $\{\vec{e}_n\}^{N}_{n = 1}$ are orthogonal with $\norm{\vec{e}_n}^2 = L$, then we can use these vectors as a basis.
  That is, for any $\vec{x} \in V_N$ we can write:
  \[
    \vec{x} = \sum_{n = 1}^{N} \hat{x}_n \vec{e}_n
  \]
  where, since $\vec{e}_n \cdot \vec{e}_m = \delta_{n m}$, we can extract the coefficients using:
  \[
    \vec{x} \cdot \vec{e}_m = \sum_{n = 1}^{N} \hat{x}_n \vec{e}_n \cdot \vec{e}_m = L\hat{x}_{m}
  \]
  So for every $\vec{x} \in V_N$ we have:
  \[
    \vec{x} = \sum_{n = 1}^{N} \hat{x}_n \vec{e}_n \text{ where } \hat{x}_n = \frac{1}{L} \vec{x} \cdot \vec{e}_n
  \]
\end{remark}
We now note that $V$ is an infinite vector space as no finite subset of $\{e_n\}_{n \in \Z}$ is linearly dependent.
Can we express $f \in V$ as an infinite sum of scaled elements of the basis?
For now, we look past this and try anyways.

Suppose that $f \in V$ can be written in the form:
\[
  f(\theta) = \sum_{n \in \Z} \hat{f}_n e_n(\theta) \equiv \sum_{n \in \Z} \hat{f}_{n} e^{2\pi i n \theta/L}
\]
We can use the same ideas to extract the coefficients as we did for the finite vector space.
\begin{align*}
  \langle f, \vec{e}_m \rangle &= \sum_{n \in \Z} \hat{f}_n \langle \vec{e}_n, \vec{e}_m \rangle \\
                               &= \sum_{n \in Z} \hat{f}_n L \delta_{n m} \\
                               &= L \hat{f}_{m}
\end{align*}
where we assume that the infinite sum and integral can be swapped.
Thus the coefficients $\{\hat{f}_n\}$ are given by:
\[
  \hat{f}_n = \frac{1}{L} \langle f, \vec{e}_n \rangle = \frac{1}{L} \int_{0}^{L} f(\theta) e^{-2\pi i n \theta / L} \d{\theta}
\]
If we set $L = 2\pi$, then we have recovered Fourier's claim.
\begin{definition}[Complex Fourier Series]
  For an $L$-periodic function $f: \R \to \C$, we define its \textit{complex Fourier series} as:
  \[
    \sum_{n \in \Z} \hat{f}_n e^{2\pi i n \theta / L}
  \]
  where:
  \[
    \hat{f}_n = \frac{1}{L} \int_{0}^{L} f(\theta) e^{-2\pi i n \theta / L} \d{\theta}
  \]
  are called the \textit{complex Fourier coefficients}.
\end{definition}
\begin{remark}[Notation]
  For an $L$-periodic $f$ we write:
  \[
    f(\theta) \sim \sum_{n \in \Z} \hat{f}_n e^{2\pi i n \theta / L}
  \]
  to mean that the series on the right hand side is the Fourier series of the function $f$ on the left hand side.

  We would like to replace $\sim$ with $=$ but \textbf{cannot} always do this.
\end{remark}
We can also write the Fourier series as sums of sines and cosines.
We split the sum into three sets of indices $\Z = \{n = 0\} \cup \{n > 0\} \cup \{n < 0\}$:
\begin{align*}
  \sum_{n \in \Z} \hat{f}_{n} e^{2\pi i n \theta / L} &= \hat{f}_0 + \sum_{n = 1}^{\infty} \hat{f}_{n} e^{2\pi i n \theta/L} + \sum_{n = 1}^{\infty} \hat{f}_{-n} e^{-2 \pi i n \theta/L} \\
                                                      &= \hat{f}_{0} + \sum_{n = 1}^{\infty} \hat{f}_{n} \left[\cos\left(\frac{2 \pi n \theta}{L}\right) + i \sin\left(\frac{2 \pi n \theta}{L}\right)\right] \\
                                                      &\quad+ \sum_{n = 1}^{\infty} \hat{f}_{-n} \left[\cos\left(\frac{2 \pi n \theta}{L}\right) - i \sin\left(\frac{2 \pi n \theta}{L}\right)\right] \\
                                                      &= \hat{f}_{0} + \sum_{n = 1}^{\infty} \left[(\hat{f}_{n} + \hat{f}_{-n}) \cos\left(\frac{2 \pi n \theta}{L}\right) + i(\hat{f}_{n} - \hat{f}_{-n}) \sin \left(\frac{2 \pi n \theta}{L}\right)\right]
\end{align*}
We can then then define $a_{n} = \hat{f}_{n} + \hat{f}_{-n} \text{ and } b_n = i(\hat{f}_{n} - \hat{f}_{-n})$
\begin{definition}[Fourier Series]
  For an $L$-periodic function $f: \R \to \C$, we define its Fourier series as:
  \[
    \frac{1}{2} a_0 + \sum_{n = 1}^{\infty} \left[a_n \cos\left(\frac{2\pi n \theta}{L}\right) + b_n \sin \left(\frac{2 \pi n \theta}{L}\right)\right]
  \]
  where:
\[
  a_n = \frac{2}{L} \int_{0}^{L} f(\theta) \cos\left(\frac{2 \pi n \theta}{L}\right) \d{\theta} \text{ and }
  b_n = \frac{2}{L} \int_{0}^{L} f(\theta) \sin\left(\frac{2 \pi n \theta}{L}\right) \d{\theta}
\]
are called the \textit{Fourier coefficients} of $f$.
\end{definition}
\begin{remark}
  If $f(\theta)$ is an odd function, then:
  \begin{align*}
    a_n &= \frac{2}{L} \int_{0}^{L} f(\theta) \cos\left(\frac{2 \pi n \theta}{L}\right) \d{\theta} \\
        &= \frac{2}{L} \int_{-L/2}^{L/2} f(\theta) \cos\left(\frac{2 \pi n \theta}{L}\right) \d{\theta} \\
        &= 0
  \end{align*}
  as the integrand is a product of an odd and an even function and the bounds are symmetric.

  Similarly, if $f(\theta)$ is an even function, then $b_n = 0$.
\end{remark}
\begin{remark}[Advice]
  When asked to construct a Fourier series, it is often advisable to start by sketching the function to establish if it is even or odd \textbf{before} starting to manually compute the Fourier coefficients.
\end{remark}
\begin{example}
  Define the 1-periodic function $f: \R \to \R$ such that $f(\theta) = \theta(1 - \theta)$ on $[0, 1)$.
  The graph of $f$ looks like:
  \begin{center}
  \begin{tikzpicture}[>=stealth]
    \draw[->] (-6,0) -- (6,0) node[right] {$\theta$};
    \draw[->] (0,0) -- (0,3) node[below right] {$f(\theta)$};

    \clip (-6, 0) rectangle (6, 3);
    \begin{scope}[scale=3]
      \foreach \s in {-2, -1, 0, 1, 2} {
        \draw[thick, domain=0:1, smooth, variable=\x, xshift=\s cm] plot ({\x}, {\x * (1 - \x)});
      }
    \end{scope}
  \end{tikzpicture}
  \end{center}
  We have:
  \[
    \hat{f}_0 = \int_{0}^{1} \theta(1 - \theta) \d{\theta} = \frac{1}{6}
  \]
  For $n \neq 0$, we integrate by parts:
  \begin{align*}
    \hat{f}_n &= \int_{0}^{1} \theta(1 - \theta)e^{-2 \pi i n \theta} \d{\theta} \\
              &= \cancelto{0}{\eval{-\frac{1}{2 \pi i n} \theta(1 - \theta) e^{-2 \pi i n \theta}}{\theta = 0}{1}} + \frac{1}{2 \pi i n} \int_{0}^{1} (1 - 2 \theta) e^{-2 \pi i n} \d{\theta} \\
              &= \eval{- \frac{1}{(2 \pi i n)^2}(1 - 2 \theta) e^{-2 \pi n \theta}}{\theta = 0}{1} \\
              &= - \frac{1}{2(\pi n)^2}
  \end{align*}
  The complex Fourier series for $f$ is then:
  \[
    f(\theta) \sim \frac{1}{6} - \sum_{n \neq 0} \frac{1}{2 (n \pi)^2} e^{2 \pi i n \theta}
  \]
  We can use these coefficients to quickly find the non-complex Fourier series for $f$.
  Since $f(\theta)$ is even, we know that $b_n = 0\ \forall n$.
  To find $a_n$:
  \[
    a_n = \hat{f}_{n} + \hat{f}_{-n} = - \frac{1}{(n \pi)^2}
  \]
  or equivalently the Fourier series for $f$ is then:
  \[
    f(\theta) \sim \frac{1}{6} - \sum_{n = 1}^{\infty} \frac{1}{(n \pi)^2} \cos(2 \pi n \theta)
  \]
  We in fact have equality here, however we need some more theory to be able to show this.
\end{example}
\begin{remark}
  We will cover this in detail later, however ``nicer'' functions have Fourier coefficients which decay quickly.
  For example, the Fourier coefficients of a smooth function will decay faster than those of a discontinuous function.

  To suggest why this is the case, notice how in the above example, each time we integrate by parts, we bring down a factor of $1/n$ which would cause the coefficient will decay faster, however we also introduce a boundary term which slows down decay, unless it vanishes.
  Here, we see that the first boundary term disappears because $f$ is continuous at the boundaries, but the second one does not since $f'$ is not continuous at the boundaries.
\end{remark}
\section{Convergence of Fourier Series}
\begin{definition}[Partial Fourier Series]
  For an $L$-periodic function $f: \R \to \C$ we define the \textit{partial Fourier series} of $f$ as the finite sum:
  \[
    (S_N f)(\theta) = \sum_{|n| \leq N} \hat{f}_{n} e^{2 \pi n \theta / L}
  \]
  where:
  \[
    \hat{f}_{n} = \frac{1}{L} \int_{0}^{L} f(\theta) e^{- 2\pi i n \theta/L} \d{\theta}
  \]
\end{definition}
We would like to figure out if $S_N f \to f$ as $N \to \infty$.
But in what way do we want the sequence of functions $S_N f$ to converge to $f$?
\begin{itemize}
  \item \textbf{Pointwise Convergence}:
    \[
      \lim_{N \to \infty} |S_{N}f(\theta) - f(\theta)| = 0\ \forall \theta \in [0, L)
    \]
  \item \textbf{Uniform Convergence}:
    \[
      \lim_{N \to \infty} \left[\sup_{x \in [0, L)} |S_N f(\theta) - f(\theta)|\right] = 0
    \]
  \item \textbf{Convergence in Norm}:
    Using the norm we defined earlier in \cref{fourierNorm}, convergence in norm is:
    \[
      \lim_{N \to \infty} \norm{S_N f - f}^2 = \lim_{N \to \infty} \int_{0}^{L} |S_{N} f(\theta) - f(\theta)|^2 \d{\theta} = 0
    \]
\end{itemize}
Convergence in norm is the nicest to show, however we would like to be able to show pointwise convergence.
\begin{theorem}[Convergence of Fourier Series]
  If $f: \R \to \C$ is $L$-periodic such that the following hold: \label{convFourier}
  \begin{enumerate}
    \item $f$ is continuous on $[0, L)$ apart from at a finite number of jump discontinuities.
    \item Both $\Re(f)$ and $\Im(f)$ have a finite number of local extremum on $[0, L)$.
  \end{enumerate}
  then we have:
  \[
    \frac{1}{2}[f(\theta_{+}) + f(\theta_{-})] = \lim_{N \to \infty} S_N f(\theta)\ \forall \theta \in \R
  \]
  where $f(\theta_{\pm}) = \lim_{\varepsilon \to 0^\pm} f(\theta + \varepsilon)$.
\end{theorem}
\begin{remark}[Remarks]
  \begin{itemize}
    \item At points of continuity of $f$, $f(\theta_{+}) = f(\theta_{-})$ and so $\lim_{N \to \infty} S_{N} f(\theta) = f(\theta)$.
    \item At jump discontinuities, $\lim_{N \to \infty} S_{N} f(\theta)$ is the average value of the left and right limits at the jump.
    \item Functions satisfying \textbf{(i)} and \textbf{(ii)} are called \textit{Dirichlet functions}.
      We will only deal with such functions for the remainder of the course and so we take the relation:
      \[
        f(\theta) \sim \sum_{n \in \Z} \hat{f}_{n} e^{2 \pi i n \theta / L}
      \]
      to mean that the two sides are equal at points of continuity of $f$, and at jump discontinuities, the right and side is the average of $f$ at the jump discontinuity.
  \end{itemize}
\end{remark}
For ease, we shall only prove this result for smooth $f$.
\begin{proof}[For $f \in C^\infty$]
  We assume take $L = 2\pi$ for notational brevity, however the result still holds in the general case.
  Due to the periodicity of $S_N f$, we need only show the result for each $\theta_0 \in [-\pi, \pi)$.
  In fact, we need only show that we have convergence at $\theta = 0$ by redefining $f(\theta)$ to be $f(\theta + \theta_0)$.

  Consider the partial Fourier series evaluated at $\theta = 0$:
  \begin{align*}
    (S_N f)(\theta) &= \sum_{|n| \leq N} \hat{f}_{n} e^{in \cdot 0} \\
                    &= \sum_{|n| \leq N} \frac{1}{2\pi} \int_{-\pi}^{\pi} f(\theta) e^{-i n \theta} \d{\theta} \\
                    &= \frac{1}{2\pi} \int_{-\pi}^{\pi} \left[\sum_{|n| \leq N} e^{-i n \theta} \right] f(\theta) \d{\theta}
  \end{align*}
  For $\theta \in \R \setminus (2 \pi \Z)$, this sum can be written as:
  \begin{align*}
    \sum_{|n| \leq N} e^{-i n \theta} &= e^{i N \theta} \sum_{i = 0}^{2N} e^{- i n \theta} \\
                                      &= e^{i N \theta} \cdot \frac{1 - e^{- i (2N + 1) \theta}}{1 - e^{-i \theta}} \\
                                      &= \frac{e^{i N \theta} - e^{- i (N + 1) \theta}}{1 - e^{-i \theta}} \\
                                      &= \frac{e^{i \left(N + \frac{1}{2}\right)\theta} - e^{-i \left(N + \frac{1}{2}\right)\theta}}{e^{i \frac{\theta}{2}} - e^{-i \frac{\theta}{2}}} \\
                                      &= \frac{\sin\left[\left(N + \frac{1}{2}\right)\theta\right]}{\sin(\theta/2)}
  \end{align*}
  For $\theta \in 2 \pi \Z$, the sum is just $\sum_{|n| \leq N} 1 = 2N + 1$.

  We define the value of this sum to be the \textit{$N$-th Dirichlet kernel}:
  \[
    D_{N}(\theta) = \begin{cases}
    \frac{\sin\left[\left(N + \frac{1}{2}\right)\theta\right]}{\sin(\theta/2)} & \text{ if } \theta \in \R \setminus (2 \pi \Z) \\
    2N + 1 & \text{ if } \theta \in 2 \pi \Z
    \end{cases}
  \]
  By considering the finite sum of exponentials:
  \begin{enumerate}
    \item $D_N$ is smooth and $2\pi$-periodic.
    \item $\int_{-\pi}^{\pi} D_{N}(\theta) \d{\theta} = 2 \pi$ for all $N \geq 0$ as only $N = 0$ will contribute when integrated.
  \end{enumerate}
  We now consider the quantity:
  \begin{align}
    S_{N}f(0) - f(0) &= \frac{1}{2\pi} \int_{-\pi}^{\pi} D_N(\theta) f(\theta) \d{\theta} - f(0) \nonumber \\
                     &= \frac{1}{2\pi} \int_{-\pi}^{\pi} D_N(\theta) f(\theta) \d{\theta} - f(0) \cdot \frac{1}{2\pi} \int_{-\pi}^{\pi} D_N(\theta) \d{\theta} \nonumber \\
                     &= \frac{1}{2\pi} \int_{-\pi}^{\pi} D_N(\theta)[f(\theta) - f(0)] \d{x} \nonumber \\
                     &= \frac{1}{2\pi} \int_{-\pi}^{\pi} \sin\left[\left(N + \frac{1}{2}\right)\theta\right] F(\theta) \d{\theta} \tag{$\dag$} \label{convDag}
  \end{align}
  where we define:
  \[
    F(\theta) = \frac{\theta}{\sin(\theta/2)} \left[\frac{f(\theta) - f(0)}{\theta}\right]
  \]
  Since $f \in C^{\infty}(\R)$, it follows that $F(\theta) \in C^{\infty}(\R)$ as:
  \[
    f(\theta) - f(0) = \int_{0}^{\theta} f'(t) \d{t} = \theta \int_{0}^{1} f'(t \theta) \d{t} \implies \abs{\frac{f(\theta) - f(0)}{\theta}} \leq \int_{0}^{1} |f'(t \theta)| \d{t}
  \]
  Since $F(\theta)$ is smooth, we can therefore integrate \cref{convDag} by parts to yield:
  \begin{align*}
    S_{N} f(0) - f(0) &= \frac{1}{2 \pi} \cdot \frac{1}{N + \frac{1}{2}} \int_{-\pi}^{\pi} \cos\left[\left(N + \frac{1}{2}\right)\theta\right] F'(\theta) \d{\theta}
  \end{align*}
  and so:
  \[
    |S_N f(0) - f(0)| \leq \frac{1}{2\pi} \cdot \frac{1}{N + \frac{1}{2}} \int_{-\pi}^{\pi} |F'(\theta)| \d{\theta} \to 0 \text{ as $N \to \infty$}
  \]
  Therefore $S_N f(0) \to f(0)$ as $N \to \infty$, as required.
\end{proof}
\begin{remark}[Remarks]
  \nonexaminable
  \begin{itemize}
    \item If $f$ was not so regular, then we cannot proceed past:
      \[
        (S_N f)(0) - f(0) = \frac{1}{2\pi} \int_{-\pi}^{\pi} \sin\left[\left(N + \frac{1}{2}\right)\theta\right] F(\theta) \d{\theta}
      \]
      For Dirichlet functions $f$ which are not $C^{1}$, we can in fact deal proceed from here using the \textit{Riemann-Lebesgue Lemma}.
    \item For a while, people believed that if $\int_{0}^{L} |f(\theta)| \d{\theta} < \infty$, then:
      \[
        f(\theta) = \sum_{n \in \Z} \hat{f} e^{2 \pi i \theta / L}
      \]
      almost everywhere.
      However, later on, a counter example was constructed which causes this sum to diverges almost everywhere.
  \end{itemize}
\end{remark}

\section{Even and Odd Extensions}
Suppose we are given a function $f: [0, L) \to \C$ which we would like to calculate a Fourier series for.
To do this, $f: \R \to \C$ needs to be periodic.
We could periodically extend this by just repeating the function every interval of length $L$ and then find the Fourier series of this, however we would have to calculate both the $a_n$ and $b_n$ sets of coefficients.
However, if we first construct new $2L$-periodic functions $f_{\text{even}}$ and $f_{\text{odd}}$ which are even and odd respectively but still have $f(\theta) = f_{\text{even}}(\theta) = f_{\text{odd}}(\theta)$  for $\theta \in [0, L)$, then we need only calculate $A_n$ for $f_{\text{even}}$ and $B_n$ for $f_{\text{odd}}$.

Based on the above criterion, we arrive at the following functions on $[-L, L)$:
\begin{align*}
  f_{\text{even}}(\theta) &= \begin{cases}
  f(\theta) & \text{ if } \theta \in [0, L) \\
  f(-\theta) & \text{ if } \theta \in [-L, 0)
  \end{cases} \\
  f_{\text{odd}}(\theta) &= \begin{cases}
  f(\theta) & \text{ if } \theta \in [0, L) \\
  -f(-\theta) & \text{ if } \theta \in [-L, 0)
  \end{cases}
\end{align*}
which we periodically extend to $\R$.

These have Fourier series given by:
\begin{align*}
  f_{\text{even}}(\theta) &\sim \frac{1}{2}A_0 + \sum_{n = 1}^{\infty} A_n \cos\left(\frac{\pi n \theta}{L}\right) \\
  f_{\text{odd}}(\theta) &\sim \sum_{n = 1}^{\infty} B_n \sin\left(\frac{\pi n \theta}{L}\right)
\end{align*}
where:
\begin{align*}
  A_n &= \frac{2}{2L} \int_{-L}^{L} f_{\text{even}}(\theta) \cos\left(\frac{2 \pi n \theta}{2L}\right) \d{\theta} \\
      &= \frac{2}{L} \int_{0}^{L} f(\theta) \cos\left(\frac{\pi n \theta}{L}\right) \d{\theta}
\end{align*}
as both $f_{even}$ and $\cos$ are even functions, and:
\begin{align*}
  B_n &= \frac{2}{2L} \int_{-L}^{L} f_{\text{odd}}(\theta) \sin\left(\frac{2 \pi n \theta}{2L}\right) \d{\theta} \\
      &= \frac{2}{L} \int_{0}^{L} f(\theta) \sin\left(\frac{n \pi \theta}{L}\right) \d{\theta}
\end{align*}
as both $f_{odd}$ and $\sin$ are odd functions, so their product is even.

By construction $f(\theta) = f_{\text{even}}(\theta) = f_{\text{odd}}(\theta)$ on $[0, L)$ and so provided $f$ is a Dirichlet function, $f_{\text{even}}$ and $f_{\text{odd}}$ will also be Dirichlet functions and so by \cref{convFourier}, for $\theta \in (0, L)$, $f(\theta)$ will be exactly equal to both the Fourier series of $f_{\text{even}}$ and of $f_{\text{odd}}$.
\begin{definition}[Cosine and Sine Series]
  For a function $f: [0, L) \to \C$, its \textit{cosine and sine series} are given by:
  \[
    \frac{1}{2} A_0 + \sum_{i = 1}^{\infty} A_n \cos\left(\frac{n \pi \theta}{L}\right) \text{ and } \sum_{n = 1}^{\infty} B_n \sin\left(\frac{n \pi \theta}{L}\right)
  \]
  respectively, where $A_n$ and $B_n$ are given by:
  \[
    A_n = \frac{2}{L} \int_{0}^{L} f(\theta) \cos\left(\frac{\pi n \theta}{L}\right) \d{\theta} \text{ and } B_n = \frac{2}{L} \int_{0}^{L} f(\theta) \sin\left(\frac{n \pi \theta}{L}\right) \d{\theta}
  \]
\end{definition}
\begin{example}
  Consider the function $f: [0, 1) \to \R,\ \theta \mapsto 1$.

  \textbf{Sine Series}\par
  The odd extension of this looks like:
  \begin{center}
  \begin{tikzpicture}[>=stealth]
    \draw[->] (-4, 0) -- (4, 0) node[right] {$\theta$};
    \draw[->] (0, -2) -- (0, 2) node[right] {$f_{\text{odd}}$};

    \clip (-4, -2) rectangle (4, 2);
    \foreach \x in {-4, -2, 0, 2, 4} {
      \draw (\x, 1) -- (\x + 1, 1);
      \draw (\x-1, -1) -- (\x, -1);


      \draw[dashed] (\x, 1) -- (\x, -1);
      \draw[dashed] (\x + 1, 1) -- (\x + 1, -1);

      \node[yshift=-3, xshift=5] at (\x, 0) {\tiny$\x$};

      \filldraw[fill=black, draw=black] (\x, 1) circle (1.3pt);
      \filldraw[fill=white, draw=black] (\x, -1) circle (1.3pt);

      \filldraw[fill=black, draw=black] (\x + 1, -1) circle (1.3pt);
      \filldraw[fill=white, draw=black] (\x + 1, 1) circle (1.3pt);
    }
    \foreach \x in {-3, -1, 1, 3} \node[yshift=-3, xshift=5] at (\x, 0) {\tiny$\x$};
  \end{tikzpicture}
  \end{center}
  Computing $B_n$ we have:
  \[
    B_n = \frac{2}{1} \int_{0}^{1} \sin(n \pi \theta) \d{\theta} = \frac{2}{n \pi}[1 - (-1)^{n}]
  \]
  Since $f_{\text{odd}}$ is a Dirichlet function, for $\theta \in (0, 1)$, $f_{\text{odd}}$ is always equal to its Fourier series and so:
  \[
    1 = \frac{4}{\pi} \sum_{n \text{ odd}} \frac{\sin(n \pi \theta)}{n}\ \forall \theta \in (0, 1)
  \]

  \textbf{Cosine Series}\par
  The even extension is just $f_{\text{even}}(\theta) = 1\ \forall \theta \in \R$.
  Computing $A_n$ we have $A_0 = 2$ and $A_n = 0$ for $n > 0$, giving the tautological result:
  \[
    1 = \frac{1}{2} \cdot 2 + 0

\end{example}
\section{Regularity and Decay of Fourier Coefficients}
Fourier series are infinite sums and so we would hope that their terms tend to 0.
\subsection{Riemann Lebesgue Lemma}
\nonexaminable
\begin{theorem}[Riemann Lebesgue Lemma]
  If $\int_{0}^{L} |f(\theta)| \d{\theta} < \infty$, then: \label{RLL}
  \[
    \int_{0}^{L} f(\theta) e^{- i \lambda \theta} \d{\theta} \to 0 \text{ as } |\lambda| \to \infty
  \]
\end{theorem}
\begin{proof}
  Not lectured.
\end{proof}
If we take $\lambda = \frac{2 \pi n}{L}$ in the above then we get that:
\[
  \hat{f}_n \to 0 \text{ as } |n| \to \infty
\]
and so $a_n, b_n \to 0$ also as $n \to \infty$.
\subsection{Decay of Fourier Coefficients}
Suppose $f: \R \to \C$ is $L$-periodic and $k$-times continuously differentiable, that is, $f^{(r)}: \R \to \C$ is continuous for $r \in \{0, \ldots, k\}$.

Integrating by parts, for $n \geq 1$:
\begin{align*}
  \hat{f}_n &= \frac{1}{L} \int_{0}^{L} f(\theta) e^{-2 \pi \pi n \theta/L} \d{\theta} \\
            &= \eval{\left(-\frac{L}{2\pi i n}\right) \frac{f(\theta)}{L} e^{-2 \pi i n \theta/L}}{\theta = 0^{+}}{L^{-}} + \left(\frac{L}{2 \pi i n}\right) \frac{1}{L} \int_{0}^{L} f'(\theta) e^{-2 \pi i n \theta / L} \d{\theta}
\end{align*}
Since $f$ is continuous , $f(L^{-}) = f(0^{-}) = f(0^{+})$ and so the boundary term vanishes.

Since $f$ is $k$ times continuously differentiable, we can do iterate this process a total of $k$ times to yield:
\[
  \hat{f}_n = \left(\frac{L}{2\pi i n}\right)^{k} \frac{1}{L} \int_{0}^{L} f^{(k)}(\theta) e^{-2 \pi i n \theta / L} \d{\theta}
\]
Since $|f^{(k)}|$ is continuous it is integrable and so $\int_{0}^{L} |f^{(k)}(\theta)| \d{\theta} < \infty$.
We can therefore apply \cref{RLL} again with $\lambda = \frac{2 \pi n}{L}$, to conclude that the above integral is $o(1)$.
Thus $\hat{f}_n = o(1/n^k)$.
So for larger $k$ (i.e. more regular $f$), the Fourier coefficients $\hat{f}_n$ decay faster.

Furthermore, if it is true that $f^{(k + 1)}(\theta)$ exists and is not necessarily continuous but is integrable, i.e. $\int_{0}^{L} |f^{(k + 1)}(\theta)| \d{\theta}$, then we can iterate this process a further step before applying \cref{RLL} and so in this case $\hat{f}_{n} = o(1/n^{k + 1})$.
\section{Term-wise Differentiation}
Let $f: \R \to \C$ be $L$-periodic, continuously differentiable on $(0, L)$ with a bounded derivative.
Setting $g = f'$, we can find the Fourier coefficients for $g$.
Instead of applying integration by parts to differentiate $f$, we instead integrate $f'$ and differentiate the exponential to yield:
\begin{align*}
  \hat{g}_n &= \frac{1}{L} \int_{0}^{L} f'(\theta) e^{-2\pi i n \theta / L} \d{\theta} \\
            &= \eval{\frac{1}{L} f(\theta) e^{-2 \pi i n \theta / L}}{\theta = 0^{+}}{L^{-}} + \left(\frac{2 \pi i n}{L}\right) \frac{1}{L} \int_{0}^{L} f(\theta) e^{-2 \pi i n \theta/L} \d{\theta} \\
            &= \frac{f(L^{-}) - f(0^{+})}{L} + \left(\frac{2 \pi i n}{L}\right) \hat{f}_n
\end{align*}
Since $f$ is $L$-periodic $f(L^{-}) = f(0^{-})$ and since it is continuous on $\R$ $f(0^{-}) = f(0^{+})$, thus the boundary term disappears giving:
\[
  \hat{g}_n = \frac{2 \pi i n}{L} \hat{f}_n
\]
hence:
\begin{align*}
  f'(\theta) = g(\theta) &\sim \sum_{n \in \Z} \left(\frac{2 \pi i n}{L}\right) \hat{f}_n e^{2 \pi i n \theta/L}  \\
                         &= \sum_{n \in \Z}^{\infty} \deriv{}{\theta}[\hat{f}_n e^{2 \pi n \theta /L}]
\end{align*}
and so the Fourier series of the derivative is the term by term differentiation of the Fourier series for the original function.
\section{Parseval's Theorem}
Suppose that $f, g: \R \to \C$ are $L$-periodic and so they have Fourier series given by:
\begin{align*}
  f(\theta) &\sim \sum_{n \in \Z} \hat{f}_n e_n(\theta) \\
  g(\theta) &\sim \sum_{n \in \Z} \hat{g}_n e_n(\theta)
\end{align*}
where $e_n(\theta) = e^{2 \pi i n \theta/L}$.

Taking the inner product as defined in \cref{innerProdFourier} of $f$ and $g$, assuming that we can swap the infinite sums and integrals to utilise linearity, then:
\begin{align*}
  \langle f, g \rangle &= \sum_{n, m \in \Z} \hat{f}_n \overline{\hat{g}_m} \langle e_n, e_m \rangle \\
                       &= L \sum_{n \in \Z} \hat{f}_n \overline{\hat{g}_n}
\end{align*}
Note also that we can ignore any differences between $f$ or $g$ and their respective Fourier series as these can only occur at jump discontinuities, of which there is only finitely many and so do not change the value of the integral within the inner product.

Thus:
\[
  \frac{1}{L} \int_{0}^{L} f(\theta) \overline{g(\theta)} \d{\theta} = \sum_{n \in \Z} \hat{f}_n \overline{\hat{g}_n}
\]
which is \textit{Parseval's theorem}.
In the case of $g = f$, we obtain:
\[
  \frac{1}{L} \int_{0}^{L} |f(\theta)|^2 \d{\theta} = \sum_{n \in \Z} |\hat{f}_n|^2
\]
\begin{example}
  Let $f: \R \to \R$ be $2\pi$-periodic with $f(\theta) = \theta$ on $[-\pi, \pi)$.
  Calculating the Fourier coefficients we see that for $n = 0$:
  \[
    \hat{f}_0 = \int_{-\pi}^{\pi} \theta \d{\theta} = 0
  \]
  and for $n \geq 1$:
  \begin{align*}
    \hat{f}_n &= \frac{1}{2\pi} \int_{-\pi}^{\pi} \theta e^{-i n \theta} \d{\theta} \\
              &= -\frac{1}{2\pi i n}\left(\eval{\theta e^{-in\theta}}{-\pi}{\pi} - \int_{-\pi}^{\pi} e^{-i n \theta} \d{\theta}\right) \\
              &= -\frac{1}{2 \pi i n} \cdot 2 \pi (-1)^n \\
              &= \frac{i(-1)^n}{n}
  \end{align*}
  Thus $f$ has Fourier series:
  \[
    f(\theta) \sim \sum_{n \neq 0} \frac{i(-1)^{n}}{n} e^{i n \theta}
  \]

  To apply Parseval's theorem with $f$ and itself:
  \[
    \frac{1}{2\pi} \int_{-\pi}^{\pi} |\theta|^2 \d{\theta} = \frac{1}{\pi} \int_{0}^{\pi} \theta^2 \d{\theta} = \frac{\pi^2}{3}
  \]
  and:
  \[
    \sum_{n \in \Z} |\hat{f}_n|^2 = \sum_{n \neq 0} \frac{1}{n^2} = 2 \sum_{n = 1}^{\infty} \frac{1}{n^2}
  \]
  which yields the solution to the \textit{Basel problem}:
  \[
    \sum_{n = 1}^{\infty} \frac{1}{n^2} = \frac{\pi^2}{6}
  \]
\end{example}
