\section{Spanning Sets}
\begin{definition}[Span]
  We define the \textit{span} of a subset $S \subseteq V$ to be:
  \[
    \langle S \rangle = \left\{\finitesum_{s \in S} \lambda_s s: \lambda_s \in \F\right\}
  \]
\end{definition}
\begin{remark}[Notation]
  We introduce the notation:
  \[
    \finitesum_{s \in S} \lambda_s
  \]
  to mean a finite sum over elements of $S$.
  That is, we require that $\lambda_s = 0$ for all but finitely many elements $s \in S$.
\end{remark}
\begin{remark}
  By convention we define $\langle \emptyset \rangle = \{\vec{0}_{V}\}$.
\end{remark}
\begin{definition}
We define the following for an $\F$-vector space $V$ and a subset $S \subseteq V$:
  \begin{enumerate}
    \item We call expressions of the form:
      \[
        \finitesum_{s \in S} \lambda_s s \text{ for } \lambda_s \in \F
      \]
      \textit{linear combinations} of elements from $S$.
    \item We say that $S$ \textit{spans} V (or that $S$ is a \textit{spanning set} for $V$) if $\langle S \rangle = V$.
    \item We say that $V$ is \textit{finite dimensional} if $V$ has a finite spanning set.
  \end{enumerate}
\end{definition}
\begin{lemma}
  For an $\F$-vector space $V$ and subsets $S, T \subseteq V$: \label{spanningLemma}
  \begin{enumerate}
    \item $\langle S \rangle \leq V$
    \item If $S$ spans $V$ and $S \subseteq \langle T \rangle$, then $T$ spans $V$.
    \item $\langle S \cup T \rangle = \langle S \rangle + \langle T \rangle$.
  \end{enumerate}
\end{lemma}
\begin{proof}[\textbf{i}]
  Clearly $\vec{0} \in \langle S \rangle$ by setting $\lambda_S = 0\ \forall s \in S$.
  For $v_1, v_2 \in \langle S \rangle$ and $\lambda \in \F$, clearly $v_1 + \lambda v_2 \in \langle S \rangle$ as we can just regroup the terms to be a sum of scalar multiples of elements of $S$.
  So $\langle S \rangle \leq V$ by the subspace test (\cref{subspaceTest}).
\end{proof}
\begin{proof}[\textbf{ii}]
  By \textbf{(i)}, $\langle T \rangle \leq V$ and also since $\langle T \rangle$ is a vector space $\langle S \rangle \leq \langle T \rangle$.
  Thus:
  \[
    V = \langle S \rangle \leq \langle T \rangle \leq V \implies \langle T \rangle = V
  \]
  So $T$ spans $V$ also.
\end{proof}
\begin{proof}[\textbf{iii}]
  For any $s_1, \ldots, s_n \in S$, $t_1, \ldots, t_n \in T$, and $\lambda_1, \ldots, \lambda_n, \mu_1, \ldots, \mu_k \in \F$ we can write:
  \begin{align*}
    v &= \lambda_1 s_1 + \cdots + \lambda_n s_n + \mu_1 t_1 + \cdots + \mu_k t_k \in \langle S \cup T \rangle\\
      &= (\lambda_1 s_1 + \cdots + \lambda_n s_n) + (\mu_1 t_1 + \cdots + \mu_k t_k) \in \langle S \rangle + \langle T \rangle
  \end{align*}
  This is the form of all elements in either of the sets and so $\langle S \cup T \rangle = \langle S \rangle + \langle T \rangle$.
\end{proof}
\begin{example}
  \begin{enumerate} \label{linIndepEx}
    \item Consider the vector space $V = \R^3$ and the subsets:
      \[
        S = \left\{\begin{pmatrix}
        1 \\
        0 \\
        0 \\
        \end{pmatrix}, \begin{pmatrix}
        1 \\
        1 \\
        2 \\
        \end{pmatrix}\right\} \text{ and } T = \left\{
        \begin{pmatrix}
        2 \\
        1 \\
        2 \\
        \end{pmatrix}, \begin{pmatrix}
        0 \\
        1 \\
        2 \\
        \end{pmatrix}, \begin{pmatrix}
        -1 \\
        2 \\
        4 \\
        \end{pmatrix}\right\}
      \]
      These subsets have:
      \[
        \langle S \rangle = \langle T \rangle = \left\{\begin{pmatrix}
        x \\
        y \\
        2y \\
        \end{pmatrix}: x, y \in \R\right\} \leq \R^3
      \]
    \item Recall that $V = \R^{\N}$ is the space of all functions $\N \to \R$.
      Consider the subset $T = \{\delta_n: n \in \N\} \subseteq V$ where:
      \[
        \delta_n: \N \to \R,\ m \mapsto \begin{cases}
        1 & m = n \\
        0 & m \neq n
        \end{cases}
      \]
      You might at first think that $\langle T \rangle = \R^{\N}$, however this is \textbf{not} the case.
      For instance, if $f \in \R^{\N}$ has $f(m) = 1$ for all $m$, then $f \notin \langle T \rangle$ as can only take linear combinations of \textbf{finitely many} elements of $T$.

      Because spanning sets only include finite linear combinations, we in fact have:
      \[
        \langle T \rangle = \{g \in V: g(n) = 0 \text{ for all $n$ sufficiently large}\}
      \]
  \end{enumerate}
\end{example}
\begin{definition}[Linear Independence]
  We say that a subset $S \subseteq V$ is \textit{linearly independent} if for every linear combination:
  \[
    \finitesum_{s \in S} \lambda_s s  = \vec{0} \implies \lambda_s = 0\ \forall s \in S
  \]
  That is, the only way to write $\vec{0}$ as a linear combination of elements of $S$ is the trivial linear combination where all coefficients are 0.

  If a subset is not linearly independent, then it is said to be \textit{linearly dependent}.
\end{definition}
\begin{remark}[Remarks]
  \begin{itemize}
    \item If $\vec{0} \in S$, then $S$ is linearly dependent as we always have the non-trivial linear combination $1 \cdot \vec{0} = \vec{0}$.
    \item If $S = \{v_1, \ldots, v_n\}$ is finite, then $S$ is linearly independent if and only if for all $\lambda_1, \ldots, \lambda_n \in \F$:
      \[
        \sum_{i = 1}^{n} \lambda_i v_i = \vec{0} \implies \lambda_1 = \cdots = \lambda_n = 0
      \]
    \item If $S$ is infinite, then $S$ is linearly independent if and only if every finite subset of $S$ is linearly independent.
    \item Every subset of a linearly independent set is linearly independent.
    \item By convention, we consider the empty set $\emptyset$ to be linearly independent.
  \end{itemize}
\end{remark}
\begin{example}
  \begin{enumerate}
    \item Returning to \cref{linIndepEx} \textbf{(i)}, we see that $T$ is linearly dependent as:
      \[
        1 \cdot \begin{pmatrix}
        2 \\
        1 \\
        2 \\
        \end{pmatrix} - 5 \cdot \begin{pmatrix}
        0 \\
        1 \\
        2 \\
        \end{pmatrix} + 2 \cdot \begin{pmatrix}
        -1 \\
        2 \\
        4 \\
        \end{pmatrix} = \vec{0}
      \]
      But $S$ is linearly independent as for $\lambda, \mu \in \R$:
      \[
        \lambda \begin{pmatrix}
        1 \\
        0 \\
        0 \\
        \end{pmatrix} + \mu \begin{pmatrix}
        1 \\
        1 \\
        2 \\
        \end{pmatrix} = \vec{0} \implies \lambda = \mu = 0
      \]
    \item Returning to \cref{linIndepEx} \textbf{(ii)} where $T = \{\delta_n: n \in \N\}$.
      Let the function $f$ be an arbitrary linear combination of elements of $T$:
      \[
        f = \finitesum_{n \in \N} \lambda_n \delta_n
      \]
      If $f \equiv 0$, then for any $m \in \N$:
      \[
        0 = f(m) = \finitesum_{n \in \N} \lambda_n \delta_n(m) = \lambda_m
      \]
      Thus $\lambda_m = 0$ for all $m$ and so $T$ is linearly independent.
  \end{enumerate}
\end{example}
\section{Bases}
\begin{definition}[Basis]
  We say that the subset $S \subseteq V$ is a basis for $V$ if it is linearly independent and spans $V$.
\end{definition}
\begin{example} \label{basesExamples}
  \begin{enumerate}
    \item Let $\vec{e}_i = (0, \ldots, 0, 1, 0, \ldots, 0)^{\trans} \in \F^{n}$ where $1$ is in the $i$-th position.
      The set $\{\vec{e}_i: i \in \{1, \ldots, n\}\}$ is called the \textit{standard basis} for $\F^{n}$.
    \item The set:
      \[
        \left\{\begin{pmatrix}
        1 \\
        0 \\
        0 \\
        \end{pmatrix}, \begin{pmatrix}
        1 \\
        1 \\
        0 \\
        \end{pmatrix}, \begin{pmatrix}
        1 \\
        1 \\
        1 \\
        \end{pmatrix}\right\}
      \]
      is a basis for $\F^3$.
    \item Recall that $P(\R)$ is the space of real polynomial functions.
      For $n \geq 0$, define the function $p_n \in P(\R)$ by:
      \[
        p_n: \R \to \R,\ x \mapsto x^n
      \]
      The set $\{p_n: n \geq 0\}$ is then a basis for $P(\R)$.
  \end{enumerate}
\end{example}
\begin{proposition}
  If $S \subseteq V$ is a finite spanning set for $V$, then there exists a subset $S' \subseteq S$ such that $S'$ is a basis for $V$. \label{finiteSpanningSub}
\end{proposition}
\begin{proof}
  Since $S$ spans $V$, by well ordering, there exists a smallest subset $S' \subseteq S$ such that $S'$ spans $V$.
  If $S' = \emptyset$, then we are done as then $v = \{\vec{0}\}$, in which case $\emptyset$ is a basis for $V$.
  Otherwise, since $S$ is finite, so is $S'$ and so we can write $S' = \{v_1, \ldots, v_n\}$.

  Suppose for contradiction that $S'$ does not span $V$.
  Thus there exists $\lambda_1, \ldots, \lambda_n \in \F$ which are not all 0 such that:
  \[
    \sum_{i = 1}^{n} \lambda_i v_i = \vec{0}
  \]
  Without loss of generality we can assume that $\lambda_1 \neq 0$ and so:
  \[
    v_1 = - \frac{1}{\lambda_1} \sum_{i = 2}^{n} \lambda_i v_i \in \langle S' \setminus \{v_1\} \rangle
  \]
  Therefore $S' \subseteq \langle S' \setminus \{v_1\} \rangle$ and so we can apply \cref{spanningLemma} \textbf{(ii)} to conclude that $S' \setminus \{v_1\}$ spans $V$, however this is a contradiction as $S' \setminus \{v_1\}$ is smaller than $S_1$.

  Thus $S'$ must span $V$ and so is a basis for $V$.
\end{proof}
\begin{remark}
  We could also view the above proof as a process where starting from $S = \{v_1, \ldots, v_n\}$, if the current set is linearly dependent, then we can remove a vector from it and still have it span $V$.
  We then repeat this until we get a basis for $V$, or end up with $\emptyset$ which is a basis for $\{\vec{0}\}$.
\end{remark}
\begin{corollary}
  A vector space $V$ is finite dimensional if and only if it has a finite basis. \label{finiteVBasis}
\end{corollary}
\begin{proof}
  \begin{proofdirection}{Assume $V$ is finite dimensional}
    So we have a finite spanning set $S$ and therefore by \cref{finiteSpanningSub} we have a finite basis $S' \subseteq S$.
  \end{proofdirection}
  \begin{proofdirection}{Assume $V$ has a finite basis}
    This basis must span $V$ so $V$ has a finite spanning set and thus is finite.
  \end{proofdirection}
\end{proof}
\begin{remark}[Warning]
  Whilst all finite vector spaces must have a basis, we require the \textit{axiom of choice} to show that every infinite vector space has a basis.
\end{remark}
\section{Steinitz Exchange Lemma}
\begin{theorem}[Steinitz Exchange Lemma]
  If $V$ is an $\F$-vector space, with finite subsets $T, S \subseteq V$ where $S$ is linearly independent and $T$ is a spanning set for $V$, then: \label{steinitz}
  \begin{enumerate}
    \item $|S| \leq |T|$
    \item There exists a subset $T' \subseteq T$ such that $S \cup T'$ spans $V$ and
      \[
        |T'| = |T| - |S|
      \]
  \end{enumerate}
\end{theorem}
\begin{remark}[Intuition]
  We can remove certain elements of $T$ and replace them by elements of $S$ and still have the resulting set span $V$.
\end{remark}
\begin{proof}
  Since $S$ and $T$ are finite $|S| = m$, $|T| = n$ and so we can write $S = \{s_1, \ldots, s_m\}$ and $T = \{t_1, \ldots, t_n\}$.
  If $S = \emptyset$, then we are done by taking $T' = T$.
  Otherwise we proceed as follows.

  \textbf{Step 1}\par
  Since $\langle T \rangle = V$, we can express the elements of $S$ as linear combinations of elements in $T$.
  Thus there exists $\lambda_i \in \F$ such that:
  \[
    s_1 = \sum_{i = 1}^{N} \lambda_i t_i
  \]
  We note that $s_1 \neq \vec{0}$ since $S$ is linearly independent and so without loss of generality we can assume that $\lambda_1 \neq 0$.
  Hence:
  \[
    t_1 = \frac{1}{\lambda_1}\left(s_1 - \sum_{i = 2}^{n} \lambda_i t_i\right)
  \]
  so $\{s_1, t_2, \ldots, t_n\}$ spans $T$ and hence spans $V$ by \cref{spanningLemma} \textbf{(ii)}.

  \textbf{Step $k + 1$ for $k < m$}\par
  We assume that $\{s_1, \ldots, s_k, t_{k + 1}, \ldots, t_n\}$ spans $V$.
  We then write:
  \[
    s_{k + 1} = \sum_{i = k + 1}^{n} \lambda_i t_i + \sum_{j = 1}^{k} \mu_i s_j
  \]
  We want to make sure that we can always substitute out a $t_i$.
  Suppose that we are not able to do that, i.e. all of the $\lambda_i$ are 0.
  We then have:
  \[
    s_{k + 1} - \sum_{i = 1}^{k} \mu_j s_j = \vec{0}
  \]
  which is a contradiction as this is a non-trivial linear combination of elements of $S$.
  So without loss of generality we can assume $\lambda_{k + 1} \neq 0$ and so:
  \[
    t_{k + 1} = \frac{1}{\lambda_{k + 1}}\left(s_{k + 1} - \sum_{i = k + 2}^{n} \lambda_i t_i - \sum_{j = 1}^{k} \mu_j s_j\right)
  \]
  hence $\{s_1, \ldots, s_{k + 1}, t_{k + 2}, \ldots, t_n\}$ spans $V$.

  \textbf{Conclusion}\par
  Suppose that this process terminates with us running out of elements of $T$ before we run out of elements of $S$, i.e. $m > n$, then we would have that $\{s_1, \ldots, s_n\}$ spans $V$, which would contradict the linear independence of $V$ as we could write $s_m$ as a linear combination of $\{s_1, \ldots, s_n\}$.

  Thus we must have $m \leq n$ i.e. $|S| \leq |T|$ and so the terminating set is of the form $\{s_1, \ldots, s_m, t_{m + 1}, \ldots, t_n\}$.
  This spans $V$ and so picking $T' = \{t_{m + 1}, \ldots, t_n\} \subseteq T$ works.
\end{proof}
\begin{corollary}
  If $V$ is a finite dimensional $\F$-vector space, then: \label{basisSize}
  \begin{enumerate}
    \item Every basis for $V$ is finite
    \item If $B_1$ and $B_2$ are finite bases for $V$, then $|B_1| = |B_2|$
  \end{enumerate}
\end{corollary}
\begin{proof}[\textbf{i}]
  By \cref{finiteVBasis}, $V$ has a finite basis $B$.
  Suppose for contradiction that $B'$ is an infinite basis for $V$.
  Since $B'$ is a basis, we can pick $|B| + 1$ elements from $B$ to construct a $B'' \subseteq B'$ with $|B''| = |B| + 1$ that is linearly independent.
  Then $S = B''$ is linearly independent and $T = B$ spans $V$, so by \cref{steinitz}, $|B| + 1 = |S| \leq |T| = |B|$, which is a contradiction.
\end{proof}
\begin{proof}[\textbf{ii}]
  Since $B_1$ and $B_2$ are bases, they are both linearly independent and both span $V$.

  Setting $S = B_1$ and $T = B_2$, gives $|B_1| \leq |B_2|$ and similarly taking $S = B_2$ and $T = B_1$ gives $|B_2| \leq |B_1|$.
  Thus $|B_1| = |B_2|$ as required.
\end{proof}
\begin{definition}[Dimension]
  The \textit{dimension}, denoted $\dim(V)$, of a finite dimensional vector space $V$ is the common size of any basis for $V$.
\end{definition}
\begin{example}
  Returning to \cref{basesExamples} \textbf{(i)}, since we were able to find an $n$ element basis for $\F^n$, we know that $\dim(\F^n) = n$.
\end{example}
\begin{corollary}
  If $V$ is a \textbf{finite dimensional} $\F$-vector space, with finite subsets $T, S \subseteq V$ where $S$ is linearly independent and $T$ is a spanning set for $V$, then: \label{dimIneq}
  \begin{enumerate}
    \item $|S| \leq \dim(V) \leq |T|$
    \item $|S| = \dim(V)$ if and only if $S$ is a basis
    \item $|T| = \dim(V)$ if and only if $T$ is a basis
  \end{enumerate}
\end{corollary}
\begin{remark}
  So if we know the dimension of $V$, for example by finding a ``simple'' basis for $V$, then if we have another set of size $\dim(V)$, we need only check that it is either linearly independent or spanning as the other follows by the above.
\end{remark}
\begin{proof}[\textbf{i}]
  Since $V$ is finite, by \cref{finiteVBasis}, there exists a finite basis $B$ for $V$.
  $B$ is a basis so is both linearly independent and spanning.
  We can there fore apply \cref{steinitz} \textbf{(i)} with $S, B$ and $B, T$ giving:
  \[
    |S| \leq |B| = \dim(V) = |B| \leq |T|
  \]
  as required.
\end{proof}
\begin{proof}[\textbf{ii}]
  \begin{proofdirection}{Assume $S$ is a basis}
    So by \cref{basisSize}, $|S| = \dim(V)$.
  \end{proofdirection}
  \begin{proofdirection}{Assume $|S| = \dim(V)$}
    As in \textbf{(i)}, there is a basis $B$ for $V$ which must have $|B| = \dim(V) = |S|$
    Since $S$ is linearly independent we can apply \cref{steinitz} \textbf{(ii)} to $S$ and $B$ and so there exists $B' \subseteq B$ such that $S \cup B'$ spans V and $|B'| = |B| - |S| = 0$.
    But $|B'| = 0 \implies B' = \emptyset$, so $S$ spans $V$ and is therefore a basis.
  \end{proofdirection}
\end{proof}
\begin{proof}[\textbf{iii}]
  \begin{proofdirection}{Assume $T$ is a basis}
    So by \cref{basisSize}, $|T| = \dim(V)$.
  \end{proofdirection}
  \begin{proofdirection}{Assume $|T| = \dim(V)$}
    Since $T$ is a finite spanning set of $V$, by \cref{finiteSpanningSub}, there must be a subset $T' \subseteq T$ such that $T'$ is a basis for $V$.
    Thus $|T'| = \dim(V) = |T| \implies T' = T$, so $T$ is a basis for $V$.
  \end{proofdirection}
\end{proof}
\begin{proposition}
  If $V$ is a finite dimensional $\F$-vector space and $U \leq V$, then: \label{finiteSubspace}
  \begin{enumerate}
    \item $\dim(U) \leq \dim(V)$, so $U$ is finite dimensional
    \item $U = V$ if and only if $\dim(U) = \dim(V)$
  \end{enumerate}
\end{proposition}
\begin{proof}[\textbf{i}]
  If $U = \{\vec{0}\}$, then there is nothing to prove.
  \quietinduction
  {$n = 1$}{
    Since $U \neq \{\vec{0}\}$, we can take some $u_1 \in U \setminus \{\vec{0}\}$ and so the set $S_1 = \{u_1\}$ is linearly independent.
  }
  {$n = k$}{
    So we assume that there exists a set $S_k = \{u_1, \ldots, u_k\} \subseteq U$ of size $k$ which is linearly independent.

    By \cref{dimIneq} \textbf{(i)}, $k \leq \dim(V)$.
    If $\langle S_k \rangle = U$, then $S_k$ is a finite basis for $U$ with $\dim(U) = k \leq \dim(V)$ and so we terminate the process.
    Otherwise we proceed as follows.
  }
  {$n = k + 1$}{
    We can take some $u_{k + 1} \in U \setminus \langle S_k \rangle$ and then let $S_{k + 1} = \{u_1, \ldots, u_k, u_{k+ 1}\}$.
    Suppose, for contradiction, that $S_{k + 1}$ is linearly dependent, so there exists $\lambda_1, \ldots, \lambda_{k + 1} \in \F$ not all 0 such that:
    \[
      \vec{0} = \sum_{i = 1}^{k + 1} \lambda_i u_i
    \]
    If $\lambda_{k + 1} = 0$, then we would have have a non-trivial linear combination of vectors from $S_k$ equal to $\vec{0}$, contradicting the linear independence  of $S_k$.
    We must therefore have $\lambda_{k + 1} \neq 0$ and so:
    \[
      u_{k + 1} = - \frac{1}{\lambda_{k + 1}} \sum_{i = 1}^{k} \lambda_i u_i \in \langle S_k \rangle
    \]
    which is a contradiction as $u_{k + 1} \notin \langle S_k \rangle$ by construction.
    Thus $S_{k + 1}$ is linearly independent.
  }
\end{proof}
\begin{proof}[\textbf{ii}]
  \begin{proofdirection}{Assume $U = V$}
    Clearly both have the same dimension as basis sizes are unique by \cref{basisSize}.
  \end{proofdirection}
  \begin{proofdirection}{Assume $\dim(U) = \dim(V)$}
    So there is a basis $B$ for $U$ of size $\dim(V)$.
    But $B \subseteq U \subseteq V$ and $B$ is linearly independent, so by \cref{dimIneq} \textbf{(ii)}, $B$ is a basis for $V$.
    Thus $U = \langle B \rangle = V$.
  \end{proofdirection}
\end{proof}
\begin{proposition}[Extending a Basis]
  Let $V$ be a finite dimensional $\F$-vector space and $U \leq V$.
  If $B_U$ is a basis for $U$, then there exists a basis $B_V$ for $V$ such that $B_U \subseteq B_V$. \label{basisExtension}
\end{proposition}
\begin{proof}
  Since $V$ is finite dimensional, by \cref{finiteVBasis}, there exists a finite basis $T$ for $V$.
  We then apply Steinitz-Exchange (\cref{steinitz}) with $S = B_U$ and $T$ to obtain a subset $T' \subseteq T$ such that $B_U \cup T'$ spans $V$ and $|T'| = |T| - |B_U| = \dim(V) - \dim(U)$.
  We then have:
  \[
    |B_U \cup T'| \leq |B_U| + |T'| = \dim(U) + (\dim(V) - \dim(U)) = \dim(V)
  \]
  So $B_U \cup T'$ spans $V$ and has $|B_U \cup T'| = \dim(V)$, thus by \cref{dimIneq} \textbf{(iii)}, $B_U \cup T'$ is a basis for $V$.
  Thus taking $B_V = B_U \cup T'$ has $B_U \subseteq B_V$.
\end{proof}
\section{Bases and Linear Maps}
\begin{definition}[Rank and Nullity]
  For a linear map $\alpha: V \to W$ we define its \textit{rank} as:
  \[
    \rank \alpha = \dim(\im \alpha)
  \]
  and its \textit{nullity} as:
  \[
    \nullity \alpha = \dim(\ker \alpha)
  \]
\end{definition}
\begin{remark}[Recap]
  Recall from \cref{imKerProp} that for a linear map $\alpha: V \to W$, we have $\ker \alpha \leq V$ and $\im \alpha \leq W$.
\end{remark}
\begin{theorem}[Rank-Nullity Theorem]
  Let $V$ and $W$ be $\F$-vector spaces such that $V$ is finite dimensional.
  If $\alpha: V \to W$ is a linear map, then $\im \alpha$ is finite dimensional and: \label{RNT}
  \[
    \dim(V) = \rank \alpha + \nullity \alpha
  \]
\end{theorem}
\begin{remark}[Intuition]
  The kernel of $\alpha$ can be thought of as the ``information'' lost by $\alpha$ and the image of $\alpha$ can be thought of as the ``information'' preserved by $\alpha$, so it makes sense that the sum of the dimensions of these two quantities should be the dimension of the domain of the map $V$.
\end{remark}
\begin{proof}
  Since $\ker \alpha \leq V$, by \cref{finiteSubspace}, $\ker \alpha$ is finite dimensional and so there exists a basis $\{v_1, \ldots, v_k\}$ for $\ker \alpha$.
  Using \cref{basisExtension}, we extend this to a basis $\{v_1, \ldots, v_k, \ldots, v_n\}$ for $V$.
  We claim that $B = \{\alpha(v_{k + 1}), \ldots, \alpha(v_n)\}$ is a basis for $\im \alpha$.

  \textbf{Spanning}\par
  For any $w \in \im \alpha$, there exists $v \in V$ such that $w = \alpha(v)$.
  Since $\{v_1, \ldots, v_n\}$ spans $V$, there exists $\lambda_1, \ldots, \lambda_n \in \F$ such that:
  \[
    v = \sum_{i = 1}^{n} \lambda_i v_i,\ \lambda_i \in \F
  \]
  and so we can rewrite $\alpha(w)$ as:
  \begin{align*}
    \alpha(w) &= \alpha\left(\sum_{i = 1}^{n} \lambda_i v_i\right) \\
              &= \sum_{i = 1}^{n} \lambda_i \alpha(v_i) \\
              &= \sum_{i = k + 1}^{n} \lambda_i \alpha(v_i)
  \end{align*}
  since $v_i \in \ker \alpha$ for $i \in \{1, \ldots, k\}$.
  Thus $\langle B \rangle \leq \im \alpha \leq \langle B \rangle \implies \im \alpha = \langle B \rangle$.

  \textbf{Linear Independence}\par
  For linear independence, suppose that there exists $\lambda_{k + 1}, \ldots, \lambda_n$ such that:
  \begin{align*}
    \vec{0} = \sum_{i = k + 1}^{n} \lambda_i \alpha(v_i) = \alpha\left(\sum_{i = k +1}^{n} \lambda_i v_i\right) \implies \sum_{i = k + 1}^{n} \lambda_i v_i \in \ker \alpha
  \end{align*}
  Thus as $\{v_1, \ldots, v_k\}$ are a basis for $\ker \alpha$, there exists $\lambda_1, \ldots, \lambda_k \in \F$ such that:
  \[
    \sum_{i = k + 1}^{n} \lambda_i v_i = - \sum_{i = 1}^{k} \lambda_i v_i \implies \sum_{i = 1}^{n} \lambda_i v_i = \vec{0}
  \]
  Since $\{v_1, \ldots, v_n\}$ is a basis for $V$, we must have $\lambda_1 = \cdots = \lambda_k = \lambda_{k + 1} = \cdots = \lambda_n = 0$.
  Thus $B$ is linearly independent.

  \textbf{Conclusion}\par
  So we have found a finite basis $B$ for $\im \alpha$ with $\rank \alpha = |B| = n - k = \dim(V) - \im \alpha$, as required.
\end{proof}
\begin{corollary}
  If $V$ is a finite dimensional $\F$-vector space and $U \leq V$, then $V / U$ is finite dimensional with: \label{quotDim}
  \[
    \dim(V / U) = \dim(V) - \dim(U)
  \]
\end{corollary}
\begin{proof}[Using rank nullity theorem]
  Recall from \cref{quotMapProp} that the quotient map:
  \[
    \pi_U: V \to V / U,\ v \mapsto v + U
  \]
  is surjective with $\ker \pi_U = U$.
  Applying \cref{RNT} to this, we obtain that $\im \pi_{U} = V / U$ is finite dimensional and:
  \[
    \dim(V) = \rank \pi_U + \im \pi_U = \dim(U) + \dim(V / U)
  \]
  as required.
\end{proof}
We can also prove the above directly.
Note that this is almost exactly the proof for the rank nullity theorem but with $\alpha$ replaced by $\pi_U$.
\begin{proof}[Without using rank nullity theorem]
  Since $U \leq V$, by \cref{finiteSubspace}, $U$ is finite dimensional and so there exists a basis $B_U = \{v_1, \ldots, v_k\}$ for $U$.
  Using \cref{basisExtension}, we extend this to a basis $B_V = \{v_1, \ldots, v_k, \ldots, v_n\}$ for $V$.
  We then construct the set:
  \[
    B_{V / U} = \{v_{k + 1} + U, \ldots, v_n + U\} \subseteq V / U
  \]
  and claim that $B_{V / U}$ is a basis for $V / U$.

  \textbf{Spanning}\par
  For any $v + U \in V / U$, since $v \in V$ and $\{v_1, \ldots, v_n\}$ spans $V$, there exists $\lambda_1, \ldots, \lambda_n \in \F$ such that:
  \[
    v = \sum_{i = 1}^{n} \lambda_i v_i,\ \lambda_i \in \F
  \]
  and so we can rewrite $v + U$ as:
  \[
    v + U = \left(\sum_{i = 1}^{n} \lambda_i v_i\right) + U = \sum_{i = k + 1}^{n} \lambda_u (v_i + U)
  \]
  as $v_i \in U \implies v_i + U = U$ (see \cref{cosetRelation}) for $i \in \{1, \ldots, k\}$.
  Thus $\langle B_{V / U} \rangle = V / U$

  \textbf{Linear Independence}\par
  For linear independence, suppose that there exists $\lambda_{k + 1}, \ldots, \lambda_n$ such that:
  \begin{align*}
    \vec{0} + U = \sum_{i = k + 1}^{n} \lambda_i (v_i + U) = \left(\sum_{i = k + 1}^{n} \lambda_i v_i\right) + U \implies \sum_{i = k + 1}^{n} \lambda_i v_i \in U
  \end{align*}
  Thus as $B_U = \{v_1, \ldots, v_k\}$ are a basis for $U$, there exists $\lambda_1, \ldots, \lambda_k \in \F$ such that:
  \[
    \sum_{i = k + 1}^{n} \lambda_i v_i = - \sum_{i = 1}^{k} \lambda_i v_i \implies \sum_{i = 1}^{n} \lambda_i v_i = \vec{0}
  \]
  Since $B_V = \{v_1, \ldots, v_n\}$ is a basis for $V$, we must have $\lambda_1 = \cdots = \lambda_k = \lambda_{k + 1} = \cdots = \lambda_n = 0$.
  Thus $B_{V / U}$ is linearly independent.

  \textbf{Conclusion}\par
  So we have found a finite basis $B_{V / U}$ for $V / U$ with $\dim(V / U) = |B_{V / U}| = n - k = \dim(V) - \dim(U)$, as required.
\end{proof}
\begin{corollary}[Linear pidgeonhole principle]
  If $V$ and $W$ are finite dimensional $\F$-vector spaces with $\dim(V) = \dim(W)$ and $\alpha: V \to W$ is linear, then the following are equivalent:
  \begin{enumerate}
    \item $\alpha$ is injective
    \item $\alpha$ is surjective
    \item $\alpha$ is an isomorphism
  \end{enumerate}
\end{corollary}
\begin{remark}[Intuition]
  This is analogous to how we saw in IA Numbers and Sets that functions between finite sets of the same size are injective if and only if they are surjective.
\end{remark}
\begin{proof}
  From \cref{imKerProp}:
  \[
    \textbf{(i)} \iff \ker \alpha = \{\vec{0}\} \iff \nullity \alpha = 0
  \]
  and:
  \[
    \textbf{(ii)} \iff \im \alpha = W \iff \rank \alpha = \dim(W)
  \]
  where the last equivalence follows from \cref{finiteSubspace} \textbf{(ii)}.

  Rank-nullity theorem (\cref{RNT}) tells us that:
  \[
    \dim(W) = \dim(V) = \rank \alpha + \nullity \alpha
  \]
  and so:
  \[
    \rank \alpha = \dim(W) \iff \nullity \alpha = 0
  \]
  That is, \textbf{(i)} $\iff$ \textbf{(ii)} $\iff$ \textbf{(iii)} by definition of an isomorphism (\cref{isoDef}).
\end{proof}
\begin{proposition}
  Let $V$ and $W$ be $\F$-vector spaces, which are not necessarily finite dimensional.
  If $B$ is a for $V$ and $f: B \to W$ is a function, then there exists a unique linear map $F: V \to W$ extending $f$.
  That is, the restriction of the domain of $F$ to $B$ is exactly $f$, $\at{F}{B} = f$.
\end{proposition}
\begin{remark}[Intuition]
  If two linear maps $F$ and $G$ have $\at{F}{B} = \at{G}{B}$, i.e. they act on a basis the same way, then $F = G$ by the above.
  So linear maps are uniquely determined by their action on a basis.
\end{remark}
\begin{proof}\par
  \textbf{Existence}\par
  For any $v \in V$, since $B$ spans $V$, there exists $\lambda_b \in \F$ for $b \in B$ such that:
  \begin{equation}
    v = \finitesum_{b \in B} \lambda_b b \label{actionUniqueDag} \tag{\dag}
  \end{equation}
  We then define the map $F: V \to W$ using these coefficients as:
  \[
    F(V) = \finitesum_{b \in B} \lambda_b f(b)
  \]
  \textbf{Well Defined}\par
  We want to show that our choice of coefficients in \cref{actionUniqueDag} was unique as then $F$ is well defined.
  Suppose that we can also write $v = \sum_{b \in B} \mu_B b$ for coefficients $\mu_b \in \F$ for $b \in B$.
  We then have:
  \[
    \vec{0} = \sum_{b \in B} (\lambda_b - \mu_b) b
  \]
  So by linear independence of $B$, $\lambda_b = \mu_b$ for all $b \in B$ and so the coefficients are unique.

  \textbf{Linearity}\par
  For any $v, u \in V$ we can write $v = \finitesum_{b \in B} \lambda_b b$ and $u = \finitesum_{b \in B} \mu_b b$ with $\lambda_b, \mu_b \in \F$ for $b \in B$.
  For $\gamma \in \F$ we have:
  \begin{align*}
    F(u + \gamma v) &= F\left(\finitesum_{b \in B} (\mu_b + \gamma \lambda_b) v\right) \\
                     &= \finitesum_{b \in B} (\mu_b + \gamma \lambda_b) f(v) \text{ by definition} \\
                     &= \finitesum_{b \in B} \mu_b f(v) + \gamma \finitesum_{b \in B} \lambda_b f(v) \\
                     &= F(u) + \gamma F(v)
  \end{align*}
  and so $F$ is linear.

  \textbf{Uniqueness}\par
  If $\overline{F}: V \to W$ is linear with $\at{\overline{F}}{B} = f$, then for any $v \in V$ we have:
  \[
    \overline{F}(v) = \overline{F}\left(\finitesum_{b \in B} \lambda_b b\right)  = \finitesum_{b \in B} \lambda_B \overline{F}(b) = \finitesum_{b \in B} \lambda_b f(b) = F(v)
  \]
  and so $\overline{F} = F$.
\end{proof}
\begin{remark}
  During the proof of the above we also showed the useful fact that the coefficients of basis vectors in a linear combination are unique.
\end{remark}
