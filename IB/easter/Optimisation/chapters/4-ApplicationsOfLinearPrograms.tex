\chapter{Applications of Linear Programs}
\section{Game Theory}
\subsection{Two Person Games}
Consider a game with two players where $P_1$ can pick from any of $\{1, \ldots, m\}$ actions and $P_2$ can pick from any of $\{1, \ldots, n\}$ actions.
\begin{definition}[Payoff Matrix]
  The \textit{payoff matrix} $A$ for $P_1$ is an $m \times n$ matrix with entries $A_{i j}$, where $A_{i j}$ is the score attained by $P_1$ if $P_1$ plays action $i$ and $P_2$ plays action $j$.
\end{definition}
\begin{example}
  For a game of rock paper scissors, each player can pick either rock (call this 1), paper (call this 2), or scissors (call this 3).
  The payoff matrix for $P_1$ is:
  \[
    A = \begin{pmatrix}
    0 & -1 & 1 \\
    1 & 0 & -1 \\
    -1 & 1 & 0 \\
    \end{pmatrix}
  \]
\end{example}
If $P_1$ plays $i$ then $P_1$ can expect to have payoff:
\[
  i \mapsto \min_{j \in \{1, \ldots, n\}} A_{i j}
\]
$P_1$ now tries to solve:
\[
  \max_{i \in \{1, \ldots, m\}} \min_{j \in \{1, \ldots, n\}} A_{i j}
\]
and $P_2$ will try to solve:
\[
  \min_{j \in \{1, \ldots, n\}} \max_{i \in \{1, \ldots, m\}} A_{i j}
\]
\begin{example}
  Consider the payoff matrix:
  \[
    A = \begin{pmatrix}
    1 & 2 \\
    3 & 4 \\
    \end{pmatrix}
  \]
  For this payoff matrix, $P_1$ decides they should play $i = 2$ and $P_2$ decides they should play $j = 1$ and so both their decisions coincide on the cell $A_{2 1} = 3$.

  Now consider the payoff matrix:
  \[
    A' = \begin{pmatrix}
    4 & 2 \\
    1 & 3 \\
    \end{pmatrix}
  \]
  In this case $P_1$ will play $i = 2$ and then will expect $P_2$ to choose row $j = 1$.
  However $P_2$ will chose $j = 2$ and expect $P_1$ to choose $i = 2$.
  Unlike with $A$, the choice of the two players does not coincide.
  In games where players alternate turns, this is ok, however in games where both players play simultaneously, the players end up stuck in a loop.
\end{example}
\begin{definition}[Strategy]
  A \textit{strategy} is a probability distribution for the actions that a player picks.
\end{definition}
\begin{definition}[Pure and Mixed Strategies]
  A \textit{pure strategy} is a strategy where one single action has probability 1, i.e. the player will always pick a particular action.
  If a strategy is not pure, then we call it a \textit{mixed strategy}.
\end{definition}
Let the strategy for $P_1$ be $i$ with probability $p_i$ and let the strategy for $P_2$ be $j$ with probability $q_i$.

$P_1$'s expected score is this then:
\[
  \sum_{i = 1}^{m} p_i A_{i j}
\]
So for optimal play, $P_1$ wants to adopt the strategy $\vec{p} = (p_1, \ldots, p_m)^{\trans}$ that maximises:
\[
  \max_{\vec{p} \in P_m} \left(\min_{j \in \{1, \ldots, n\}} \sum_{i = 1}^{m} p_i A_{i j}\right)
\]
where $P_m$ is the set of discrete probability distributions for $\{1, \ldots, m\}$.
\subsection{Linear Programming for Strategies}
We can express $P_1$'s optimisation problem as:
\begin{maxi}
  {\R}{v}{}{}
  \addConstraint{A^{\trans} \vec{p}}{\geq v e}
  \addConstraint{\vec{e}^{\trans} \vec{p}}{= 1}
  \addConstraint{\vec{p}}{\geq \vec{0}}
\end{maxi}
where $v$ will be the maximal score for $P_1$ and $\vec{e} = (1, \ldots, 1)^{\trans} \in \R^{m}$.
The $\vec{e}^{\trans} \vec{p} = 1$ constraint ensures that $\sum_{i} p_i = 1$ and the $\vec{p} \geq 0$ constraint ensures that $p_i \geq 0\ \forall i$ so that the $p_i$ are a valid probability distribution.

Similarly for $P_2$, we can express their optimisation problem as:
\begin{mini*}
  {\R}{w}{}{}
  \addConstraint{A\vec{q}}{\leq w \vec{e}'}
  \addConstraint{\vec{e}'^{\trans} \vec{p}}{= 1}
  \addConstraint{\vec{q}}{\geq \vec{0}}
\end{mini*}
where $w$ will be the maximal score for $P_2$ and $\vec{e}' = (1, \ldots, 1)^{\trans} \in \R^{n}$

These two problems actually turn out to be dual to one another.
To find the dual of $P_2$'s problem, we find the Lagrangian:
\[
  L(w, \vec{q}, \vec{s}, \vec{\lambda}_1, \lambda_2) = w + \vec{\lambda}^{\trans}_1(A\vec{q} + \vec{s} - w \vec{e}) - \lambda_2(\vec{e}^{\trans} \vec{q} - 1)
\]
The set $\Lambda$ is then:
\[
  \Lambda = \{\vec{\lambda}_1: \vec{\lambda}^{\trans}_1 \vec{e} = 1,\ \vec{\lambda}^{\trans}_1 A \geq \lambda_2\vec{e}^{\trans}, \vec{\lambda_1} \geq \vec{0}\}
\]
and then when $\vec{\lambda} \in \Lambda$, $\min L = \lambda_2$.
Thus the dual of the problem is:
\begin{maxi*}
  {\R}{\lambda_2}{}{}
  \addConstraint{\vec{\lambda}^{\trans}_1 \vec{e}}{= 1}
  \addConstraint{\vec{\lambda}_1}{\geq \vec{0}}
  \addConstraint{A^{\trans} \vec{\lambda_1}}{\geq \lambda_2 \vec{e}'}
\end{maxi*}
Upon relabelling $\lambda_2$ by $v$ and $\vec{\lambda}_1$ by $\vec{p}$ we obtain $P_1$'s optimisation problem.
\begin{theorem}[Optimal Strategies]
  A strategy $\vec{p}$ is optimal for $P_1$ if and only if there exists $\vec{q}$ and $v$ such that:
  \begin{enumerate}
    \item $A^{\trans} \vec{p} \geq v \vec{e}$, $\vec{e}^{\trans} p = 1$, $\vec{p} \geq \vec{0}$ (Primal feasibility)
    \item $A\vec{q} \leq v \vec{e}$, $\vec{e}^{\trans} \vec{q} = 1$, $\vec{q} \geq \vec{0}$ (Dual feasibility)
    \item $v = \vec{p}^{\trans} A \vec{q}$ (Complimentary slackness)
  \end{enumerate}
\end{theorem}
\begin{proof}
  We need only check the complimentary slackness equations.
  $(\vec{p}, v)$ and $(\vec{q}, w)$ are primal/dual optimal if and only if both:
  \[
    (A \vec{q} - w \vec{e})^{\trans} \vec{p} = 0 \text{ and } \vec{q}^{\trans}(A^{\trans} \vec{p} - v\vec{e}) = 0
  \]
  which is equivalent to $v = w = \vec{p}^{\trans} A \vec{q}$.
\end{proof}
\subsection{Finding Optimal Strategies}
\begin{enumerate}
  \item Look for a ``saddle point'', if such a point exists then a pure strategy is optimal.
  \item Look for ``dominating actions''.
    These are actions which a player would obviously not play.
    For example, suppose for $P_1$ $\exists i_1, i_2$ such that $A_{i_1 j} \geq A_{i_2 j}$ for all $j$.
    In this case, we can simply drop row $i_2$.
  \item Formula and solve the linear program, either graphically or by using the simplex method.
\end{enumerate}
\begin{example}
  Consider the payoff matrix for $P_1$:
  \[
    A = \begin{pmatrix}
    2 & 3 & 4 \\
    3 & 1 & 1/2 \\
    1 & 3 & 2 \\
    \end{pmatrix}
  \]
  We first notice that every entry in the third row is less than or equal to the first row and so we can drop the third row:
  \[
    A = \begin{pmatrix}
    2 & 3 & 4 \\
    3 & 1 & 1/2 \\
    \end{pmatrix}
  \]
  There is no saddle so $P_1$ adopts a mixed strategy $(p, 1 - p, 0)$ and thus the linear program is:
  \begin{maxi*}
    {\R}{v}{}{}
    \addConstraint{2p + 3(1 - p)}{\geq v}
    \addConstraint{3p + (1 - p)}{\geq v}
    \addConstraint{4p - \frac{1 - p}{2}}{\geq v}
    \addConstraint{0 \leq p}{\leq 1}
  \end{maxi*}
  We can solve this graphically by plotting these straight lines in the $(p, v)$-plane and finding the maximal point $v$ in the region bounded by the lines.
  After solving, we see that $p = 2/3$ and so $P_1$ adopts the mixed strategy $(2/3, 1/3, 0)$.
\end{example}
\section{Network Flows}
\subsection{Graphs}
Network flow problems are defined on \textit{graphs}.
\begin{definition}[Directed Graph]
  A \textit{directed graph} $G = (V, E)$ where $V$ is a set of \textit{vertices} and $E \subseteq V \times V$ is a set of ordered pairs of vertices called \textit{edges}, for example $(v_1, v_2)$ represents a directed edge from $v_1 \to v_2$ for $v_1, v_2 \in V$.
\end{definition}
Consider a graph $G = (V, E)$ with $n$ edges, i.e. $V = \{v_1, \ldots, v_n\}$.
\begin{definition}[Flow Vector]
  We define the flow vector $\vec{b} \in \R^{n}$ to be such that $b_i$ is the flow \textit{entering} vertex $v_i$.
\end{definition}
We also assume that the total quantity flow within the network is 0, that is:
\[
  \sum_{i = 1}^{n} b_i = 0
\]
\begin{definition}[Cost Matrix]
  The cost matrix $C$ is an $n \times n$ matrix where $C_{i j}$ is the cost per unit flow along the edge $v_i \to v_j$.
\end{definition}
\begin{definition}[Capacity Constraints]
  The $n \times n$ matrices $\overline{M}$ and $\underline{M}$ restrict the capacity through each edge, i.e. the flow along the edge $v_i \to v_j$, called $x_{i j}$ must satisfy $x_{i j} \in [\underline{M}_{i j}, \overline{M}_{i j}]$.
\end{definition}
\begin{definition}[Network Flow Problem]
  Let $G = (V, E)$  with $|V| = n$.
  The minimum cost flow is a linear program given by:
  \begin{mini*}
    {x_{i j}}{\sum_{(i, j) \in E} C_{i j}x_{i j}}{}{}
    \addConstraint{b_i + \sum_{(j, i) \in E} x_{j i}}{= \sum_{(i, j) \in E} x_{i j}}
    \addConstraint{\underline{M}_{i j} \leq x_{i j}}{\leq \overline{M}_{i j}}
  \end{mini*}
\end{definition}
\subsection{Transport Problem}
A transport problem is a special case of network flow problems where there are \textbf{no intermediate vertices}.
Suppose that we have $n$ suppliers $i \in \{1, \ldots, n\}$ and $m$ consumers $j \in \{1, \ldots, m\}$.
Supplier $i$ produces a quantity $s_i$ and consumer $j$ demands a quantity $d_j$.
\begin{definition}[Transport Problem]
  The \textit{transport problem is:}
  \begin{mini*}
    {x_{i j}}{\sum_{\substack{1 \leq i \leq n \\ 1 \leq j \leq m}} C_{i j}x_{i j}}{}{}
    \addConstraint{\sum_{j = 1}^{m} x_{i j}}{= S_i\ \forall i \in \{1, \ldots, n\}}
    \addConstraint{\sum_{i = 1}^{n} x_{i j}}{= d_j\ \forall j \in \{1, \ldots, m\}}
    \addConstraint{x_{i j}}{\geq 0\ \forall i \in {1, \ldots, n} \text{ and } j \in \{1, \ldots, m\}}
  \end{mini*}
\end{definition}
\begin{remark}
  Since $\sum_{i = 1}^{n} S_i = \sum_{i = 1}^{n} d_j$, we have some redundant constraints and  so cannot directly apply the simplex algorithm.
  We could remove the redundant constraints, however we can instead use the \textit{Transport Algorithm}, which we will introduce later.
\end{remark}
\begin{theorem}
  Every minimum cost network flow problem (i.e. $\overline{M}_{i j} < \infty\ \forall i, j$) can be recast as an equivalent transport problem.
\end{theorem}
\begin{remark}
  If $C_{i j} \geq 0\ \forall i, j$, then without loss of generality then we can assume $\overline{M}_{i j} < \infty$.
  In fact, we could set:
  \[
    \overline{M}_{i j} = \sum_{i \in V} |b_i|
  \]
  for those $(i, j)$ where $\overline{M}_{i j} = \infty$.
\end{remark}
\begin{proof}
  We first observe that we can set $\underline{M}$ to $O$ by defining $\widetilde{x}_{i j} = x_{i j} - \underline{M}_{i j}$.
  This transforms the conditions to:
  \begin{align*}
    0 \leq \widetilde{x}_{i j} &\leq \overline{M}_{i j} - \underline{M}_{i j} \\
    \sum_{i, j} C_{i j}x_{i j} &= \sum_{i, j} C_{i j} \widetilde{x}_{i j} + \sum_{i, j} \underline{M}_{i j} C_{i j}
  \end{align*}
  The last term of this is just some constant and so we can reformulate the flow condition by replacing $b_i$ by:
  \[
    \widetilde{b}_i = b_i + \sum_{(j, i) \in E} \underline{M}_{j i} - \sum_{(i, j) \in E} \underline{M}_{i j}
  \]
  and so WLOG we can assume $\underline{M} = O$.

  For every vertex $i$, construct a consumer, so we construct $|V|$ consumers.
  For every edge $(i, j)$, construct a supplier, so we construct $|E|$ suppliers.

  Now let the supply from supplier $(i, j)$ be $\overline{M}_{i j}$ and let the cost of transport from this supplier to consumer $i$ be 0, and the cost to $j$ be $C_{i j}$.
  These are the only edges we add to the graph.

  We let the amount of goods for consumer $i$ be:
  \[
    \sum_{k: (i, k) \in E} \overline{M}_{i k} - b_i
  \]
  and of consumer $j$:
  \[
    \sum_{k: (j, k) \in E} \overline{M}_{j k} - b_j
  \]

  Now suppose that we send a flow $x_{i j}$ from supplier $(i, j)$ to consumer $j$, because of the above, the flow to $j$ is then the remaining $\overline{M}_{i j} - x_{i j}$, because we made the cost to consume $i$ be 0, the overall cost is:
  \[
    \sum_{(i, j) \in E} C_{i j} x_{i j}
  \]
  Thus, the cost function in both problems is the same.

  Consumer $i$ has inward flow:
  \[
    \sum_{k: (i, k) \in E} (\overline{M}_{i k} - x_{i k}) + \sum_{k: (k, i) \in E} x_{k i}
  \]
  which must be equal to the demand, and so:
  \[
    \sum_{k: (i, k) \in E} \overline{M}_{i k} - b_i = \sum_{k: (i, k) \in E} (\overline{M}_{i k} - x_{i k}) + \sum_{k: (k, i) \in E} x_{k i}
  \]
  Upon simplifying, we obtain:
  \[
    b_i + \sum_{k: (k, i) \in E} x_{k i} = \sum_{k: (i, k) \in E} x_{i k}
  \]
  which is the same as in the original flow problem.

  So both problems have the same feasible set and cost functions, and so are equivalent.
\end{proof}
\subsection{Transport Algorithm}
\begin{theorem}[Optimality Conditions]
  If for some feasible $x$, we have dual variables $\vec{\lambda} \in \R^{n}$ and $\vec{\mu} \in \R^{m}$  such that for all $i \in \{1, \ldots, n\}$ and  $j \in \{1, \ldots, m\}$ we have:
  \begin{align*}
    C_{i j} \geq \lambda_i + \mu_j \\
    (C_{i j} - (\lambda_i + \mu_j)) x_{i j} = 0
  \end{align*}
  then $x$ is optimal.
\end{theorem}
\begin{remark}
  The components $\lambda_i$, correspond to supplier $i$ and the components $\mu_j$ correspond to consumer $j$.
\end{remark}
\begin{proof}
  The Lagrangian is given by:
  \begin{align*}
    L(x, \vec{\lambda}, \vec{\mu}) &= \sum_{i = 1}^{n} \sum_{j = 1}^{m} C_{i j}x_{i j} - \sum_{i = 1}^{n} \lambda_i \left(\sum_{j = 1}^{m} x_{i j} - S_i\right) - \sum_{j  =1}^{m} \mu_j \left(\sum_{i = 1}^{n} x_{ij} - d_j\right) \\
                                   &= \sum_{i = 1}^{n} \sum_{j = 1}^{m} (C_{i j} - \lambda_i - \mu_j)x_{i j} + \sum_{i = 1}^{n} \lambda_i S_i + \sum_{j = 1}^{m} \mu_j d_j
  \end{align*}
  The conditions in the theorem  mean that the given $(\vec{\lambda}, \vec{\mu})$ are dual feasible and complementary slackness holds.
  Thus using optimality conditions for Linear Programs, we conclude that $x$ is optimal. %TODO
\end{proof}
\begin{remark}
  For a given basic feasible solution $x$, when we solve for $(\vec{\lambda}, \vec{\mu})$ using complementary slackness, we can assume that $\lambda_1 = 0$ without loss of generality as we can just shift the other components around.
\end{remark}
\subsubsection{Transport Algorithm}
We first draw the \textit{Transport Tableau}:
\begin{center}
\begin{tabular}{c|c|c|c|c|c}
 & $\mu_1$ & $\mu_2$ & $\cdots$ & $\mu_m$ &  \\
 \hline
$\lambda_1$ & $\substack{\lambda_1 + \mu_1 \\ x_{1 1} \vert C_{1 1}}$ & $\substack{\lambda_1 + \mu_2 \\ x_{1 2} \vert C_{1 2}}$ & $\cdots$ & $\substack{\lambda_1 + \mu_m \\ x_{1 m} \vert C_{1 m}}$ & $s_1$ \\
\hline
$\lambda_1$ & $\substack{\lambda_2 + \mu_1 \\ x_{2 1} \vert C_{2 1}}$ & $\substack{\lambda_2 + \mu_2 \\ x_{2 2} \vert C_{2 2}}$ & $\cdots$ & $\substack{\lambda_2 + \mu_m \\ x_{2 m} \vert C_{2 m}}$ & $s_2$ \\
\hline
$\vdots$ & $\vdots$ & $\vdots$ & $\ddots$ & $\vdots$ & $\vdots$ \\
\hline
$\lambda_1$ & $\substack{\lambda_n + \mu_1 \\ x_{1 1} \vert C_{n 1}}$ & $\substack{\lambda_n + \mu_2 \\ x_{n 2} \vert C_{n 2}}$ & $\cdots$ & $\substack{\lambda_n + \mu_m \\ x_{n m} \vert C_{n m}}$ & $s_n$ \\
\hline
 & $d_1$ & $d_2$ & $\cdots$ & $d_m$ &
\end{tabular}
\end{center}
where numbers $d_1, \ldots, d_n$ and $s_1, \ldots, s_n$ follow directly from the problem and are not updated during the algorithm.
\begin{example}
  Consider the following transport problem:
  \begin{align*}
    S_1 &= 14,\ S_2 = 10,\ S_3 = 9 \\
    d_1 &= 12,\ d_2 = 5,\ d_3 = 8,\ d_4 = 8
  \end{align*}
  with cost matrix:
  \[
    C = \begin{pmatrix}
    5 & 3 & 4 & 6 \\
    2 & 7 & 4 & 1 \\
    5 & 6 & 2 & 4 \\
    \end{pmatrix}
  \]
  We need to find any basic feasible solution to start the algorithm.

  To find this basic feasible solution, we use the \textit{north-west corner rule}, this gives as much supply as possible to the first consumer starting with the first producer, and then as much as possible to the second, starting with the first producer and so on.
  Considering the values of $\vec{\lambda}$ and $\vec{\mu}$, we have 6 complementary slackness equations from $(C_{i j} - (\lambda_i + \mu_j))x_{i j} = 0$, corresponding to each of the non-zero $x_{i j}$.
  Recalling that we can set $\lambda_1 = 0$ without loss of generality, this along the other constraints gives:
  \[
    \vec{\lambda} = \begin{pmatrix}
    0 \\
    4 \\
    2 \\
    \end{pmatrix} \text{ and } \vec{\mu} = \begin{pmatrix}
    5 \\
    3 \\
    0 \\
    2 \\
    \end{pmatrix}
  \]
  So the intial tableau is:
  \begin{center}
  \begin{tabular}{c|c|c|c|c|c}
   & 5 & 3 & 0 & 2 &  \\
   \hline
  0 & $\substack{5 \\ 12 \vert 5}$ & $\substack{3 \\ 2 \vert 3}$ & $\substack{0 \\ 0 \vert 4}$ & $\substack{2 \\ 0 \vert 6}$ & 14 \\
  \hline
  4 & $\substack{9 \\ 0 \vert 2}$ & $\substack{7 \\ 3 \vert 7}$ & $\substack{4 \\ 7 \vert 4}$ & $\substack{6 \\ 0 \vert 1}$ & 10 \\
  \hline
  2 & $\substack{7 \\ 0 \vert 5}$ & $\substack{5 \\ 0 \vert 6}$ & $\substack{2 \\ 1 \vert 2}$ & $\substack{4 \\ 8 \vert 4}$ & 9 \\
  \hline
   & 12 & 5 & 8 & 8 & \\
  \end{tabular}
  \end{center}
  Now we are at a basic feasible solution we can start the main body of the algorithm.

  We then chose a cell with $C_{i j} < \lambda_i + \mu_j$, for example $i = 2$ and $j = 1$.
  Suppose that we sent a quantity $\theta$ along the edge between supplier $i$ and consumer $j$.
  Upon consulting the graph to find the new quantities along other edges, the smallest new quantity is $3 - \theta$.
  Sine we want to send as much down this edge as possible without violating the constraints, we pick $\theta = 3$.
  The tableaux is then updated accordingly:
  \begin{center}
  \begin{tabular}{c|c|c|c|c|c}
   & 5 & 3 & 0 & 2 &  \\
   \hline
  0 & $\substack{5 \\ 9 \vert 5}$ & $\substack{3 \\ 5 \vert 3}$ & $\substack{0 \\ 0 \vert 4}$ & $\substack{2 \\ 0 \vert 6}$ & 14 \\
  \hline
  -3 & $\substack{2 \\ 3 \vert 2}$ & $\substack{0 \\ 0 \vert 7}$ & $\substack{4 \\ 7 \vert 4}$ & $\substack{6 \\ 0 \vert 1}$ & 10 \\
  \hline
  -5 & $\substack{0 \\ 0 \vert 5}$ & $\substack{-2 \\ 0 \vert 6}$ & $\substack{2 \\ 1 \vert 2}$ & $\substack{4 \\ 8 \vert 4}$ & 9 \\
  \hline
   & 12 & 5 & 8 & 8 & \\
  \end{tabular}
  \end{center}
  We then choose another cell with $\lambda_i + \mu_j > C_{i j}$, and repeat as before.
  If there is no such cell left, the optimality conditions are satisfied and so we are done.
\end{example}
