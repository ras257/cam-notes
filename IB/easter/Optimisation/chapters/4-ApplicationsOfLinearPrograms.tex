\chapter{Applications of Linear Programs}
\section{Game Theory}
\subsection{Two Person Zero-Sum Games}
\begin{definition}[Two Person Zero-Sum Game]
  A \textit{two person zero-sum} game is an abstract game with two players in which $P_1$ can pick from any of $\{1, \ldots, m\}$ actions and $P_2$ can pick from any of $\{1, \ldots, n\}$ actions.
  We say the game is \textit{zero-sum} because when \textbf{$P_1$ attains score, $P_2$ loses an equal amount of score} and vice versa.
\end{definition}
\begin{remark}[Note]
  In some games players will play simultaneously and in others players will alternate turns.
\end{remark}
\begin{definition}[Payoff Matrix]
  The \textit{payoff matrix} $A$ for $P_1$ is an $m \times n$ matrix with entries $A_{i j}$, where $A_{i j}$ is the score attained by $P_1$ if $P_1$ plays action $i$ and $P_2$ plays action $j$.
\end{definition}
\begin{remark}[Remarks]
  \begin{itemize}
    \item Note that the payoff matrix $B$ for $P_2$ is also an $m \times n$ matrix however $B_{i j}$ now represents the score attained by $P_2$  if $P_1$ plays $i$ and $P_2$ plays $j$ (i.e. we only change who is getting the score and not which cell it is in).
    \item In a zero sum game, $B = -A$.
  \end{itemize}
\end{remark}
\begin{example} \label{RPSExample}
  Rock, paper, scissors in an example of a zero-sum game.
  Each player can pick either rock (call this 1), paper (call this 2), or scissors (call this 3).
  The payoff matrix for $P_1$ is:
  \[
    A = \begin{pmatrix}
    0 & -1 & 1 \\
    1 & 0 & -1 \\
    -1 & 1 & 0 \\
    \end{pmatrix}
  \]
  We will consider two versions of the game:
  \begin{itemize}
    \item Consider first a strange version of rock, paper, scissors where one player plays before the other.

      If $P_1$ starts and picks $i$, then $P_1$ knows $P_2$ will try and minimise $P_1$'s gain (or equivalently $P_2$'s loss) and so will choose $j$ to minimise $A_{i j}$.
      Thus if $P_1$ plays $i$, they expect to have a payoff of:
      \[
        \min_{j \in \{1, \ldots, n\}} A_{i j}
      \]
      Thus since $P_1$ wants to maximise their payoff over their possible choices, $P_1$ now tries to solve:
      \[
        \max_{i \in \{1, \ldots, m\}} \min_{j \in \{1, \ldots, n\}} A_{i j} = -1
      \]
      so $P_2$ wins, which makes sense as if you show your move first in rock, paper, scissors you will always lose if the other player is playing rationally.

      Similarly, if $P_2$ starts, they will try to solve:
      \[
        \min_{j \in \{1, \ldots, n\}} \max_{i \in \{1, \ldots, m\}} A_{i j} = 1
      \]
      so $P_1$ wins.
    \item Now consider the standard version of the game where both players pick simultaneously.
      In this case, the two players will get stuck in a loop indefinitely both trying to play perfectly whilst knowing the other player is also doing the same.
  \end{itemize}
\end{example}
We can formalise the idea of whether the players will get stuck in a loop using \textit{saddle points}.
\begin{lemma}
  In any payoff matrix: \label{payoffEquality}
  \[
    \alpha = \max_{i \in \{1, \ldots, m\}} \min_{j \in \{1, \ldots, n\}} A_{i j} \leq \min_{j \in \{1, \ldots, n\}} \max_{i \in \{1, \ldots, m\}} A_{i j} = \beta
  \]
\end{lemma}
\begin{proof}
  \nonexaminable
  For any $i, j$  we have:
  \[
    \min_{j \in \{1, \ldots, n\}} A_{i j} \leq A_{i j} \leq \max_{i \in \{1, \ldots, m\}} A_{i j} \implies \min_{j \in \{1, \ldots, n\}} A_{i j} \leq \max_{i \in \{1, \ldots, m\}} A_{i j}\ \forall i, j
  \]
  Since this is true for all $i$ we can take a maximum of the left hand side over $i$ whilst preserving the inequality:
  \[
    \alpha = \max_{i \in \{1, \ldots, m\}} \min_{j \in \{1, \ldots, n\}} A_{i j} \leq \max_{i \in \{1, \ldots, m\}} A_{i j}\ \forall j
  \]
  and since this is true for all $j$ we can take a minimum of the right hand side over $j$ whilst preserving the inequality:
  \[
    \alpha = \max_{i \in \{1, \ldots, m\}} \min_{j \in \{1, \ldots, n\}} A_{i j} \leq \min_{j \in \{1, \ldots, n\}}  \max_{i \in \{1, \ldots, m\}} A_{i j} = \beta
  \]
  as required.
\end{proof}
\subsection{Saddle Points}
\begin{definition}[Saddle Point]
  For a payoff matrix $A$, we say that $(k, l)$ is a \textit{saddle point} if in the payoff matrix $A_{k l}$ is \textbf{an entry that is both the minimum of its row and the maximum of its column}:
  \[
    A_{k l} = \min_{j \in \{1, \ldots, n\}} A_{k j} = \max_{i \in \{1, \ldots, m\}} A_{i l}
  \]
  That is, $k$ is the best response for $P_1$ if $P_2$ picks $l$ \textbf{and} $l$ is the best response for $P_2$ if $P_1$ picks $k$.

  We then call the value $A_{k l}$ the \textit{value} of the game.
\end{definition}
\begin{proposition}
  A pay off matrix $A$ has a saddle point if and only if:
  \[
    \alpha = \max_{i \in \{1, \ldots, m\}} \min_{j \in \{1, \ldots, n\}} A_{i j} = \min_{j \in \{1, \ldots, n\}} \max_{i \in \{1, \ldots, m\}} A_{i j} = \beta
  \]
  This shared value is also the value of the game.
\end{proposition}
\begin{proof}
  \nonexaminable
  \begin{proofdirection}{Assume $A$ has a saddle point}
    We have have:
    \[
      \alpha \geq \min_{j \in \{1, \ldots, n\}} A_{k j} = A_{k l} = \max_{i \in \{1, \ldots, m\}} A_{i l} \geq \beta \implies \alpha \geq \beta
    \]
    But from \cref{payoffEquality} we also have $\alpha \leq \beta$ and so $\alpha = \beta = A_{k l}$.
  \end{proofdirection}
  \begin{proofdirection}{Assume $\alpha = \beta$}
    By definition of $\alpha$, there exists $k \in \{1, \ldots, m\}$ such that:
    \[
      \alpha = \min_{j \in \{1, \ldots, n\}} A_{k j} \leq A_{k j}\ \forall j
    \]
    Similarly, by definition of $\beta$, there exists $l \in \{1, \ldots, n\}$ such that:
    \[
      \beta = \max_{i \in \{1, \ldots, m\}} A_{i l} \geq A_{i l}\ \forall i
    \]
    Combining, there exists $k, l$ such that:
    \[
      A_{i l} \leq \alpha = \beta \leq A_{k j}\ \forall i, j
    \]
    Taking $i = k$ and $j = l$ we get $A_{k l} = \alpha = \beta$ and so:
    \[
      A_{k l} = \min_{j \in \{1, \ldots, n\}} A_{k j} = \max_{i \in \{1, \ldots, m\}} A_{i l}
    \]
    i.e. $(k, l)$ is a saddle point.
  \end{proofdirection}
\end{proof}
\begin{remark}[Remarks]
  \begin{itemize}
    \item In the reverse direction above there could be multiple choices for the saddle point and so a game can have multiple saddle points, however they must all have the same value.
    \item There can be entries of the payoff matrix which have the value of the game but are not saddle points.
  \end{itemize}
\end{remark}
\begin{example}
  Consider the payoff matrix:
  \[
    A = \begin{pmatrix}
    1 & 2 \\
    3 & 4 \\
    \end{pmatrix}
  \]
  For this payoff matrix:
  \begin{itemize}
    \item $P_1$ decides they should play $i = 2$ and expects $P_2$ to play $j = 1$.
    \item $P_2$ decides they should play $j = 1$ and expects $P_1$ to play $i = 2$.
  \end{itemize}
  Thus $(2, 1)$ is a saddle point of $A$ and the value of the game is $A_{2 1} = 3$.

  Since there is a saddle point, with these player strategies:
  \begin{itemize}
    \item If the game is played with alternating turns, it \textbf{does not} matter who starts.
    \item The game can be played simultaneously without the players getting stuck in a loop trying to find the optimal strategy.
  \end{itemize}
\end{example}
\begin{example}
  Consider the payoff matrix:
  \[
    A' = \begin{pmatrix}
    4 & 2 \\
    1 & 3 \\
    \end{pmatrix}
  \]
  For this payoff matrix:
  \begin{itemize}
    \item $P_1$ decides they should play $i = 1$ and expects $P_2$ to play $j = 2$.
    \item $P_2$ decides they should play $j = 2$ and expects $P_1$ to play $i = 2$.
  \end{itemize}
  So $A'$ does not have a saddle point, this means that with these strategies:
  \begin{itemize}
    \item If the game is played with alternating turns, it \textbf{does matter} who starts.
    \item The game \textbf{cannot} be played simultaneously without the players getting stuck in a loop trying to find the optimal strategy.

      For example, from $P_1$'s perspective, they might decide from the above that they should play $i = 1$, but then will realise that $P_2$ would pick $j = 2$.
      But now given that $P_2$ will play $j = 2$, it would now be better for $P_1$ to choose $i = 2$.
      However, $P_2$ would also realise this and so would pick $j = 1$, \textit{and so on...}
  \end{itemize}
\end{example}
\begin{remark}
  The payoff matrix for the rock, paper, scissors game in \cref{RPSExample} does not have a saddle point since:
  \[
    \max_{i \in \{1, \ldots, m\}} \min_{j \in \{1, \ldots, n\}} A_{i j} \neq \min_{j \in \{1, \ldots, n\}} \max_{i \in \{1, \ldots, m\}} A_{i j}
  \]
  which is why the players get stuck when trying to play simultaneously.
\end{remark}
\subsection{Strategies}
\begin{definition}[Strategy]
  A \textit{strategy} is a probability distribution for the actions that a player picks.
\end{definition}
\begin{definition}[Pure and Mixed Strategies]
  A \textit{pure strategy} is a strategy where one single action has probability 1, i.e. the player will always pick a particular action.
  If a strategy is not pure, then we call it a \textit{mixed strategy}.
\end{definition}
\begin{remark}
  In the previous section, the players were using a pure strategy.
  With mixed strategy, the players actions have probabilities and so games with a saddle point can played simultaneously whilst still allowing for optimal play.
\end{remark}
We represent the mixed strategy for $P_1$ as a vector $\vec{p} \in \R^{m}$ where $p_i \in [0, 1]$ is the discrete probability distribution that $P_1$ picks action $i$ and similarly represent the mixed strategy for $P_2$ with the vector $\vec{q} \in \R^{n}$.

Given that $P_2$ picks $j$, $P_1$'s expected score is:
\[
  \sum_{i = 1}^{m} p_i A_{i j}
\]
From $P_1$'s perspective, they expect $P_2$ to pick to minimise this expected score over $j$ and so for optimal play, $P_1$ wants to adopt the mixed strategy $\vec{p}$ that solves:
\begin{equation}
  \max_{\vec{p} \in P_m} \left(\min_{j \in \{1, \ldots, n\}} \sum_{i = 1}^{m} p_i A_{i j}\right) \label{p1prob}
\end{equation}
where $P_m$ is the set of all discrete probability distributions on $\{1, \ldots, m\}$.

Similarly, from $P_2$'s perspective, they want to adopt the mixed strategy $\vec{q}$ that solves:
\[
  \min_{\vec{q} \in P_n} \left(\max_{i \in \{1, \ldots, m\}} \sum_{j = 1}^{n} q_j A_{i j}\right)
\]
Note that min and max are swapped because $A_{i j}$ is the gain for $P_1$ and loss for $P_2$.
\subsection{Linear Programming for Strategies}
We can express the optimisation problems for each player as linear programs.

Considering $P_1$'s problem from \cref{p1Prob}, we first note that we can express the expectation as:
\[
  \sum_{i = 1}^{m} p_i A_{i j} = [A^{\trans} \vec{p}]_j
\]
We want to maximise over $\vec{p}$ the minimum over $j$ of this.
The minimum over $j$ of this will be the smallest component of $A^{\trans} \vec{p} \in \R^{n}$.
The solution is therefore the largest $v \in \R$ over all $\vec{p} \in \R^{m}$ such that \textbf{every component} of $A^{\trans} \vec{p}$ is bigger than $v$.
This means that the limiting component is the smallest component of $A^{\trans} \vec{p}$, i.e. the inner minimum.
Therefore $P_1$'s problem is equivalent to the linear program:
\begin{maxi}
  {v \in \R, \vec{p} \in \R^{m}}{v}{}{\label{p1Lin}}
  \addConstraint{A^{\trans} \vec{p}}{\geq v \vec{1}_n}
  \addConstraint{\vec{1}^{\trans}_m \vec{p}}{= 1}
  \addConstraint{\vec{p}}{\geq \vec{0}}
\end{maxi}
where $\vec{1}_k = (1, \ldots, 1)^{\trans} \in \R^{k}$.
The optimal solution $\vec{p}^{*}$ is the best mixed strategy for $P_1$ and the optimal value is its expected score.
The other two constraints ensure that $p_i$ is a valid probability distribution since $\vec{p}$ is a valid probability distribution if and only if $\vec{1}^{\trans}_m \vec{p} = 1$ and $\vec{p} \geq 0$.

Similarly for $P_2$, we can express their problem as:
\begin{mini}
  {w \in \R, \vec{q} \in \R^{n}}{w}{}{\label{p2Lin}}
  \addConstraint{A\vec{q}}{\leq w \vec{1}_m}
  \addConstraint{\vec{1}^{\trans}_n \vec{q}}{= 1}
  \addConstraint{\vec{q}}{\geq \vec{0}}
\end{mini}
\begin{theorem}
  The optimisation problems for $P_1$, in \cref{p1Lin}, and $P_2$, in \cref{p2Lin}, are dual. \label{gameDual}
\end{theorem}
\begin{proof}
  To find the dual of $P_2$'s problem, we first add slack variables:
  \begin{mini*}
    {w \in \R, \vec{q} \in \R^{n}, \vec{s} \in \R^{m}}{w}{}{}
    \addConstraint{A\vec{q} + \vec{s}}{= w \vec{1}_m}
    \addConstraint{\vec{1}^{\trans}_n \vec{q}}{= 1}
    \addConstraint{\vec{q}}{\geq \vec{0}}
    \addConstraint{\vec{s}}{\geq \vec{0}}
  \end{mini*}
  We then package all of our variables up into $(w, \vec{q}, \vec{s})$ and introduce parameters $\vec{\lambda}_1 \in \R^{m}$ and $\lambda_2 \in \R$ to construct the Lagrangian:
  \begin{align*}
    L(w, \vec{q}, \vec{s}, \vec{\lambda}_1, \lambda_2) &= w - \vec{\lambda}^{\trans}_1(A\vec{q} + \vec{s} - w \vec{1}_m) - \lambda_2(\vec{1}^{\trans}_n \vec{q} - 1) \\
                                                       &= (1 + \vec{\lambda}^{\trans}_1 \vec{1}_m) w - (\vec{\lambda}^{\trans}_1 A + \lambda_2 \vec{1}^{\trans}_n) \vec{q} - \vec{\lambda}^{\trans}_1 \vec{s} + \lambda_2
  \end{align*}
  So the dual objective function is:
  \[
    g(\vec{\lambda}_1, \lambda_2) = \min_{w \in \R, \vec{q} \geq \vec{0}, \vec{s} \geq \vec{0}} [(1 + \vec{\lambda}^{\trans}_1 \vec{1}_m) w - (\vec{\lambda}^{\trans}_1 A + \lambda_2 \vec{1}^{\trans}_n) \vec{q} - \vec{\lambda}^{\trans}_1 \vec{s} + \lambda_2]
  \]
  The set $\Lambda$ for which the dual objective function is then:
  \[
    \Lambda = \{(\vec{\lambda}_1, \lambda_2): -\vec{\lambda}^{\trans}_1 \vec{1}_m = 1,\ -\vec{\lambda}^{\trans}_1 A \geq \lambda_2\vec{1}^{\trans}_n, -\vec{\lambda_1} \geq \vec{0}\}
  \]
  and then when $\vec{\lambda} \in \Lambda$, $g(\vec{\lambda}_1, \lambda_2) = \lambda_2$.

  Therefore, the dual of the problem is:
  \begin{maxi*}
    {\lambda_2 \in \R, \vec{\lambda}_1 \in \R^{m}}{\lambda_2}{}{}
    \addConstraint{- \vec{\lambda}^{\trans}_1 A}{\geq \lambda_2 \vec{1}^{\trans}_n}
    \addConstraint{-\vec{\lambda}^{\trans}_1 \vec{1}_m}{= 1}
    \addConstraint{-\vec{\lambda}_1}{\geq \vec{0}}
  \end{maxi*}
  We then relabel $\lambda_2 \to v$, and let $\vec{p} = -\vec{\lambda}_1$.
  Finally, we take transposes of the first two constraints to arrive at:
  \begin{maxi*}
    {v \in \R, \vec{p} \in \R^{m}}{v}{}{}
    \addConstraint{A^{\trans} \vec{p}}{\geq v \vec{1}_n}
    \addConstraint{\vec{1}^{\trans}_m \vec{p}}{= 1}
    \addConstraint{\vec{p}}{\geq \vec{0}}
  \end{maxi*}
  which is as in \cref{p1Lin}.
\end{proof}
\begin{corollary}[Optimal Strategies]
  A strategy $\vec{p}$ is optimal for $P_1$ if and only if there exists $\vec{q}$ and $v$ such that: \label{optimalStrategies}
  \begin{enumerate}
    \item $A^{\trans} \vec{p} \geq v \vec{1}_n$, $\vec{1}^{\trans}_m \vec{p} = 1$, $\vec{p} \geq \vec{0}$ (Primal feasibility)
    \item $A\vec{q} \leq v \vec{1}_m$, $\vec{1}^{\trans}_n \vec{q} = 1$, $\vec{q} \geq \vec{0}$ (Dual feasibility) -- Note that $v$ is repeated here.
    \item $v = \vec{p}^{\trans} A \vec{q}$ (Complementary slackness)
  \end{enumerate}
\end{corollary}
\begin{proof}
  Using \cref{optimalityConds}, we need only check the complementary slackness equations.
  We know from \cref{optimalityConds} that $(\vec{p}, v)$ and $(\vec{q}, w)$ are primal/dual optimal if and only if both:
  \[
    (A \vec{q} - w \vec{1}_m)^{\trans} \vec{p} = 0 \text{ and } \vec{q}^{\trans}(A^{\trans} \vec{p} - v\vec{1}_n) = 0
  \]
  We see that since $\vec{1}^{\trans}_m \vec{p} = 1$:
  \[
    (A \vec{q} - w \vec{1}_m)^{\trans} \vec{p} = 0 \iff \vec{q}^{\trans} A^{\trans} \vec{p} = w \vec{1}^{\trans}_m \vec{p} = w
  \]
  and since $\vec{q}^{\trans} \vec{1}_n = 1$:
  \[
    \vec{q}^{\trans}(A^{\trans} \vec{p} - v\vec{1}_n) = 0 \iff \vec{q}^{\trans} A^{\trans} \vec{p} = v \vec{q}^{\trans} \vec{1}_n = v
  \]
  Thus since $\vec{q}^{\trans} A^{\trans} \vec{p} = \vec{p}^{\trans} A \vec{q}$:
  \[
    (A \vec{q} - w \vec{1}_m)^{\trans} \vec{p} = 0 \text{ and } \vec{q}^{\trans}(A^{\trans} \vec{p} - v\vec{1}_n) = 0 \iff v = w = \vec{p}^{\trans} A \vec{q}
  \]
\end{proof}
\begin{remark}
  Since $v = w$, the expected payoff for both players is always the same.
\end{remark}
\subsection{Finding Optimal Strategies}
We can use the following steps to find optimal strategies for each player.
\begin{enumerate}
  \item Look for a saddle point of the payoff matrix, if such a point exists then a pure strategy where both players always pick this point is optimal.
  \item Simplify the payoff matrix by looking for actions which a player would obviously not play and so we can remove their corresponding rows/columns.

    For example, suppose for $P_1$ that $\exists i_1, i_2$ such that $A_{i_1 j} \geq A_{i_2 j}$ for all $j$.
    There would never be a reason for $P_1$ to pick $i_2$ as its payoff is worse regardless of $P_2$'s choice and so we can simply drop row $i_2$.
  \item Formulate and solve the linear program for $P_1$, either graphically or by using the simplex method.
    This yields $P_1$'s strategy and the expected payoff for both players.
  \item Use complementary slackness and \cref{optimalStrategies} to determine $P_2$'s strategy.
\end{enumerate}
\begin{example}
  Consider the payoff matrix for $P_1$:
  \[
    A = \begin{pmatrix}
    2 & 3 & 4 \\
    3 & 1 & 1/2 \\
    1 & 3 & 2 \\
    \end{pmatrix}
  \]
  We first notice that every entry in the third row is less than or equal to the first row and so we can drop the third row:
  \[
    A = \begin{pmatrix}
    2 & 3 & 4 \\
    3 & 1 & 1/2 \\
    \end{pmatrix}
  \]
  There is no saddle point since:
  \[
    \max_{i \in \{1, \ldots, m\}} \min_{j \in \{1, \ldots, n\}} A_{i j} = 2 \neq 3 = \min_{j \in \{1, \ldots, n\}} \max_{i \in \{1, \ldots, m\}} A_{i j}
  \]
  A mixed strategy is therefore optimal and so $P_1$ adopts the mixed strategy $(p, 1 - p, 0)$ and thus the linear program is:
  \begin{maxi*}
    {p \in \R, v \in \R}{v}{}{}
    \addConstraint{2p + 3(1 - p)}{\geq v}
    \addConstraint{3p + (1 - p)}{\geq v}
    \addConstraint{4p + \frac{1}{2}(1 - p)}{\geq v}
    \addConstraint{0 \leq p}{\leq 1}
  \end{maxi*}
  In this case we do not need to use the simplex method and we can just solve this graphically in $\R^2$.
  To do this we plot the feasible set in the $(p, v)$-plane and find the maximal point $v$ within this region.

  After solving, we see that $p = \frac{2}{3}$ and so $P_1$ adopts the mixed strategy $\left(\frac{2}{3}, \frac{1}{3}, 0\right)$ with expected payoff $v = \frac{7}{3}$.
  From \cref{gameDual}, we know that the problem for $P_2$'s strategy is dual to this and so we can use complementary slackness to find their strategy.

  We first note that $2p + 3(1 - p) = \frac{7}{3}$ and $3p + (1 - p) = \frac{7}{3}$ but $4p + \frac{1}{2}(1 - p) \geq \frac{7}{3}$, thus the first two constraints are tight but there is still slack in the third constraint.
  Thus we must have $q_3 = 0$ in the dual and so we can write $P_2$'s mixed strategy as $(q, 1 - q, 0)$ and use $\vec{p}^{\trans} A \vec{q} = \frac{7}{3}$ to find $q$, yielding $q = \frac{2}{3}$.
  So $P_2$ adopts the same mixed strategy as $P_1$.
\end{example}
\section{Network Flows}
\subsection{Graphs}
Network flow problems are defined on \textit{graphs}.
\begin{definition}[Directed Graph]
  A \textit{directed graph} $G = (V, E)$ where $V$ is a set of \textit{vertices} and $E \subseteq V \times V$ is a set of ordered pairs of vertices called \textit{edges}, for example $(v_1, v_2)$ represents a directed edge from $v_1 \to v_2$ for $v_1, v_2 \in V$.
\end{definition}
Consider a graph $G = (V, E)$ with $n$ edges, i.e. $V = \{v_1, \ldots, v_n\}$.
\begin{definition}[Flow Vector]
  We define the flow vector $\vec{b} \in \R^{n}$ to be such that $b_i$ is the flow \textit{entering} vertex $v_i$.
\end{definition}
We also assume that the total quantity flow within the network is 0, that is:
\[
  \sum_{i = 1}^{n} b_i = 0
\]
\begin{definition}[Cost Matrix]
  The cost matrix $C$ is an $n \times n$ matrix where $C_{i j}$ is the cost per unit flow along the edge $v_i \to v_j$.
\end{definition}
\begin{definition}[Capacity Constraints]
  The $n \times n$ matrices $\overline{M}$ and $\underline{M}$ restrict the capacity through each edge, i.e. the flow along the edge $v_i \to v_j$, called $x_{i j}$ must satisfy $x_{i j} \in [\underline{M}_{i j}, \overline{M}_{i j}]$.
\end{definition}
\begin{definition}[Network Flow Problem]
  Let $G = (V, E)$  with $|V| = n$.
  The minimum cost flow is a linear program given by:
  \begin{mini*}
    {x_{i j}}{\sum_{(i, j) \in E} C_{i j}x_{i j}}{}{}
    \addConstraint{b_i + \sum_{(j, i) \in E} x_{j i}}{= \sum_{(i, j) \in E} x_{i j}}
    \addConstraint{\underline{M}_{i j} \leq x_{i j}}{\leq \overline{M}_{i j}}
  \end{mini*}
\end{definition}
\subsection{Transport Problem}
A transport problem is a special case of network flow problems where there are \textbf{no intermediate vertices}.
Suppose that we have $n$ suppliers $i \in \{1, \ldots, n\}$ and $m$ consumers $j \in \{1, \ldots, m\}$.
Supplier $i$ produces a quantity $s_i$ and consumer $j$ demands a quantity $d_j$.
\begin{definition}[Transport Problem]
  The \textit{transport problem is:}
  \begin{mini*}
    {x_{i j}}{\sum_{\substack{1 \leq i \leq n \\ 1 \leq j \leq m}} C_{i j}x_{i j}}{}{}
    \addConstraint{\sum_{j = 1}^{m} x_{i j}}{= S_i\ \forall i \in \{1, \ldots, n\}}
    \addConstraint{\sum_{i = 1}^{n} x_{i j}}{= d_j\ \forall j \in \{1, \ldots, m\}}
    \addConstraint{x_{i j}}{\geq 0\ \forall i \in {1, \ldots, n} \text{ and } j \in \{1, \ldots, m\}}
  \end{mini*}
\end{definition}
\begin{remark}
  Since $\sum_{i = 1}^{n} S_i = \sum_{i = 1}^{n} d_j$, we have some redundant constraints and  so cannot directly apply the simplex algorithm.
  We could remove the redundant constraints, however we can instead use the \textit{Transport Algorithm}, which we will introduce later.
\end{remark}
\begin{theorem}
  Every minimum cost network flow problem (i.e. $\overline{M}_{i j} < \infty\ \forall i, j$) can be recast as an equivalent transport problem.
\end{theorem}
\begin{remark}
  If $C_{i j} \geq 0\ \forall i, j$, then without loss of generality then we can assume $\overline{M}_{i j} < \infty$.
  In fact, we could set:
  \[
    \overline{M}_{i j} = \sum_{i \in V} |b_i|
  \]
  for those $(i, j)$ where $\overline{M}_{i j} = \infty$.
\end{remark}
\begin{proof}
  We first observe that we can set $\underline{M}$ to $O$ by defining $\widetilde{x}_{i j} = x_{i j} - \underline{M}_{i j}$.
  This transforms the conditions to:
  \begin{align*}
    0 \leq \widetilde{x}_{i j} &\leq \overline{M}_{i j} - \underline{M}_{i j} \\
    \sum_{i, j} C_{i j}x_{i j} &= \sum_{i, j} C_{i j} \widetilde{x}_{i j} + \sum_{i, j} \underline{M}_{i j} C_{i j}
  \end{align*}
  The last term of this is just some constant and so we can reformulate the flow condition by replacing $b_i$ by:
  \[
    \widetilde{b}_i = b_i + \sum_{(j, i) \in E} \underline{M}_{j i} - \sum_{(i, j) \in E} \underline{M}_{i j}
  \]
  and so WLOG we can assume $\underline{M} = O$.

  For every vertex $i$, construct a consumer, so we construct $|V|$ consumers.
  For every edge $(i, j)$, construct a supplier, so we construct $|E|$ suppliers.

  Now let the supply from supplier $(i, j)$ be $\overline{M}_{i j}$ and let the cost of transport from this supplier to consumer $i$ be 0, and the cost to $j$ be $C_{i j}$.
  These are the only edges we add to the graph.

  We let the amount of goods for consumer $i$ be:
  \[
    \sum_{k: (i, k) \in E} \overline{M}_{i k} - b_i
  \]
  and of consumer $j$:
  \[
    \sum_{k: (j, k) \in E} \overline{M}_{j k} - b_j
  \]

  Now suppose that we send a flow $x_{i j}$ from supplier $(i, j)$ to consumer $j$, because of the above, the flow to $j$ is then the remaining $\overline{M}_{i j} - x_{i j}$, because we made the cost to consume $i$ be 0, the overall cost is:
  \[
    \sum_{(i, j) \in E} C_{i j} x_{i j}
  \]
  Thus, the cost function in both problems is the same.

  Consumer $i$ has inward flow:
  \[
    \sum_{k: (i, k) \in E} (\overline{M}_{i k} - x_{i k}) + \sum_{k: (k, i) \in E} x_{k i}
  \]
  which must be equal to the demand, and so:
  \[
    \sum_{k: (i, k) \in E} \overline{M}_{i k} - b_i = \sum_{k: (i, k) \in E} (\overline{M}_{i k} - x_{i k}) + \sum_{k: (k, i) \in E} x_{k i}
  \]
  Upon simplifying, we obtain:
  \[
    b_i + \sum_{k: (k, i) \in E} x_{k i} = \sum_{k: (i, k) \in E} x_{i k}
  \]
  which is the same as in the original flow problem.

  So both problems have the same feasible set and cost functions, and so are equivalent.
\end{proof}
\subsection{Transport Algorithm}
\begin{theorem}[Optimality Conditions]
  If for some feasible $x$, we have dual variables $\vec{\lambda} \in \R^{n}$ and $\vec{\mu} \in \R^{m}$  such that for all $i \in \{1, \ldots, n\}$ and  $j \in \{1, \ldots, m\}$ we have:
  \begin{align*}
    C_{i j} \geq \lambda_i + \mu_j \\
    (C_{i j} - (\lambda_i + \mu_j)) x_{i j} = 0
  \end{align*}
  then $x$ is optimal.
\end{theorem}
\begin{remark}
  The components $\lambda_i$, correspond to supplier $i$ and the components $\mu_j$ correspond to consumer $j$.
\end{remark}
\begin{proof}
  The Lagrangian is given by:
  \begin{align*}
    L(x, \vec{\lambda}, \vec{\mu}) &= \sum_{i = 1}^{n} \sum_{j = 1}^{m} C_{i j}x_{i j} - \sum_{i = 1}^{n} \lambda_i \left(\sum_{j = 1}^{m} x_{i j} - S_i\right) - \sum_{j  =1}^{m} \mu_j \left(\sum_{i = 1}^{n} x_{ij} - d_j\right) \\
                                   &= \sum_{i = 1}^{n} \sum_{j = 1}^{m} (C_{i j} - \lambda_i - \mu_j)x_{i j} + \sum_{i = 1}^{n} \lambda_i S_i + \sum_{j = 1}^{m} \mu_j d_j
  \end{align*}
  The conditions in the theorem  mean that the given $(\vec{\lambda}, \vec{\mu})$ are dual feasible and complementary slackness holds.
  Thus using optimality conditions for Linear Programs, we conclude that $x$ is optimal. %TODO
\end{proof}
\begin{remark}
  For a given basic feasible solution $x$, when we solve for $(\vec{\lambda}, \vec{\mu})$ using complementary slackness, we can assume that $\lambda_1 = 0$ without loss of generality as we can just shift the other components around.
\end{remark}
\subsubsection{Transport Algorithm}
We first draw the \textit{Transport Tableau}:
\begin{center}
\begin{tabular}{c|c|c|c|c|c}
 & $\mu_1$ & $\mu_2$ & $\cdots$ & $\mu_m$ &  \\
 \hline
$\lambda_1$ & $\substack{\lambda_1 + \mu_1 \\ x_{1 1} \vert C_{1 1}}$ & $\substack{\lambda_1 + \mu_2 \\ x_{1 2} \vert C_{1 2}}$ & $\cdots$ & $\substack{\lambda_1 + \mu_m \\ x_{1 m} \vert C_{1 m}}$ & $s_1$ \\
\hline
$\lambda_1$ & $\substack{\lambda_2 + \mu_1 \\ x_{2 1} \vert C_{2 1}}$ & $\substack{\lambda_2 + \mu_2 \\ x_{2 2} \vert C_{2 2}}$ & $\cdots$ & $\substack{\lambda_2 + \mu_m \\ x_{2 m} \vert C_{2 m}}$ & $s_2$ \\
\hline
$\vdots$ & $\vdots$ & $\vdots$ & $\ddots$ & $\vdots$ & $\vdots$ \\
\hline
$\lambda_1$ & $\substack{\lambda_n + \mu_1 \\ x_{1 1} \vert C_{n 1}}$ & $\substack{\lambda_n + \mu_2 \\ x_{n 2} \vert C_{n 2}}$ & $\cdots$ & $\substack{\lambda_n + \mu_m \\ x_{n m} \vert C_{n m}}$ & $s_n$ \\
\hline
 & $d_1$ & $d_2$ & $\cdots$ & $d_m$ &
\end{tabular}
\end{center}
where numbers $d_1, \ldots, d_n$ and $s_1, \ldots, s_n$ follow directly from the problem and are not updated during the algorithm.
\begin{example}
  Consider the following transport problem:
  \begin{align*}
    S_1 &= 14,\ S_2 = 10,\ S_3 = 9 \\
    d_1 &= 12,\ d_2 = 5,\ d_3 = 8,\ d_4 = 8
  \end{align*}
  with cost matrix:
  \[
    C = \begin{pmatrix}
    5 & 3 & 4 & 6 \\
    2 & 7 & 4 & 1 \\
    5 & 6 & 2 & 4 \\
    \end{pmatrix}
  \]
  We need to find any basic feasible solution to start the algorithm.

  To find this basic feasible solution, we use the \textit{north-west corner rule}, this gives as much supply as possible to the first consumer starting with the first producer, and then as much as possible to the second, starting with the first producer and so on.
  Considering the values of $\vec{\lambda}$ and $\vec{\mu}$, we have 6 complementary slackness equations from $(C_{i j} - (\lambda_i + \mu_j))x_{i j} = 0$, corresponding to each of the non-zero $x_{i j}$.
  Recalling that we can set $\lambda_1 = 0$ without loss of generality, this along the other constraints gives:
  \[
    \vec{\lambda} = \begin{pmatrix}
    0 \\
    4 \\
    2 \\
    \end{pmatrix} \text{ and } \vec{\mu} = \begin{pmatrix}
    5 \\
    3 \\
    0 \\
    2 \\
    \end{pmatrix}
  \]
  So the intial tableau is:
  \begin{center}
  \begin{tabular}{c|c|c|c|c|c}
   & 5 & 3 & 0 & 2 &  \\
   \hline
  0 & $\substack{5 \\ 12 \vert 5}$ & $\substack{3 \\ 2 \vert 3}$ & $\substack{0 \\ 0 \vert 4}$ & $\substack{2 \\ 0 \vert 6}$ & 14 \\
  \hline
  4 & $\substack{9 \\ 0 \vert 2}$ & $\substack{7 \\ 3 \vert 7}$ & $\substack{4 \\ 7 \vert 4}$ & $\substack{6 \\ 0 \vert 1}$ & 10 \\
  \hline
  2 & $\substack{7 \\ 0 \vert 5}$ & $\substack{5 \\ 0 \vert 6}$ & $\substack{2 \\ 1 \vert 2}$ & $\substack{4 \\ 8 \vert 4}$ & 9 \\
  \hline
   & 12 & 5 & 8 & 8 & \\
  \end{tabular}
  \end{center}
  Now we are at a basic feasible solution we can start the main body of the algorithm.

  We then chose a cell with $C_{i j} < \lambda_i + \mu_j$, for example $i = 2$ and $j = 1$.
  Suppose that we sent a quantity $\theta$ along the edge between supplier $i$ and consumer $j$.
  Upon consulting the graph to find the new quantities along other edges, the smallest new quantity is $3 - \theta$.
  Sine we want to send as much down this edge as possible without violating the constraints, we pick $\theta = 3$.
  The tableaux is then updated accordingly:
  \begin{center}
  \begin{tabular}{c|c|c|c|c|c}
   & 5 & 3 & 0 & 2 &  \\
   \hline
  0 & $\substack{5 \\ 9 \vert 5}$ & $\substack{3 \\ 5 \vert 3}$ & $\substack{0 \\ 0 \vert 4}$ & $\substack{2 \\ 0 \vert 6}$ & 14 \\
  \hline
  -3 & $\substack{2 \\ 3 \vert 2}$ & $\substack{0 \\ 0 \vert 7}$ & $\substack{4 \\ 7 \vert 4}$ & $\substack{6 \\ 0 \vert 1}$ & 10 \\
  \hline
  -5 & $\substack{0 \\ 0 \vert 5}$ & $\substack{-2 \\ 0 \vert 6}$ & $\substack{2 \\ 1 \vert 2}$ & $\substack{4 \\ 8 \vert 4}$ & 9 \\
  \hline
   & 12 & 5 & 8 & 8 & \\
  \end{tabular}
  \end{center}
  We then choose another cell with $\lambda_i + \mu_j > C_{i j}$, and repeat as before.
  If there is no such cell left, the optimality conditions are satisfied and so we are done.
\end{example}
\section{Max-Flow Min-Cut}
\subsection{Problem Statement}
For a graph $G = (V, E)$ with $V = \{v_1, \ldots, v_n\}$, consider a transport problem where there is a single source vertex, $v_1$, and a single sink vertex $v_n$.
\begin{definition}[Max Flow Problem]
  The \textit{max-flow} problem is the following:
  \begin{maxi*}
    {x, \delta}{\delta}{}{}
    \addConstraint{\sum_{j: (1, j) \in E} x_{1 j}}{= \delta}
    \addConstraint{\sum_{i: (i, n) \in E} x_{i n}}{= \delta}
    \addConstraint{\sum_{j: (i, j) \in E} x_{i j}}{= \sum_{i: (i, j) \in E} x_{j i}\ \forall i \notin \{1, n\}}
    \addConstraint{0 \leq x_{i j}}{\leq C_{i j}}\ \forall (i, j) \in E
  \end{maxi*}
\end{definition}
If we define:
\[
  \phi_i(x) = \sum_{j: (i, j) \in E} x_{i j} - \sum_{j: (j, i) \in E} x_{j i}
\]
then the problem can be equivalently reformulated as:
\begin{maxi*}
  {x, \delta}{\delta}{}{}
  \addConstraint{\phi_1(x)}{= \delta}
  \addConstraint{\phi_n(x)}{= \delta}
  \addConstraint{\phi_i(x)}{= 0}\ \forall i \notin \{1, n\}
  \addConstraint{0 \leq x_{i j}}{\leq C_{i j}}\ \forall (i, j) \in E
\end{maxi*}
and further if we define:
\[
  b_i = \begin{cases}
  +1 & \text{ if } i = 1 \\
  -1 & \text{ if } i = n \\
  0 & \text{ otherwise }
  \end{cases}
\]
then the problem becomes:
\begin{maxi*}
  {x , \delta}{\delta}{}{}
  \addConstraint{\phi_i(x) - b_i \delta}{= 0\ \forall i \in \{1, \ldots, n\}}
  \addConstraint{0 \leq x_{i j}}{\leq C_{i j}\ \forall (i, j) \in E}
\end{maxi*}
\subsection{Cuts}
\begin{definition}[Cut]
  A \textit{cut} of a graph $G$ is a partition of $V$ into $S$ and $V \setminus S$.
\end{definition}
\begin{definition}[Cut Capacity]
  The \textit{capacity} of a cut is defined as:
  \[
    C(S) = \sum_{(i, j) \in E, i \in S, j \notin S} C_{i j}
  \]
\end{definition}
\begin{theorem}
  If $x$ be a feasible flow with value $\delta$, then for any cut $(S, V \setminus S)$ such that $v_1 \in S$ and $v_n \in V \setminus S$, we have: \label{cutIneq}
  \[
    \delta \leq C(S)
  \]
\end{theorem}
\begin{remark}
  From this we can conclude that the max flow is less than or equal to the min cut:
  \[
    \max_{x} \delta \leq \min_{\delta} C(S)
  \]
\end{remark}
\begin{proof}
  For any sets $A, B \subseteq V$ let:
  \[
    f_x(A, B) = \sum_{(i, j) \in E, i \in A, j \in B} x_{i j}
  \]
  Note that $A$ and $B$ need not be disjoint.
  We then sum up the flow constraints over all vertices in $S$:
  \begin{align*}
    \delta &= \sum_{i \in S} \left[\sum_{j: (i, j) \in E} x_{i j} - \sum_{j: (j, i) \in E} x_{i j}\right] \\
           &= f_x(S, V) - f_x(V, S) \\
           &= (f_x(S, S) + f_x(S, V \setminus S)) - (f_x(S, S) + f_x(V\setminus S, S)) \\
           &= f_x(S, V \setminus S) - f_x(V \setminus S, S) \\
           &\leq f_x(S, V \setminus S) \text{ as $f_x(V\setminus S, S) \geq 0$}\\
           &\leq C(S) \text{ as $x_{i j} \leq C_{i j}$}
  \end{align*}
\end{proof}
\begin{theorem}[Max Flow Min-Cut Theorem]
  Let $\delta^{*}$ be the optimal value of the flow from the source to the sink.
  We then have:
  \[
    \delta^{*} = \min\{C(S): C \subseteq V, v_1 \in S, v_n \in V \setminus S\}
  \]
  i.e. there exists a cut such that the capacity of the cut is the maximal flow.
\end{theorem}
\begin{proof}
  From \cref{cutIneq}, we know that $\delta \leq C(S)$ and so it suffices to find a set $S$ such that $C(S) = \delta^{*}$.

  An \textit{augmenting path}, is a path between two vertices in the undirected graph such that every \textit{forward edge} $(i, j)$ satisfies $x_{i j} < C_{i j}$ (i.e. the flow is not full), and every \textit{reverse edge} $(i, j)$ satisfies $x_{i j} > 0$.
  If $x$ is optimal, then $x$ cannot have any augmenting paths from 1 to $n$ as if an augmenting path exists, we can increase the flow along that path.

  Set:
  \[
    S = \{1\} \cup \{i: \text{there is an augmenting path from 1 to $i$}\}
  \]
  For the set $S$, sum up the flow constraints of $S$, from the proof of \cref{cutIneq}, we have:
  \[
    \delta^{*} = f_x(S, V \setminus S) - f_x(V \setminus S, S)
  \]
  We must have $f_x(S, V \setminus S) = C(S)$ and $f_x(V \setminus S, S) = 0$ and so $\delta^{*} = C(S)$.
  This is because any edge from $S$ to $V \setminus S$ must saturated as this cannot be an augmenting path, and any reverse edges must be empty.
\end{proof}
\subsection{Ford-Fulkerson Algorithm}
We can use the \textit{ford-fulkerson algorithm} to solve the max-flow min-cut problem.
\subsubsection{Algorithm}
\begin{enumerate}
  \item Start with any feasible flow $x$, for example, we can alway pick $x_{i j} = 0$.
  \item Search for an augmenting path and then direct as much flow as possible along that path.
  \item Repeat step \textbf{(ii)} until no more augmenting paths exist.
\end{enumerate}
We can then employ \cref{cutIneq} to find an optimal cut by selecting the verties with augmenting paths from $1$.
\begin{remark}
  For graphs with integer capacities, the Ford-Fulkerson algorithm will always converge in finite time.
\end{remark}
