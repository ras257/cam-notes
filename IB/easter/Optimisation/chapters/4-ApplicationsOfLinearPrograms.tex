\chapter{Applications of Linear Programs}
\section{Game Theory}
\subsection{Two Person Zero-Sum Games}
\begin{definition}[Two Person Zero-Sum Game]
  A \textit{two person zero-sum} game is an abstract game with two players in which $P_1$ can pick from any of $\{1, \ldots, m\}$ actions and $P_2$ can pick from any of $\{1, \ldots, n\}$ actions.
  We say the game is \textit{zero-sum} because when \textbf{$P_1$ attains score, $P_2$ loses an equal amount of score} and vice versa.
\end{definition}
\begin{remark}[Note]
  In some games players will play simultaneously and in others players will alternate turns.
\end{remark}
\begin{definition}[Payoff Matrix]
  The \textit{payoff matrix} $A$ for $P_1$ is an $m \times n$ matrix with entries $A_{i j}$, where $A_{i j}$ is the score attained by $P_1$ if $P_1$ plays action $i$ and $P_2$ plays action $j$.
\end{definition}
\begin{remark}[Remarks]
  \begin{itemize}
    \item Note that the payoff matrix $B$ for $P_2$ is also an $m \times n$ matrix however $B_{i j}$ now represents the score attained by $P_2$  if $P_1$ plays $i$ and $P_2$ plays $j$ (i.e. we only change who is getting the score and not which cell it is in).
    \item In a zero sum game, $B = -A$.
  \end{itemize}
\end{remark}
\begin{example} \label{RPSExample}
  Rock, paper, scissors in an example of a zero-sum game.
  Each player can pick either rock (call this 1), paper (call this 2), or scissors (call this 3).
  The payoff matrix for $P_1$ is:
  \[
    A = \begin{pmatrix}
    0 & -1 & 1 \\
    1 & 0 & -1 \\
    -1 & 1 & 0 \\
    \end{pmatrix}
  \]
  We will consider two versions of the game:
  \begin{itemize}
    \item Consider first a strange version of rock, paper, scissors where one player plays before the other.

      If $P_1$ starts and picks $i$, then $P_1$ knows $P_2$ will try and minimise $P_1$'s gain (or equivalently $P_2$'s loss) and so will choose $j$ to minimise $A_{i j}$.
      Thus if $P_1$ plays $i$, they expect to have a payoff of:
      \[
        \min_{j \in \{1, \ldots, n\}} A_{i j}
      \]
      Thus since $P_1$ wants to maximise their payoff over their possible choices, $P_1$ now tries to solve:
      \[
        \max_{i \in \{1, \ldots, m\}} \min_{j \in \{1, \ldots, n\}} A_{i j} = -1
      \]
      so $P_2$ wins, which makes sense as if you show your move first in rock, paper, scissors you will always lose if the other player is playing rationally.

      Similarly, if $P_2$ starts, they will try to solve:
      \[
        \min_{j \in \{1, \ldots, n\}} \max_{i \in \{1, \ldots, m\}} A_{i j} = 1
      \]
      so $P_1$ wins.
    \item Now consider the standard version of the game where both players pick simultaneously.
      In this case, the two players will get stuck in a loop indefinitely both trying to play perfectly whilst knowing the other player is also doing the same.
  \end{itemize}
\end{example}
We can formalise the idea of whether the players will get stuck in a loop using \textit{saddle points}.
\begin{lemma}
  In any payoff matrix: \label{payoffEquality}
  \[
    \alpha = \max_{i \in \{1, \ldots, m\}} \min_{j \in \{1, \ldots, n\}} A_{i j} \leq \min_{j \in \{1, \ldots, n\}} \max_{i \in \{1, \ldots, m\}} A_{i j} = \beta
  \]
\end{lemma}
\begin{proof}
  \nonexaminable
  For any $i, j$  we have:
  \[
    \min_{j \in \{1, \ldots, n\}} A_{i j} \leq A_{i j} \leq \max_{i \in \{1, \ldots, m\}} A_{i j} \implies \min_{j \in \{1, \ldots, n\}} A_{i j} \leq \max_{i \in \{1, \ldots, m\}} A_{i j}\ \forall i, j
  \]
  Since this is true for all $i$ we can take a maximum of the left hand side over $i$ whilst preserving the inequality:
  \[
    \alpha = \max_{i \in \{1, \ldots, m\}} \min_{j \in \{1, \ldots, n\}} A_{i j} \leq \max_{i \in \{1, \ldots, m\}} A_{i j}\ \forall j
  \]
  and since this is true for all $j$ we can take a minimum of the right hand side over $j$ whilst preserving the inequality:
  \[
    \alpha = \max_{i \in \{1, \ldots, m\}} \min_{j \in \{1, \ldots, n\}} A_{i j} \leq \min_{j \in \{1, \ldots, n\}}  \max_{i \in \{1, \ldots, m\}} A_{i j} = \beta
  \]
  as required.
\end{proof}
\subsection{Saddle Points}
\begin{definition}[Saddle Point]
  For a payoff matrix $A$, we say that $(k, l)$ is a \textit{saddle point} if in the payoff matrix $A_{k l}$ is \textbf{an entry that is both the minimum of its row and the maximum of its column}:
  \[
    A_{k l} = \min_{j \in \{1, \ldots, n\}} A_{k j} = \max_{i \in \{1, \ldots, m\}} A_{i l}
  \]
  That is, $k$ is the best response for $P_1$ if $P_2$ picks $l$ \textbf{and} $l$ is the best response for $P_2$ if $P_1$ picks $k$.

  We then call the value $A_{k l}$ the \textit{value} of the game.
\end{definition}
\begin{proposition}
  A pay off matrix $A$ has a saddle point if and only if:
  \[
    \alpha = \max_{i \in \{1, \ldots, m\}} \min_{j \in \{1, \ldots, n\}} A_{i j} = \min_{j \in \{1, \ldots, n\}} \max_{i \in \{1, \ldots, m\}} A_{i j} = \beta
  \]
  This shared value is also the value of the game.
\end{proposition}
\begin{proof}
  \nonexaminable
  \begin{proofdirection}{Assume $A$ has a saddle point}
    We have have:
    \[
      \alpha \geq \min_{j \in \{1, \ldots, n\}} A_{k j} = A_{k l} = \max_{i \in \{1, \ldots, m\}} A_{i l} \geq \beta \implies \alpha \geq \beta
    \]
    But from \cref{payoffEquality} we also have $\alpha \leq \beta$ and so $\alpha = \beta = A_{k l}$.
  \end{proofdirection}
  \begin{proofdirection}{Assume $\alpha = \beta$}
    By definition of $\alpha$, there exists $k \in \{1, \ldots, m\}$ such that:
    \[
      \alpha = \min_{j \in \{1, \ldots, n\}} A_{k j} \leq A_{k j}\ \forall j
    \]
    Similarly, by definition of $\beta$, there exists $l \in \{1, \ldots, n\}$ such that:
    \[
      \beta = \max_{i \in \{1, \ldots, m\}} A_{i l} \geq A_{i l}\ \forall i
    \]
    Combining, there exists $k, l$ such that:
    \[
      A_{i l} \leq \alpha = \beta \leq A_{k j}\ \forall i, j
    \]
    Taking $i = k$ and $j = l$ we get $A_{k l} = \alpha = \beta$ and so:
    \[
      A_{k l} = \min_{j \in \{1, \ldots, n\}} A_{k j} = \max_{i \in \{1, \ldots, m\}} A_{i l}
    \]
    i.e. $(k, l)$ is a saddle point.
  \end{proofdirection}
\end{proof}
\begin{remark}[Remarks]
  \begin{itemize}
    \item In the reverse direction above there could be multiple choices for the saddle point and so a game can have multiple saddle points, however they must all have the same value.
    \item There can be entries of the payoff matrix which have the value of the game but are not saddle points.
  \end{itemize}
\end{remark}
\begin{example}
  Consider the payoff matrix:
  \[
    A = \begin{pmatrix}
    1 & 2 \\
    3 & 4 \\
    \end{pmatrix}
  \]
  For this payoff matrix:
  \begin{itemize}
    \item $P_1$ decides they should play $i = 2$ and expects $P_2$ to play $j = 1$.
    \item $P_2$ decides they should play $j = 1$ and expects $P_1$ to play $i = 2$.
  \end{itemize}
  Thus $(2, 1)$ is a saddle point of $A$ and the value of the game is $A_{2 1} = 3$.

  Since there is a saddle point, with these player strategies:
  \begin{itemize}
    \item If the game is played with alternating turns, it \textbf{does not} matter who starts.
    \item The game can be played simultaneously without the players getting stuck in a loop trying to find the optimal strategy.
  \end{itemize}
\end{example}
\begin{example}
  Consider the payoff matrix:
  \[
    A' = \begin{pmatrix}
    4 & 2 \\
    1 & 3 \\
    \end{pmatrix}
  \]
  For this payoff matrix:
  \begin{itemize}
    \item $P_1$ decides they should play $i = 1$ and expects $P_2$ to play $j = 2$.
    \item $P_2$ decides they should play $j = 2$ and expects $P_1$ to play $i = 2$.
  \end{itemize}
  So $A'$ does not have a saddle point, this means that with these strategies:
  \begin{itemize}
    \item If the game is played with alternating turns, it \textbf{does matter} who starts.
    \item The game \textbf{cannot} be played simultaneously without the players getting stuck in a loop trying to find the optimal strategy.

      For example, from $P_1$'s perspective, they might decide from the above that they should play $i = 1$, but then will realise that $P_2$ would pick $j = 2$.
      But now given that $P_2$ will play $j = 2$, it would now be better for $P_1$ to choose $i = 2$.
      However, $P_2$ would also realise this and so would pick $j = 1$, \textit{and so on...}
  \end{itemize}
\end{example}
\begin{remark}
  The payoff matrix for the rock, paper, scissors game in \cref{RPSExample} does not have a saddle point since:
  \[
    \max_{i \in \{1, \ldots, m\}} \min_{j \in \{1, \ldots, n\}} A_{i j} \neq \min_{j \in \{1, \ldots, n\}} \max_{i \in \{1, \ldots, m\}} A_{i j}
  \]
  which is why the players get stuck when trying to play simultaneously.
\end{remark}
\subsection{Strategies}
\begin{definition}[Strategy]
  A \textit{strategy} is a probability distribution for the actions that a player picks.
\end{definition}
\begin{definition}[Pure and Mixed Strategies]
  A \textit{pure strategy} is a strategy where one single action has probability 1, i.e. the player will always pick a particular action.
  If a strategy is not pure, then we call it a \textit{mixed strategy}.
\end{definition}
\begin{remark}
  In the previous section, the players were using a pure strategy.
  With mixed strategy, the players actions have probabilities and so games with a saddle point can played simultaneously whilst still allowing for optimal play.
\end{remark}
We represent the mixed strategy for $P_1$ as a vector $\vec{p} \in \R^{m}$ where $p_i \in [0, 1]$ is the discrete probability distribution that $P_1$ picks action $i$ and similarly represent the mixed strategy for $P_2$ with the vector $\vec{q} \in \R^{n}$.

Given that $P_2$ picks $j$, $P_1$'s expected score is:
\[
  \sum_{i = 1}^{m} p_i A_{i j}
\]
From $P_1$'s perspective, they expect $P_2$ to pick to minimise this expected score over $j$ and so for optimal play, $P_1$ wants to adopt the mixed strategy $\vec{p}$ that solves:
\begin{equation}
  \max_{\vec{p} \in P_m} \left(\min_{j \in \{1, \ldots, n\}} \sum_{i = 1}^{m} p_i A_{i j}\right) \label{p1prob}
\end{equation}
where $P_m$ is the set of all discrete probability distributions on $\{1, \ldots, m\}$.

Similarly, from $P_2$'s perspective, they want to adopt the mixed strategy $\vec{q}$ that solves:
\[
  \min_{\vec{q} \in P_n} \left(\max_{i \in \{1, \ldots, m\}} \sum_{j = 1}^{n} q_j A_{i j}\right)
\]
Note that min and max are swapped because $A_{i j}$ is the gain for $P_1$ and loss for $P_2$.
\subsection{Linear Programming for Strategies}
We can express the optimisation problems for each player as linear programs.

Considering $P_1$'s problem from \cref{p1Prob}, we first note that we can express the expectation as:
\[
  \sum_{i = 1}^{m} p_i A_{i j} = [A^{\trans} \vec{p}]_j
\]
We want to maximise over $\vec{p}$ the minimum over $j$ of this.
The minimum over $j$ of this will be the smallest component of $A^{\trans} \vec{p} \in \R^{n}$.
The solution is therefore the largest $v \in \R$ over all $\vec{p} \in \R^{m}$ such that \textbf{every component} of $A^{\trans} \vec{p}$ is bigger than $v$.
This means that the limiting component is the smallest component of $A^{\trans} \vec{p}$, i.e. the inner minimum.
Therefore $P_1$'s problem is equivalent to the linear program:
\begin{maxi}
  {v \in \R, \vec{p} \in \R^{m}}{v}{}{\label{p1Lin}}
  \addConstraint{A^{\trans} \vec{p}}{\geq v \vec{1}_n}
  \addConstraint{\vec{1}^{\trans}_m \vec{p}}{= 1}
  \addConstraint{\vec{p}}{\geq \vec{0}}
\end{maxi}
where $\vec{1}_k = (1, \ldots, 1)^{\trans} \in \R^{k}$.
The optimal solution $\vec{p}^{*}$ is the best mixed strategy for $P_1$ and the optimal value is its expected score.
The other two constraints ensure that $p_i$ is a valid probability distribution since $\vec{p}$ is a valid probability distribution if and only if $\vec{1}^{\trans}_m \vec{p} = 1$ and $\vec{p} \geq 0$.

Similarly for $P_2$, we can express their problem as:
\begin{mini}
  {w \in \R, \vec{q} \in \R^{n}}{w}{}{\label{p2Lin}}
  \addConstraint{A\vec{q}}{\leq w \vec{1}_m}
  \addConstraint{\vec{1}^{\trans}_n \vec{q}}{= 1}
  \addConstraint{\vec{q}}{\geq \vec{0}}
\end{mini}
\begin{theorem}
  The optimisation problems for $P_1$, in \cref{p1Lin}, and $P_2$, in \cref{p2Lin}, are dual. \label{gameDual}
\end{theorem}
\begin{proof}
  To find the dual of $P_2$'s problem, we first add slack variables:
  \begin{mini*}
    {w \in \R, \vec{q} \in \R^{n}, \vec{s} \in \R^{m}}{w}{}{}
    \addConstraint{A\vec{q} + \vec{s}}{= w \vec{1}_m}
    \addConstraint{\vec{1}^{\trans}_n \vec{q}}{= 1}
    \addConstraint{\vec{q}}{\geq \vec{0}}
    \addConstraint{\vec{s}}{\geq \vec{0}}
  \end{mini*}
  We then package all of our variables up into $(w, \vec{q}, \vec{s})$ and introduce parameters $\vec{\lambda}_1 \in \R^{m}$ and $\lambda_2 \in \R$ to construct the Lagrangian:
  \begin{align*}
    L(w, \vec{q}, \vec{s}, \vec{\lambda}_1, \lambda_2) &= w - \vec{\lambda}^{\trans}_1(A\vec{q} + \vec{s} - w \vec{1}_m) - \lambda_2(\vec{1}^{\trans}_n \vec{q} - 1) \\
                                                       &= (1 + \vec{\lambda}^{\trans}_1 \vec{1}_m) w - (\vec{\lambda}^{\trans}_1 A + \lambda_2 \vec{1}^{\trans}_n) \vec{q} - \vec{\lambda}^{\trans}_1 \vec{s} + \lambda_2
  \end{align*}
  So the dual objective function is:
  \[
    g(\vec{\lambda}_1, \lambda_2) = \min_{w \in \R, \vec{q} \geq \vec{0}, \vec{s} \geq \vec{0}} [(1 + \vec{\lambda}^{\trans}_1 \vec{1}_m) w - (\vec{\lambda}^{\trans}_1 A + \lambda_2 \vec{1}^{\trans}_n) \vec{q} - \vec{\lambda}^{\trans}_1 \vec{s} + \lambda_2]
  \]
  The set $\Lambda$ for which the dual objective function is then:
  \[
    \Lambda = \{(\vec{\lambda}_1, \lambda_2): -\vec{\lambda}^{\trans}_1 \vec{1}_m = 1,\ -\vec{\lambda}^{\trans}_1 A \geq \lambda_2\vec{1}^{\trans}_n, -\vec{\lambda_1} \geq \vec{0}\}
  \]
  and then when $\vec{\lambda} \in \Lambda$, $g(\vec{\lambda}_1, \lambda_2) = \lambda_2$.

  Therefore, the dual of the problem is:
  \begin{maxi*}
    {\lambda_2 \in \R, \vec{\lambda}_1 \in \R^{m}}{\lambda_2}{}{}
    \addConstraint{- \vec{\lambda}^{\trans}_1 A}{\geq \lambda_2 \vec{1}^{\trans}_n}
    \addConstraint{-\vec{\lambda}^{\trans}_1 \vec{1}_m}{= 1}
    \addConstraint{-\vec{\lambda}_1}{\geq \vec{0}}
  \end{maxi*}
  We then relabel $\lambda_2 \to v$, and let $\vec{p} = -\vec{\lambda}_1$.
  Finally, we take transposes of the first two constraints to arrive at:
  \begin{maxi*}
    {v \in \R, \vec{p} \in \R^{m}}{v}{}{}
    \addConstraint{A^{\trans} \vec{p}}{\geq v \vec{1}_n}
    \addConstraint{\vec{1}^{\trans}_m \vec{p}}{= 1}
    \addConstraint{\vec{p}}{\geq \vec{0}}
  \end{maxi*}
  which is as in \cref{p1Lin}.
\end{proof}
\begin{corollary}[Optimal Strategies]
  A strategy $\vec{p}$ is optimal for $P_1$ if and only if there exists $\vec{q}$ and $v$ such that: \label{optimalStrategies}
  \begin{enumerate}
    \item $A^{\trans} \vec{p} \geq v \vec{1}_n$, $\vec{1}^{\trans}_m \vec{p} = 1$, $\vec{p} \geq \vec{0}$ (Primal feasibility)
    \item $A\vec{q} \leq v \vec{1}_m$, $\vec{1}^{\trans}_n \vec{q} = 1$, $\vec{q} \geq \vec{0}$ (Dual feasibility) -- Note that $v$ is repeated here.
    \item $v = \vec{p}^{\trans} A \vec{q}$ (Complementary slackness)
  \end{enumerate}
\end{corollary}
\begin{proof}
  Using \cref{optimalityConds}, we need only check the complementary slackness equations.
  We know from \cref{optimalityConds} that $(\vec{p}, v)$ and $(\vec{q}, w)$ are primal/dual optimal if and only if both:
  \[
    (A \vec{q} - w \vec{1}_m)^{\trans} \vec{p} = 0 \text{ and } \vec{q}^{\trans}(A^{\trans} \vec{p} - v\vec{1}_n) = 0
  \]
  We see that since $\vec{1}^{\trans}_m \vec{p} = 1$:
  \[
    (A \vec{q} - w \vec{1}_m)^{\trans} \vec{p} = 0 \iff \vec{q}^{\trans} A^{\trans} \vec{p} = w \vec{1}^{\trans}_m \vec{p} = w
  \]
  and since $\vec{q}^{\trans} \vec{1}_n = 1$:
  \[
    \vec{q}^{\trans}(A^{\trans} \vec{p} - v\vec{1}_n) = 0 \iff \vec{q}^{\trans} A^{\trans} \vec{p} = v \vec{q}^{\trans} \vec{1}_n = v
  \]
  Thus since $\vec{q}^{\trans} A^{\trans} \vec{p} = \vec{p}^{\trans} A \vec{q}$:
  \[
    (A \vec{q} - w \vec{1}_m)^{\trans} \vec{p} = 0 \text{ and } \vec{q}^{\trans}(A^{\trans} \vec{p} - v\vec{1}_n) = 0 \iff v = w = \vec{p}^{\trans} A \vec{q}
  \]
\end{proof}
\begin{remark}
  Since $v = w$, the expected payoff for both players is always the same.
\end{remark}
\subsection{Finding Optimal Strategies}
We can use the following steps to find optimal strategies for each player.
\begin{enumerate}
  \item Look for a saddle point of the payoff matrix, if such a point exists then a pure strategy where both players always pick this point is optimal.
  \item Simplify the payoff matrix by looking for actions which a player would obviously not play and so we can remove their corresponding rows/columns.

    For example, suppose for $P_1$ that $\exists i_1, i_2$ such that $A_{i_1 j} \geq A_{i_2 j}$ for all $j$.
    There would never be a reason for $P_1$ to pick $i_2$ as its payoff is worse regardless of $P_2$'s choice and so we can simply drop row $i_2$.
  \item Formulate and solve the linear program for $P_1$, either graphically or by using the simplex method.
    This yields $P_1$'s strategy and the expected payoff for both players.
  \item Use complementary slackness and \cref{optimalStrategies} to determine $P_2$'s strategy.
\end{enumerate}
\begin{example}
  Consider the payoff matrix for $P_1$:
  \[
    A = \begin{pmatrix}
    2 & 3 & 4 \\
    3 & 1 & 1/2 \\
    1 & 3 & 2 \\
    \end{pmatrix}
  \]
  We first notice that every entry in the third row is less than or equal to the first row and so we can drop the third row:
  \[
    A = \begin{pmatrix}
    2 & 3 & 4 \\
    3 & 1 & 1/2 \\
    \end{pmatrix}
  \]
  There is no saddle point since:
  \[
    \max_{i \in \{1, \ldots, m\}} \min_{j \in \{1, \ldots, n\}} A_{i j} = 2 \neq 3 = \min_{j \in \{1, \ldots, n\}} \max_{i \in \{1, \ldots, m\}} A_{i j}
  \]
  A mixed strategy is therefore optimal and so $P_1$ adopts the mixed strategy $(p, 1 - p, 0)$ and thus the linear program is:
  \begin{maxi*}
    {p \in \R, v \in \R}{v}{}{}
    \addConstraint{2p + 3(1 - p)}{\geq v}
    \addConstraint{3p + (1 - p)}{\geq v}
    \addConstraint{4p + \frac{1}{2}(1 - p)}{\geq v}
    \addConstraint{0 \leq p}{\leq 1}
  \end{maxi*}
  In this case we do not need to use the simplex method and we can just solve this graphically in $\R^2$.
  To do this we plot the feasible set in the $(p, v)$-plane and find the maximal point $v$ within this region.

  After solving, we see that $p = \frac{2}{3}$ and so $P_1$ adopts the mixed strategy $\left(\frac{2}{3}, \frac{1}{3}, 0\right)$ with expected payoff $v = \frac{7}{3}$.
  From \cref{gameDual}, we know that the problem for $P_2$'s strategy is dual to this and so we can use complementary slackness to find their strategy.

  We first note that $2p + 3(1 - p) = \frac{7}{3}$ and $3p + (1 - p) = \frac{7}{3}$ but $4p + \frac{1}{2}(1 - p) \geq \frac{7}{3}$, thus the first two constraints are tight but there is still slack in the third constraint.
  Thus we must have $q_3 = 0$ in the dual and so we can write $P_2$'s mixed strategy as $(q, 1 - q, 0)$ and use $\vec{p}^{\trans} A \vec{q} = \frac{7}{3}$ to find $q$, yielding $q = \frac{2}{3}$.
  So $P_2$ adopts the same mixed strategy as $P_1$.
\end{example}
\section{Network Flows}
\subsection{Graphs}
Network flow problems are optimisation problems defined on \textit{graphs}.
\begin{definition}[Directed Graph]
  A \textit{directed graph} $G = (V, E)$ consists of a set $V$ of \textit{vertices} and a set $E \subseteq V \times V$ of ordered pairs of vertices called \textit{edges}, for example $(v_1, v_2)$ represents a directed edge from $v_1 \to v_2$ for $v_1, v_2 \in V$.
\end{definition}
From now on, assume we are working with a graph $G = (V, E)$ with $n$ vertices, i.e. $V = \{v_1, \ldots, v_n\}$.
\textit{Network flow} problems involve a quantity or resource that is able to move between vertices along edges:
\begin{definition}[Flow]
  If there is an edge $(i, j) \in E$, then \textit{flow} of some quantity along this edge from $v_i \to v_j$ is represented by $x_{i j}$.
\end{definition}
Quantity can also flow into and out of the graph:
\begin{definition}[Flow Vector, Source \& Sink Vertices]
  The flow into a vertex $v_i$ from outside the graph is given by $b_i$ and we define $\vec{b} \in \R^{n}$ with these components to be the \textit{flow vector}.

  If $b_i > 0$, then we call $v_i$ called a \textit{source} and if $b_i < 0$, then we call $v_i$ a \textit{sink}.
\end{definition}
We also associate a cost with moving a quantity between two vertices:
\begin{definition}[Cost Matrix]
  The \textit{cost matrix} $C$ is an $n \times n$ matrix where $C_{i j}$ is the cost per unit flow along the edge $v_i \to v_j$.
  That is, the cost for the flow along the edge $v_i \to v_j$ is given by $C_{i j} x_{i j}$.
\end{definition}
We can also restrict the minimum and maximum flow between two vertices:
\begin{definition}[Capacity Constraints]
  The $n \times n$ matrices $\overline{M}$ and $\underline{M}$ restrict the \textit{capacity} through each edge, i.e. the flow along the edge $v_i \to v_j$ must satisfy $\underline{M}_{i j} \leq x_{i j} \leq \overline{M}_{i j}$.
\end{definition}
\subsection{Minimum Cost Flow}
\begin{definition}[Minimum Cost Flow]
  The \textit{minimum cost flow} problem is a linear program given by:
  \begin{mini}
    {x_{i j} \in \R\ \forall i, j}{\sum_{(i, j) \in E} C_{i j}x_{i j}}{}{\label{minCost}}
    \addConstraint{b_i + \sum_{j: (j, i) \in E} x_{j i}}{= \sum_{j: (i, j) \in E} x_{i j}}\ \forall i \in \{1, \ldots, n\}
    \addConstraint{\underline{M}_{i j} \leq x_{i j}}{\leq \overline{M}_{i j}\ \forall (i, j) \in E}
  \end{mini}
  The quantity we are trying to minimise is the total cost over the whole graph, the first constraint ensures that the total flow into a vertex is equal to the total flow out (i.e. flow is conserved, a quantity cannot just sit at a vertex), and the second represents the capacity constraints.
\end{definition}
If a graph is feasible for the above problem, then using flow conservation:
\begin{align*}
  \sum_{i = 1}^{n} \left[b_i + \sum_{j: (j, i) \in E} x_{j i}\right] &= \sum_{i = 1}^{n} \sum_{j: (i, j) \in E} x_{i j} \\
  \implies \sum_{i = 1}^{n} b_i + \sum_{(i, j) \in E} x_{i j} &= \sum_{(i, j) \in E} x_{i j} \\
  \implies \sum_{i = 1}^{n} b_i &= 0
\end{align*}
This means that the net flow of quantity into and out of the graph as a whole is 0.
We usually assume that this holds otherwise the graph cannot be feasible for the problem.
\subsection{Transport Problem}
The transport problem is a special case of the minimum cost flow problem.

Suppose that we have $n$ suppliers $i \in \{1, \ldots, n\}$ and $m$ consumers $j \in \{1, \ldots, m\}$.
Each supplier $i$ produces a quantity $s_i$ and each consumer $j$ demands a quantity $d_j$.
We assume that the total supply perfectly matches the total demand, that is:
\[
  \sum_{i = 1}^{n} s_i = \sum_{j = 1}^{m} d_j
\]
The flow from supplier $i$ to consumer $j$ is denoted $x_{i j}$ and its associated cost per unit flow is $C_{i j}$.
We also insist that $x_{i j} \geq 0$ to ensure that quantity can only flow from suppliers to consumers.

In this problem $G$ is a \textit{bipartite graph} where every supplier is connected by a directed edge to every consumer for a total of $n + m$ vertices and $nm$ directed edges:

We can now formulate the optimisation problem:
\begin{definition}[Transport Problem]
  The \textit{transport problem} is a linear program given by:
  \begin{mini*}
    {x_{i j} \in \R\ \forall i, j}{\sum_{\substack{1 \leq i \leq n \\ 1 \leq j \leq m}} C_{i j}x_{i j}}{}{}
    \addConstraint{\sum_{j = 1}^{m} x_{i j}}{= s_i\ \forall i \in \{1, \ldots, n\}}
    \addConstraint{\sum_{i = 1}^{n} x_{i j}}{= d_j\ \forall j \in \{1, \ldots, m\}}
    \addConstraint{x_{i j}}{\geq 0\ \forall i \in \{1, \ldots, n\} \text{ and } j \in \{1, \ldots, m\}}
  \end{mini*}
  Again, we are trying to minimise the total cost over the whole graph, the first set of constraints ensures that all of the supply produced by each supplier is used, and the second set of constraints ensures that the demand of every supplier is exactly satisfied.
\end{definition}
\begin{remark}[Remarks]
  \begin{itemize}
    \item We can reformulate the first two sets of constraints using matrix multiplication and the objective function as a dot product and so this problem is already in the standard form for linear programs from \cref{LPStandard}.
    \item Since we assume that $\sum_{i = 1}^{n} s_i = \sum_{i = 1}^{n} d_j$, we have some redundant constraints within the first two sets of constraints and so cannot directly apply the simplex algorithm.
      We could remove the redundant constraints, however we can instead use the \textit{Transport Algorithm}, which we will introduce later.
    \item Here we use a formulation where every supplier is connected to every consumer, however we can equally have a transport problem where some of these routes are prohibited by introducing an edge set.
  \end{itemize}
\end{remark}
We can clearly recast any transport problem as a minimum cost flow problem as this is just a special type of graph with a minimum flow of 0 on each vertex.
Interestingly, it turns out that we can recast any minimum cost flow problem \textbf{with finite capacities} as a transport problem.
This means that we need only study the transport problem in order to solve any more general minimum cost flow problem.
\begin{theorem}
  Every minimum cost flow problem \textbf{with finite capacities} (i.e. $\overline{M}_{i j} < \infty\ \forall i, j$) can be recast as an equivalent transport problem.
\end{theorem}
\begin{remark}
  If $C_{i j} \geq 0\ \forall i, j$ (i.e. all costs are non-negative), then can equivalently reformulate the problem with $\overline{M}_{i j} < \infty$ by setting:
  \[
    \overline{M}_{i j} = \sum_{i \in \{1, \ldots, n\}} |b_i| < \infty
  \]
  for those $(i, j)$ where $\overline{M}_{i j} = \infty$ as the flow on an edge clearly cannot be greater than this.

  Note that this may not work if we have any negative costs since there could be a loop of edges with negative cost, in which case we would want to send an infinite quantity around these edges to send the objective function to $-\infty$.
\end{remark}
\begin{proof}\par
  \textbf{Assuming that $\underline{M} = O$}\par
  Starting from the problem in \cref{minCost}, we first observe that we can assume that $\underline{M} = O$ (the $n \times n$ zero-matrix).
  This is because if we define $\widetilde{x}_{i j} = x_{i j} - \underline{M}_{i j}$, the regional constraints become:
  \[
    0 \leq \widetilde{x}_{i j} \leq \overline{M}_{i j} - \underline{M}_{i j}
  \]
  The objective function becomes:
  \[
    \sum_{(i, j) \in E} C_{i j}x_{i j} = \sum_{(i, j) \in E} C_{i j} \widetilde{x}_{i j} + \sum_{(i, j) \in E} C_{i j} \underline{M}_{i j}
  \]
  The second term on the right hand side is just a constant of the problem and so minimising $\sum C_{i j} x_{i j}$ is equivalent to minimising $\sum C_{i j} \widetilde{x}_{i j}$.
  The flow constraint becomes:
  \[
    b_i + \sum_{j: (j, i) \in E} \widetilde{x}_{j i} + \sum_{j: (j, i) \in E} \underline{M}_{i j} = \sum_{j: (i, j) \in E} \widetilde{x}_{i j} + \sum_{j: (i, j) \in E} \underline{M}_{i j}
  \]
  We then define:
  \[
    \widetilde{b}_i = b_i + \sum_{j: (j, i) \in E} \underline{M}_{j i} - \sum_{j: (i, j) \in E} \underline{M}_{i j}
  \]
  Thus the problem can be equivalently reformulated up to a constant difference in the optimal value and so assume $\underline{M} = O$.

  \textbf{Suppliers and Consumers}\par
  For every vertex $v_i \in V$, construct a consumer $i$ with demand:
  \[
    \sum_{k: (i, k) \in E} \overline{M}_{i k} - b_i
  \]
  and for every edge $(i, j) \in E$, construct a supplier $ij$ with supply $\overline{M}_{i j} < \infty$.
  Finally, let the cost of transport from supplier $ij$ to consumer $i$ be 0 and the cost from $ij$ to consumer $j$ be $C_{i j}$.
  Pictorially, this is:
  \begin{center}
  \begin{tikzpicture}[>=stealth]
    \node[shape=circle, draw=black] (S1) at (0, -1) {$ij$};
    \path[dashed,->] (-2, -1) edge node[above] {$\overline{M}_{i j}$} (S1);
    \node[yshift=18] at (0, 0) {Suppliers};

    \node[shape=circle,draw=black] (D1) at (3, 0) {$i$};
    \node[shape=circle,draw=black] (D2) at (3, -2) {$j$};
    \path[dashed, <-] (5, 0) edge (D1) node[right] {$\sum\limits_{k: (i, k) \in E} \overline{M}_{i k} - b_i$};
    \path[dashed, <-] (5, -2) edge (D2) node[right] {$\sum\limits_{k: (j, k) \in E} \overline{M}_{j k} - b_j$};
    \node[yshift=20] at (D1) {Consumers};

    \path[->] (S1) edge node[above left] {$0$} (D1);
    \path[->] (S1) edge node[below left] {$C_{i j}$} (D2);
    \path[->, gray!70] (1.5, 0.75) edge node[below, yshift=5] {$\vdots$} (D1);
    \path[->, gray!70] (1.5, -2.75) edge node[above] {$\vdots$} (D2);
  \end{tikzpicture}
  \end{center}

  \textbf{Equivalence of Feasibility}\par
  Now suppose that we have a feasible flow for the new transport problem.
  Each supplier $ij$ only sends to consumers $i$ and $j$ with total supply $\overline{M}_{i j}$, and so we must a flow $x_{i j}$ to consumer $j$ and the remaining $\overline{M}_{i j} - x_{i j}$ to consumer $i$.
  The total flow coming into consumer $i$ is then:
  \[
    \sum_{k: (i, k) \in E} (\overline{M}_{i k} - x_{i k}) + \sum_{k: (k, i) \in E} x_{k i}
  \]
  Since the flow is feasible for the new problem, this must be equal to the demand, and so:
  \begin{align*}
    \sum_{k: (i, k) \in E} \overline{M}_{i k} - b_i &= \sum_{k: (i, k) \in E} (\overline{M}_{i k} - x_{i k}) + \sum_{k: (k, i) \in E} x_{k i} \\
    \iff b_i + \sum_{k: (k, i) \in E} x_{k i} &= \sum_{k: (i, k) \in E} x_{i k}
  \end{align*}
  which is precisely feasibility for original minimum flow problem along with the fact that transport flows are non-negative.

  \textbf{Equivalence of Cost Function}\par
  The cost function for our new transport problem is:
  \[
    \sum_{(i, j) \in E} \left[0 \cdot (\overline{M}_{i j} - x_{i j}) + C_{i j} \cdot x_{i j}\right] = \sum_{(i, j) \in E} C_{i j} x_{i j}
  \]
  which is identical to that of the minimum cost flow problem.
  Thus both problems have the same feasibility conditions and cost functions, and so are equivalent.
\end{proof}
\subsection{Transport Algorithm}
\begin{theorem}[Optimality Conditions]
  If for some feasible flow $(x_{i j})$ we have dual variables $\vec{\lambda} \in \R^{n}$ and $\vec{\mu} \in \R^{m}$  such that for all $i \in \{1, \ldots, n\}$ and $j \in \{1, \ldots, m\}$ we have: \label{transportOpti}
  \begin{align*}
    C_{i j} &\geq \lambda_i + \mu_j \\
    (C_{i j} - (\lambda_i + \mu_j)) x_{i j} &= 0
  \end{align*}
  then this flow is optimal.
\end{theorem}
\begin{remark}
  The components $\lambda_i$ correspond to supplier $i$ and the components $\mu_j$ correspond to consumer $j$.
\end{remark}
\begin{proof}
  The Lagrangian is given by:
  \begin{align*}
    L(x, \vec{\lambda}, \vec{\mu}) &= \sum_{i = 1}^{n} \sum_{j = 1}^{m} C_{i j}x_{i j} - \sum_{i = 1}^{n} \lambda_i \left(\sum_{j = 1}^{m} x_{i j} - s_i\right) - \sum_{j  =1}^{m} \mu_j \left(\sum_{i = 1}^{n} x_{ij} - d_j\right) \\
                                   &= \sum_{i = 1}^{n} \sum_{j = 1}^{m} (C_{i j} - \lambda_i - \mu_j)x_{i j} + \sum_{i = 1}^{n} \lambda_i s_i + \sum_{j = 1}^{m} \mu_j d_j
  \end{align*}
  Since $x_{i j} \geq 0$, the feasibility set for the dual is:
  \[
    \Lambda = \{\vec{\lambda} \in \R^{n},\ \vec{\mu} \in \R^{m}: C_{i j} \geq \lambda_i + \mu_j \}
  \]
  Therefore, the first assumption of the theorem implies that the given $(\vec{\lambda}, \vec{\mu})$ is dual feasible and the second theorem assumption tells us that complementary slackness holds.
  Thus using \cref{optimalityConds}, we conclude that the flow $x$ is optimal.
\end{proof}
\begin{remark} \label{wlogRemark}
  If we are given a basic feasible solution flow $x$, then we can use the above to try and solve for $(\vec{\lambda}, \vec{\mu})$ using complimentary slackness.
  If we are able to find such a $(\vec{\lambda}, \vec{\mu})$ then we know immediately that $x$ is optimal.

  We can make this easier to solve by noting that if we increase all of the $\lambda_i$ by a constant $k$ and decrease all of the $\mu_j$ by $k$ then $\lambda_i + \mu_j$ remains unchanged.
  Thus by taking $k = -\lambda_1$ we can without loss of generality assume that $\lambda_1 = 0$ when solving.
\end{remark}
\subsection{Transport Algorithm}
Instead of using the simplex algorithm to solve transportation problems, we can instead use the \textit{transport algorithm}.
Similarly to using the simplex algorithm with a tableau, we employ a \textit{transport tableau}, which contains all the data for the problem and algorithm state whilst allowing us to solve the problem.

Here is the structure of the transport tableau:
\begin{center}
  \newcommand{\transportcell}[2]{$\begin{array}{c|c} \multicolumn{2}{c}{\lambda_{#1} + \mu_{#2}} \\ \cline{2-2} x_{{#1} {#2}} & C_{{#1} {#2}} \end{array}$}
\begin{tabular}{c|c|c|c|c|c}
  & $\mu_1$ & $\mu_2$ & $\cdots$ & $\mu_m$ &  \\
  \hline
  $\lambda_1$ & \transportcell{1}{1} & \transportcell{1}{2} & $\cdots$ & \transportcell{1}{m} & $s_1$ \\
  \hline
  $\lambda_2$ & \transportcell{2}{1} & \transportcell{2}{2} & $\cdots$ & \transportcell{2}{m} & $s_2$ \\
  \hline
  $\vdots$ & $\vdots$ & $\vdots$ & $\ddots$ & $\vdots$ & $\vdots$ \\
  \hline
  $\lambda_n$ & \transportcell{n}{1} & \transportcell{n}{2} & $\cdots$ & \transportcell{n}{m} & $s_n$ \\
  \hline
  & $d_1$ & $d_2$ & $\cdots$ & $d_m$ &
\end{tabular}
\end{center}
Note that because the values $d_i$, $s_i$, and $C_{i j}$ are constants of the problem, they are \textbf{not updated} during the algorithm.

\subsubsection{North-West Corner Rule}
As with the simplex method, we need to start with a basic feasible solution of flows.
We can find such an $x$ using the \textit{north-west corner rule}.
\begin{enumerate}
  \item Start with some flow $x_{i j}$ and increase it until either the supply $s_i$ is used or demand $d_j$ is met.
    Typically, we start with $x_{1 1}$.
  \item If the supply is used, move to the next supplier (i.e. $x_{(i + 1) j}$) and repeat \textbf{(i)}
  \item If the demand is met, move to the next consumer (i.e. $x_{i (j + 1)}$) and repeat \textbf{(i)}
\end{enumerate}
If supply and demand are met simultaneously then the north-west corner method has succeeded and the problem is non-degenerate.

We will now illustrate how to perform the transport algorithm with an example:
\subsubsection{Transport Algorithm Example}
Consider a transport problem with three suppliers and four consumers with supplies and demands:
\begin{align*}
  s_1 &= 14,\ s_2 = 10,\ s_3 = 9 \\
  d_1 &= 12,\ d_2 = 5,\ d_3 = 8,\ d_4 = 8
\end{align*}
and cost matrix:
\[
  C = \begin{pmatrix}
  5 & 3 & 4 & 6 \\
  2 & 7 & 4 & 1 \\
  5 & 6 & 2 & 4 \\
  \end{pmatrix}
\]
We now use the north-west corner rule to find a basic feasible solution.
\begin{enumerate}
  \item Starting with $x_{1 1}$, we set $x_{1 1} = 12$ so that $d_1$ is satisfied and then move on to $x_{1 2}$ which we set $x_{1 2} = 2$ so that $s_1$.

    Since supplier 2 is used, we move onto supplier 3.

\end{enumerate}

\begin{definition}[Spanning Tree]
  A \textit{spanning tree} of an \textbf{undirected graph} is a subset of $G$ without any cycles such that every vertex can be reached from every other vertex.
\end{definition}
\begin{theorem}
  When constructing a basic feasible flow using the north-west corner rule, if we construct a \textbf{undirected graph} using the set of edges with positive flow, then we obtain a \textit{spanning tree} $T$ with exactly $m + n - 1$ edges.
\end{theorem}
\begin{proof}
  Not lectured.
\end{proof}
This means that our system of equations will always have $m + n - 1$ unknowns (as $\vec{\lambda} \in \R^{n}, \vec{\mu} \in \R^{m}$ and we assume $\lambda_1 = 0$) and $m + n - 1$ equations from the positive edges and so will always have a unique solution.

We now repeat the following until $C_{i j} \geq \lambda_i + \mu_j\ \forall (i, j)$:
\begin{enumerate}
  \item Pick some $(i, j)$ such that $C_{i j} < \lambda_i + \mu_j$.
  \item Introduce an edge from supplier $i$ to supplier $j$.
    Since the positive edges form a spanning tree, doing this will always introduce a cycle.
  \item Send some flow $x_{i j} = \theta$ from $i$ to $j$.
    This will exceed the demand of consumer $j$ and so we decrease the supply along the other edge into consumer $j$ by $\theta$.
    This in turn will mean the supply of another supplier is not being used up and so we increase the supply along the corresponding edge by $\theta$ \textit{and so on...}

    We repeat this, alternating the sign of $\theta$, until we have gone round the cycle and are back at supplier $i$.
  \item We now increase $\theta$ until $x_{i^{*} j^{*}} = 0$ for some edge in the cycle.
  \item Create a fresh transport tableau and fill in the new flows.
  \item We now notice that $x_{i j} = \theta > 0$ and $x_{i^{*} j^{*}} = 0$ and so the positive edges still form a spanning tree with $m + n - 1$ edges.
    We can therefore find a new dual $\vec{\lambda}$ and $\vec{\mu}$ for this BFS using complimentary slackness as we did for the starting BFS.
    Copy these new dual solutions into the tableau.
\end{enumerate}

\section{Max-Flow Min-Cut}
\subsection{Problem Statement}
Let $G = (V, E)$ be a directed graph with $V = \{v_1, \ldots, v_n\}$ where there is a single source vertex $v_1$ and a single sink vertex $v_n$.
Suppose also that each edge $(i, j) \in E$ has maximum capacity $C_{i j}$.
\begin{definition}[Max-Flow Problem]
  The \textit{max-flow problem} finds the maximum possible flow $\delta$ through the graph between the source and sink vertex.
  It can be stated as:
  \begin{maxi*}
    {\delta \in \R, x_{i j} \in \R\ \forall (i, j) \in E}{\delta}{}{}
    \addConstraint{\sum_{j: (1, j) \in E} x_{1 j} - \sum_{i: (i, 1) \in E} x_{i 1}}{= \delta}
    \addConstraint{\sum_{j: (n, j) \in E} x_{n j} - \sum_{i: (i, 1) \in E} x_{i n}}{= -\delta}
    \addConstraint{\sum_{k: (i, k) \in E} x_{i k} - \sum_{k: (k, i) \in E} x_{k i}}{= 0\ \forall i \notin \{1, n\}}
    \addConstraint{0 \leq x_{i j}}{\leq C_{i j}}\ \forall (i, j) \in E
  \end{maxi*}
  The quantities in the above can be explained as follows:
  \begin{itemize}
    \item $\delta$ is the flow into the source vertex and also the flow out of the sink vertex.
    \item The first constraint ensures that the total flow entering the graph at the source vertex $v_1$ is $\delta$ and the second constraint ensures that the total flow exiting the graph at the sink vertex $v_n$ is also $\delta$.
    \item The third constraint ensures that the flow is conserved at all intermediate vertices (i.e. total flow exiting the graph at all other vertices is 0).
    \item Finally, the last constraint enforces the maximum edge capacities.
  \end{itemize}
\end{definition}
It is convenient to define $\phi_i(x)$ to be the flow entering the graph at vertex $i$, i.e. the difference between the total exiting $v_i$ from edges and the total entering $v_i$ from edges:
\[
  \phi_i(x) = \sum_{j: (i, j) \in E} x_{i j} - \sum_{j: (j, i) \in E} x_{j i}
\]
then the problem can be equivalently reformulated as:
\begin{maxi*}
  {\delta \in \R, x_{i j} \in \R\ \forall (i, j) \in E}{\delta}{}{}
  \addConstraint{\phi_1(x)}{= \delta}
  \addConstraint{\phi_n(x)}{= -\delta}
  \addConstraint{\phi_i(x)}{= 0}\ \forall i \notin \{1, n\}
  \addConstraint{0 \leq x_{i j}}{\leq C_{i j}}\ \forall (i, j) \in E
\end{maxi*}
Further, if we define:
\[
  b_i = \begin{cases}
  +1 & \text{ if } i = 1 \\
  -1 & \text{ if } i = n \\
  0 & \text{ otherwise }
  \end{cases}
\]
then the problem becomes:
\begin{maxi*}
  {\delta \in \R, x_{i j} \in \R\ \forall (i, j) \in E}{\delta}{}{}
  \addConstraint{\phi_i(x) - b_i \delta}{= 0\ \forall i \in \{1, \ldots, n\}}
  \addConstraint{0 \leq x_{i j}}{\leq C_{i j}\ \forall (i, j) \in E}
\end{maxi*}
\subsection{Cuts}
Since the max flow problem is just a linear program, we could solve it using techniques such as the simplex algorithm.
However, for this problem, we can utilise an entirely different method with \textit{cuts}.
\begin{definition}[Cut]
  A \textit{cut} of a graph $G$ is a partition of the vertex set $V$ into $S$ and $V \setminus S$.
\end{definition}
\begin{definition}[Cut Capacity]
  The \textit{capacity} of a cut is defined as:
  \[
    C(S) = \sum_{\{(i, j) \in E: i \in S, j \notin S\}} C_{i j}
  \]
  That is, the total capacity of the edges which go from partition $S$ to partition $V \setminus S$.
\end{definition}
\begin{theorem}
  Let $(x_{i j})$ be a feasible flow for the max-flow problem with value $\delta$.
  For any cut $(S, V \setminus S)$ such that $v_1 \in S$ and $v_n \in V \setminus S$ (i.e. $v_1$ and $v_n$ are separated by the cut), we have: \label{cutIneq}
  \[
    \delta \leq C(S)
  \]
\end{theorem}
\begin{remark}
  Since this is true for all feasible flows and for all cuts that separate the source and the sink vertex, it follows that:
  \[
    \max_{x_{i j}} \delta \leq \min_{S} C(S)
  \]
  That is, $\text{maximum flow} \leq \text{minimum cut}$
\end{remark}
\begin{proof}
  For any sets $A, B \subseteq V$ we define:
  \[
    f_x(A, B) = \sum_{\{(i, j) \in E: i \in A, j \in B\}} x_{i j}
  \]
  which is the total flow from vertices in $A$ into vertices in $B$.
  Note that $A$ and $B$ need not be disjoint.

  We then sum up the flow constraints over all vertices in $S$:
  \[
    \sum_{i \in S} \left[\sum_{j: (i, j) \in E} x_{i j} - \sum_{j: (j, i) \in E} x_{i j}\right] = \delta
  \]
  This is just $\delta$ since the summand is 0 for $i \neq 1$ (as $n \notin S$ by assumption) and $\delta$ for $i = 1$.

  We can also rewrite and bound this sum using $f_x$:
  \begin{align*}
    \delta &= \sum_{i \in S} \left[\sum_{j: (i, j) \in E} x_{i j} - \sum_{j: (j, i) \in E} x_{i j}\right] \\
           &= \sum_{i \in S} \left[\sum_{j: (i, j) \in E} x_{i j}\right] - \sum_{i \in S} \left[\sum_{j: (j, i) \in E} x_{i j}\right] \\
           &= f_x(S, V) - f_x(V, S) \\
           &= [f_x(S, S) + f_x(S, V \setminus S)] - [f_x(S, S) + f_x(V\setminus S, S)] \text{ as $V = S \cup (V \setminus S)$} \\
           &= f_x(S, V \setminus S) - f_x(V \setminus S, S) \\
           &\leq f_x(S, V \setminus S) \text{ as $f_x(V\setminus S, S) \geq 0$ since $x_{i j} \geq 0$}\\
           &\leq C(S) \text{ as $x_{i j} \leq C_{i j}$}
  \end{align*}
  as required.
\end{proof}
\begin{theorem}[Max-Flow Min-Cut Theorem]
  Let $\delta^{*}$ be the optimal value of a max-flow problem with optimal flow $x$.
  We then have: \label{maxmin}
  \[
    \delta^{*} = \min\{C(S): S \subseteq V, v_1 \in S, v_n \in V \setminus S\}
  \]
  That is, the maximal flow is equal to the minimal cut capacity of the graph.
\end{theorem}
\begin{proof}
  From \cref{cutIneq}, we know that $\delta^{*} \leq \min_{S} C(S)$.
  Thus it is sufficient to find a cut $S$ such that $C(S) = \delta^{*}$ as then we have $\delta^{*} \leq \min_{S} C(S) \leq C(S) = \delta^{*}$.

  We first need to introduce a few definitions:
  \begin{itemize}
    \item We define a \textit{path} to be a sequence of vertices such that every pair of adjacent vertices in the sequence is connected by an edge, irrespective of its direction.
    \item We call an edge connecting adjacent vertices in a path a \textit{forward edge} if is directed forwards in the sequence of vertices and call it a \textit{reverse edge} otherwise.
    \item Finally, we define an \textit{augmenting path} to be a path such that every \textit{forward edge} satisfies $x_{i j} < C_{i j}$ (i.e. the flow is not full), and every \textit{reverse edge} satisfies $x_{i j} > 0$ (i.e. the flow is non-zero).
  \end{itemize}
  We now note that if $x$ is optimal solution, then $x$ \textbf{cannot} have any augmenting paths from 1 to $n$ as if an augmenting path were to exist, we can would be able to increase the flow along that path without violating constraints (thus increasing $\delta$) and so $x$ would not be optimal.

  Define the cut $S$ as follows:
  \[
    S = \{1\} \cup \{i \in V: \text{there is an augmenting path from 1 to $i$}\}
  \]
  Observe that this is a valid cut for the theorem as $1 \in S$ but $n \notin S$ since $x$ is optimal.

  For the set $S$, we now sum up the flow constraints of $S$ as in the proof of \cref{cutIneq}, we have:
  \[
    \delta^{*} = f_x(S, V \setminus S) - f_x(V \setminus S, S)
  \]
  We must have $f_x(S, V \setminus S) = C(S)$ since every edge from $S$ into $V \setminus S$ must be saturated.
  This is because if we had an edge $(i, j)$ with $i \in S$ and $j \in V \setminus S$, since we already know there is an augmenting path from $1 \to i$, if $x_{i j} < C_{i j}$, then we would be able to extend this into an augmenting path from $1 \to j$, violating the definition of $S$.
  For a similar reason, we must also have $f_x(V \setminus S, S) = 0$ since every edge from $V \setminus S$ into $S$ must have zero flow.

  Thus $\delta^{*} = C(S)$ as required.
\end{proof}
\begin{remark}
  We can also prove the above using by showing that the dual of the max-flow problem is equivalent to minimising the cut.
  This is somewhat fiddly as the initial formulation of the dual does not look like a cut problem.
\end{remark}
\subsection{Ford-Fulkerson Algorithm}
We can utilise what we just found out about augmenting paths within the \textit{Ford-Fulkerson algorithm} to solve the max-flow min-cut problem.
\subsubsection{Algorithm Steps}
\begin{enumerate}
  \item Start with any feasible flow $(x_{i j})$, for example, by picking $x_{i j} = 0$ for all $(i, j) \in E$.
  \item Search for an augmenting path from 1 to n.
    If there is no such augmenting path, the flow is optimal by \cref{maxmin} and so we stop.
    Otherwise, since there is such an augmenting path, we increase the flow along this path as much as possible.
  \item Repeat step \textbf{(ii)}.
\end{enumerate}
If we believe that there are no more augmenting paths from 1 to $n$ and our solution has flow $\delta$, we can prove this is optimal by finding a cut $S$ with capacity $\delta$ and appealing to \cref{maxmin} as then:
\[
  \delta \leq \delta^{*} = \min_{S} C(S) \leq C(S) = \delta \implies \delta^{*} = \delta
\]
\begin{remark}
  For graphs with integer capacities, the Ford-Fulkerson algorithm will always terminate after a finite number of iterations.

  This is because if we start with integral $x_{i j}$, then the algorithm maintains this throughout whilst also increasing the flow from 1 to $n$ by an integral amount with each iteration.
\end{remark}
