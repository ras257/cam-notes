\chapter{Linear Programming}
\section{Formulation}
\begin{definition}[Linear Program]
  A \textit{linear program} or \textit{LP} has general form:
  \begin{mini}
    {\vec{x} \in \R^{n}}{\vec{c}^{\trans}\vec{x}}{}{\label{linearGeneral}}
    \addConstraint{\vec{a}^{\trans}_i \vec{x}}{\geq b_i,\ i \in M_1}
    \addConstraint{\vec{a}^{\trans}_i \vec{x}}{\leq b_i,\ i \in M_2}
    \addConstraint{\vec{a}^{\trans}_i \vec{x}}{= b_i,\ i \in M_3}
    \addConstraint{x_j}{\geq 0,\ j \in N_1}
    \addConstraint{x_j}{\leq 0,\ j \in N_2}
    \addConstraint{x_j}{\text{ free},\ j \in N_3}
  \end{mini}
  for indexing sets $M_1, M_2, M_3, N_1, N_2, N_3$ and $\vec{a}_i, \vec{c} \in \R^{n}$.
  There are $m$ constraints, i.e. $|M_1| + |M_2| + |M_3| = m$, and $n$ components of $\vec{x}$, i.e. $|N_1| + |N_2| + |N_3| = n$.

  That is, a linear program is a constrained minimisation problem in which the objective function and constraints are all linear functions.
\end{definition}
It is convenient if we package up the first three types of constraints into a single matrix $A$:
\[
  A = \begin{pmatrix}
  \leftarrow \vec{a}^{\trans}_1 \rightarrow \\
  \leftarrow \vec{a}^{\trans}_2 \rightarrow \\
  \cdots \\
  \leftarrow \vec{a}^{\trans}_m \rightarrow
  \end{pmatrix}
\]
\begin{theorem}
  The dual of the problem in \cref{linearGeneral} is also a linear program with objective function:
  \begin{equation} \label{linearDual}
    g(\vec{\lambda}) = \vec{b}^{\trans} \vec{\lambda} \text{ where } \Lambda = \left\{\vec{\lambda}: \begin{cases}
    \lambda_i \geq 0 & \text{ for } i \in M_1 \\
    \lambda_i \leq 0 & \text{ for } i \in M_2 \\
    \lambda_i \text{ free} & \text{ for } i \in M_3 \\
    \vec{\lambda}^{\trans} \vec{A}_j \leq c_j & \text{ for } j \in N_1 \\
    \vec{\lambda}^{\trans} \vec{A}_j \geq c_j & \text{ for } j \in N_2 \\
    \vec{\lambda}^{\trans} \vec{A}_j = c_j & \text{ for } j \in N_3
    \end{cases}\right\}
  \end{equation}
\end{theorem}
\begin{proof}
  Adding slack variables $\vec{s} \geq \vec{0}$ to turn the inequality constraints into equality constraints:
  \begin{mini*}
    {\vec{x} \in \R^{n}, \vec{s} \geq \vec{0}}{\vec{c}^{\trans} \vec{x}}{}{}
    \addConstraint{\vec{a}^{\trans}_i \vec{x} - s_i}{= b_i,\ i \in M_1}
    \addConstraint{\vec{a}^{\trans}_i \vec{x} + s_i}{= b_i,\ i \in M_2}
    \addConstraint{\vec{a}^{\trans}_i \vec{x}}{= b_i,\ i \in M_3}
    \addConstraint{x_j}{\geq 0,\ j \in N_1}
    \addConstraint{x_j}{\leq 0,\ j \in N_2}
    \addConstraint{x_j}{\text{ free},\ j \in N_3}
  \end{mini*}
  we construct the Lagrangian as:
  \begin{align*}
    L(\vec{x}, \vec{s}, \vec{\lambda}) &= \vec{c}^{\trans} \vec{x} - \vec{\lambda}^{\trans} \\
                                       &= \vec{c}^{\trans} \vec{x} - \sum_{i \in M_1} \lambda_i (\vec{a}^{\trans}_i \vec{x} - s_i - b_i) \\
                                       &- \sum_{i \in M_2} \lambda_i(\vec{a}_i^{\trans} \vec{x} + s_i - b_i) - \sum_{i \in M_3} \lambda_i(\vec{a}^{\trans}_i \vec{x} - b_i) \\
                                       &= \sum_{j \in N_1} (c_j - \vec{\lambda}^{\trans} \vec{A}_j)x_j + \sum_{j \in N_2} (c_j - \vec{\lambda}^{\trans} \vec{A}_j)x_j + \sum_{j \in N_3} (c_j - \vec{\lambda}^{\trans} \vec{A}_j)x_j \\
                                       &+\sum_{i \in M_1} \lambda_i s_i - \sum_{i \in M_2} \lambda_i s_i + \sum_{i \in M_1 \cup M_2 \cup M_3} \lambda_i b_i
  \end{align*}
  where $\vec{A}_j$ is the $j$-th column of $A$.

  Recall from \cref{dualitySec} that the dual objective function is given by $g(\vec{\lambda}) = \min_{\vec{x} \in \R^{m}, \vec{s} \geq \vec{0}} L(\vec{x}, \vec{s}, \vec{\lambda})$.
  We need to find the set $\Lambda$ of $\vec{\lambda}$ such that a solution to this minimisation problem exists.
  Considering the above expression, we conclude that:
  \begin{itemize}
    \item For $j \in N_1$, since $x_j \geq 0$, we need $c_j - \vec{\lambda}^{\trans} \vec{A}_j \geq 0$.
    \item For $j \in N_2$, since $x_j \leq 0$, we need $c_j - \vec{\lambda}^{\trans} \vec{A}_j \leq 0$.
    \item For $j \in N_3$, since $x_j$ is unrestricted, we need $c_j - \vec{\lambda}^{\trans} \vec{A}_j = 0$.
    \item For $i \in M_1$, since $s_i \geq 0$, we need $\lambda_i \geq 0$.
    \item For $i \in M_2$, since $s_i \geq 0$, we need $\lambda_i \leq 0$.
    \item For $i \in M_3$, there are no restrictions on $\lambda_i$.
  \end{itemize}
  Thus the set of $\vec{\lambda}$ for which $g(\vec{\lambda})$ is well defined is given by:
  \[
    \Lambda = \left\{\vec{\lambda}: \begin{cases}
    \lambda_i \geq 0 & \text{ for } i \in M_1 \\
    \lambda_i \leq 0 & \text{ for } i \in M_2 \\
    \lambda_i \text{ free} & \text{ for } i \in M_3 \\
    \vec{\lambda}^{\trans} \vec{A}_j \leq c_j & \text{ for } j \in N_1 \\
    \vec{\lambda}^{\trans} \vec{A}_j \geq c_j & \text{ for } j \in N_2 \\
    \vec{\lambda}^{\trans} \vec{A}_j = c_j & \text{ for } j \in N_3
    \end{cases}\right\}
  \]
  Now if we take $\vec{\lambda} \in \Lambda$, we notice that the first five terms of the expression will always be non-negative so we choose $\vec{x} = \vec{0}$ and $\vec{s} = \vec{0}$ to minimise those terms, thus the optimal value function is:
  \[
    g(\vec{\lambda}) = \min_{\vec{x} \in \R^{n}, \vec{s} \geq \vec{0}} L(\vec{x}, \vec{s}, \vec{\lambda}) = \sum_{i \in M_1 \cup M_2 \cup M_3} \lambda_i b_i = \vec{b}^{\trans} \vec{\lambda}
  \]
  So the dual of the problem is:
  \begin{maxi*}
    {\vec{\lambda} \in \Lambda}{\vec{b}^{\trans} \vec{\lambda}}{}{}
  \end{maxi*}
  which is also a linear program.
\end{proof}
\begin{remark}
  It is useful to notice how the inequality constraints on $\vec{a}^{\trans}_i \vec{x}$ turn into inequality constraints on $\lambda_i$ with flipped sign and equality turns into $\lambda_i$ free.
  Further, sign constraints on $x_i$ turn into inequality constraints on $\vec{\lambda}^{\trans} \vec{A}_j$ (the columns of $A$ instead of the rows) with flipped signs and free $x_i$ turn into equality constraints.
\end{remark}
\section{Optimality Conditions}
\begin{theorem}
  Let $\vec{x}$ be a feasible solution to the primal problem in \cref{linearGeneral} (i.e. it satisfies the linear constraints) and let $\vec{\lambda}$ be a feasible solution to the dual problem in \cref{linearDual} (i.e. $\vec{\lambda} \in \Lambda$) \label{optimalityConds}

  We then have that these are optimal for the primal and dual respectively if and only if:
  \begin{equation}
    \lambda_i (\vec{a}^{\trans}_i \vec{x} - b_i) = 0\ \forall i \text{ and } (c_j - \vec{\lambda}^{\trans}\vec{A}_j)x_j = 0\ \forall j \label{lambdaSystem}
  \end{equation}
\end{theorem}
\begin{remark}[Complementary Slackness]
  Recall the concept of \textit{complementary slackness} from \cref{complementarySlackness}.
  The first condition above covers complementary slackness for the primal problem and the second condition covers complementary slackness for the dual problem.
\end{remark}
\begin{remark}[Use]
  If we want to check that particular feasible $\vec{x}$ is optimal, then we can try to solve \cref{lambdaSystem} as a system of equations in $\vec{\lambda}$.
  If the solution for this is dual feasible, then we know that $\vec{x}$ is indeed the optimal solution.
\end{remark}
\begin{proof}
  Define $u_i = \lambda_i(\vec{a}^{\trans}_i \vec{x} - b_i)$ and $v_j = (c_j - \vec{\lambda}^{\trans} \vec{A}_j)x_j$.
  Since $\vec{x}$ and $\vec{\lambda}$ are primal and dual feasible respectively, as noted in the previous remark, by using both the definition of $\Lambda$ and the constraints on the original problem that we have that $u_i \geq 0\ \forall i$ and $v_j \geq 0\ \forall j$.

  Taking the sum over all $i$ and $j$ respectively:
  \begin{align*}
    \sum_{i = 1}^{m} u_i &= - \vec{\lambda}^{\trans} \vec{b} + \sum_{i, j} \lambda_i A_{i j} x_j  = \vec{\lambda}^{\trans} A \vec{x} - \vec{\lambda}^{\trans} \vec{b} \\
    \sum_{j = 1}^{n} v_j &= \vec{c}^{\trans} \vec{x} - \sum_{i, j} \lambda_i A_{i j} x_j = \vec{c}^{\trans} \vec{x} - \vec{\lambda}^{\trans} A\vec{x} \\
    \implies \sum_{i = 1}^{m} u_i + \sum_{j = 1}^{n} v_j &= \vec{c}^{\trans} \vec{x} - \vec{\lambda}^{\trans} \vec{b}
  \end{align*}
  By weak duality (\cref{weakDuality}), we know that $\vec{c}^{\trans} \vec{x} \geq \vec{\lambda}^{\trans} \vec{b}$ so the above must be non-negative.
  \begin{proofdirection}{Assume they are optimal}
    Since they are primal and dual optimal respectively, by strong duality (\cref{strongDualityThm}), we have:
    \[
      \vec{c}^{\trans} \vec{x} = \min_{\vec{x}' \in \mathcal{X}(\vec{b})} \vec{c}^{\trans} \vec{x}' =  \max_{\vec{\lambda}' \in \Lambda} \vec{\lambda}' \vec{b} = \vec{\lambda}^{\trans} \vec{b}
    \]
    Thus because $u_i$ and $v_j$ are non-negative, we have $u_i = 0\ \forall i$ and $v_j = 0\ \forall j$.
  \end{proofdirection}
  \begin{proofdirection}{Assume the equality holds}
    It then follows that $\vec{c}^{\trans} \vec{x} = \vec{\lambda}^{\trans} \vec{b}$ and so by strong duality, $\vec{x}$ and $\vec{\lambda}$ are optimal.
  \end{proofdirection}
\end{proof}
\section{Standard Form for Linear Programs}
\subsection{Equivalent forms of Linear Programs} \label{LPEquivForms}
The general form of linear programs given in \cref{linearGeneral} is quite cumbersome to work with so we would like to \textit{reduce it to an equivalent form}:
\begin{enumerate}
  \item First, observe that we can write the sign constraints on the components of $\vec{x}$ in the form of the first two constraints by using $\vec{a}_i$ with a single non-zero component.
  \item We can then convert the equality constraints into pairs of inequality constraints with differing signs.
  \item Finally, we can convert all of the $\vec{a}^{\trans}_i \vec{x} \geq b_i$ inequality constraints to be in the form $\vec{a}^{\trans}_i \vec{x} \leq b_i$ by inverting the sign of $\vec{a}_i$ and $b_i$.
\end{enumerate}
Thus we have converted all of the constraints into the form $\vec{a}^{\trans}_i \vec{x} \leq b_i$ and so we can write the equivalent reduced general form as:
\begin{mini*}
  {\vec{x} \in \R^{n}}{\vec{c}^{\trans} \vec{x}}{}{}
  \addConstraint{A\vec{x}}{\leq \vec{b}}
\end{mini*}
Note that as we have done similarly before, we can convert this inequality constraint into an equality constraint by using a slack vector $\vec{s} \geq \vec{0}$.
\begin{definition}[Standard Form of an LP]
  We say that a linear program is in \textit{standard form} if it is in the form: \label{LPStandard}
  \begin{mini*}
    {\vec{x} \in \R^{n}}{\vec{c}^{\trans} \vec{x}}{}{}
    \addConstraint{A\vec{x}}{= \vec{b}}
    \addConstraint{\vec{x}}{\geq \vec{0}}
  \end{mini*}
\end{definition}
We can easily convert from the standard form to the reduced form using the method above as this is just a specific case.

We can also convert from the reduced general form to standard form.
Firstly, we add a slack vector $\vec{s}$ to convert the inequality constraint into an equality constraint.
We then observe that any $\vec{x} \in \R^{n}$ can be expressed as $\vec{x} = \vec{x}_{+} - \vec{x}_{-}$ where $\vec{x}_{+}, \vec{x}_{-} \geq \vec{0}$:
\begin{mini*}
  {\vec{x} \in \R^{n}}{\vec{c}^{\trans}(\vec{x}_{+} - \vec{x}_{-})}{}{}
  \addConstraint{A(\vec{x}_{+} - \vec{x}_{-}) + \vec{s}}{= \vec{b}}
  \addConstraint{\vec{x}_{+}}{\geq 0}
  \addConstraint{\vec{x}_{-}}{\geq 0}
  \addConstraint{\vec{s}}{\geq 0}
\end{mini*}
We can then package $\vec{x}_{+}, \vec{x}_{-}$ and $\vec{s}$ up into a higher dimensional vector $\vec{z} = (\vec{x}_{+}, \vec{x}_{-}, \vec{s})$ and construct a new vector $\vec{c}'$ and $A'$ such that $\vec{c}^{\trans}(\vec{x}_{+} - \vec{x}_{-}) = {\vec{c}'}^{\trans} \vec{z}$ and $A(\vec{x}_{+} - \vec{x}_{-}) + \vec{s} = A'\vec{z}$ so that the problem becomes:
\begin{mini*}
  {\vec{z}}{{\vec{c}'}^{\trans} \vec{z}}{}{}
  \addConstraint{A'\vec{z}}{= \vec{b}}
  \addConstraint{\vec{z}}{\geq \vec{0}}
\end{mini*}
which is a standard form problem.
\begin{remark}
  When discussing the theory behind linear programs it is often easier to refer to the general form, however when discussing algorithms for solving linear programs it is often easier to refer to the standard form.
\end{remark}
\subsection{Duals of equivalent forms}
It is also useful to find the duals of the above forms of linear programs.
\subsubsection{Dual of a reduced general form LP}
Starting from the reduced general form:
\begin{mini*}
  {\vec{x} \in \R^{n}}{\vec{c}^{\trans} \vec{x}}{}{}
  \addConstraint{A\vec{x}}{\leq \vec{b}}
\end{mini*}
with $\vec{b} \in \R^{m}$.
We first add a slack vector $\vec{s} \geq \vec{0}$:
\begin{mini*}
  {\vec{x} \in \R^{n}, \vec{s} \geq \vec{0}}{\vec{c}^{\trans} \vec{x}}{}{}
  \addConstraint{A\vec{x} + \vec{s}}{= \vec{b}}
\end{mini*}
The dual objective function is given by:
\begin{align*}
  g(\vec{\lambda}) &= \min_{\vec{x} \in \R^{n}, \vec{s} \geq \vec{0}} L(\vec{x}, \vec{s}, \vec{\lambda}) \\
                   &= \min_{\vec{x} \in \R^{n}, \vec{s} \geq \vec{0}} [\vec{c}^{\trans} \vec{x} - \vec{\lambda}^{\trans}(A\vec{x} + \vec{s} - \vec{b})] \\
                   &= \vec{\lambda}^{\trans} \vec{b} +\min_{\vec{x} \in \R^{n}, \vec{s} \geq \vec{0}} [(\vec{c}^{\trans} - \vec{\lambda}^{\trans}A)\vec{x} - \vec{\lambda}^{\trans} \vec{s}]
\end{align*}
To find $\Lambda$, since $\vec{x}$ is unrestricted, we must have $\vec{c}^{\trans} - \vec{\lambda}^{\trans} A = \vec{0} \iff A^{\trans} \vec{\lambda} = \vec{c}$ and since $\vec{s} \geq 0$ we must have $\vec{\lambda} \leq \vec{0}$ else the minimum is just $-\infty$.
Thus $\Lambda$ is:
\[
  \Lambda = \{\vec{\lambda} \in \R^{m}: A^{\trans} \vec{\lambda} = \vec{c},\ \vec{\lambda} \leq \vec{0}\}
\]
For such $\vec{\lambda}$, the optimal value will just be $\vec{\lambda}^{\trans} \vec{b}$ and so the dual problem is:
\begin{maxi*}
  {\vec{\lambda} \in \R^{m}}{\vec{\lambda}^{\trans} \vec{b}}{}{}
  \addConstraint{A^{\trans} \vec{\lambda}}{= \vec{c}}
  \addConstraint{\vec{\lambda}}{\leq \vec{0}}
\end{maxi*}
We could also set $\vec{\mu} = -\vec{\lambda}$ so that this becomes:
\begin{mini}
  {\vec{\mu} \in \R^{m}}{\vec{b}^{\trans} \vec{\mu}}{}{\label{genDual}}
  \addConstraint{A^{\trans}\vec{\mu}}{= - \vec{c}}
  \addConstraint{\vec{\mu}}{\geq \vec{0}}
\end{mini}
which is a LP in standard form.
\subsubsection{Dual of a standard form LP} \label{stdDual}
Starting from the standard form:
\begin{mini*}
  {\vec{x} \in \R^{n}}{\vec{c}^{\trans} \vec{x}}{}{}
  \addConstraint{A \vec{x}}{= \vec{b}}
  \addConstraint{\vec{x}}{\geq \vec{0}}
\end{mini*}
The dual objective function is given by:
\begin{align*}
  g(\vec{\lambda}) &= \min_{\vec{x} \geq \vec{0}} [\vec{c}^{\trans} \vec{x} - \vec{\lambda}^{\trans}(A\vec{x} - \vec{b})] \\
                   &= \vec{\lambda}^{\trans} \vec{b} + \min_{\vec{x} \geq \vec{0}} (\vec{c}^{\trans} - \vec{\lambda}^{\trans} A)\vec{x}
\end{align*}
This minimum only exists when $\vec{c}^{\trans} - \vec{\lambda}^{\trans} A \geq \vec{0} \iff A^{\trans} \vec{\lambda} \leq \vec{c}$ so $\Lambda = \{\vec{\lambda} \in \R^{m}: A^{\trans} \vec{\lambda} = \vec{c}\}$.
For such $\vec{\lambda}$, the minimum value is just $\vec{\lambda}^{\trans} \vec{b}$ and so the dual problem is:
\begin{maxi*}
  {\vec{\lambda} \in \R^{m}}{\vec{b}^{\trans} \vec{\lambda}}{}{}
  \addConstraint{A^{\trans} \vec{\lambda}}{\leq \vec{c}}
\end{maxi*}
\subsubsection{Dual of a dual program}
We know from the above if the primal problem is in standard form then its dual is:
\begin{maxi*}
  {\vec{\lambda} \in \R^{m}}{\vec{b}^{\trans} \vec{\lambda}}{}{}
  \addConstraint{A^{\trans} \vec{\lambda}}{\leq \vec{c}}
\end{maxi*}
If we then set $\vec{\mu} = -\vec{\lambda}$ this becomes:
\begin{mini*}
  {\vec{\mu} \in \R^{m}}{\vec{b}^{\trans} \vec{\mu}}{}{}
  \addConstraint{(-A^{\trans})\vec{\mu}}{\leq \vec{c}}
\end{mini*}
Which is in the general form, thus from \cref{genDual} its dual is given by:
\begin{mini*}
  {\vec{\theta} \in \R^{n}}{\vec{c}^{\trans} \vec{\theta}}{}{}
  \addConstraint{(-A^{\trans})^{\trans} \vec{\theta}}{= - \vec{b}}
  \addConstraint{\vec{\theta}}{\geq \vec{0}}
\end{mini*}
which is equivalent to:
\begin{mini*}
  {\vec{\theta} \in \R^{n}}{\vec{c}^{\trans} \vec{\theta}}{}{}
  \addConstraint{A \vec{\theta}}{= \vec{b}}
  \addConstraint{\vec{\theta}}{\geq \vec{0}}
\end{mini*}
So taking the dual of an LP twice recovers the original program.
\section{Solving Linear Programs}
We first note that $\vec{c}^{\trans} \vec{x}$ is equivalent to maximising $-\vec{c}^{\trans} \vec{x}$.
But $-\vec{c}^{\trans} \vec{x}$ is a convex function, so by solving a linear program we are trying to maximise a convex function over some restricted region.
Usefully, can make some statements about where within a convex set the maximiser of a convex function will lie.
\begin{remark}[Warning]
  In \cref{chap1} we were \textbf{minimising} convex functions over a mix of unrestricted and restricted sets, here we are trying to \textbf{maximise} convex functions only over restricted sets.
\end{remark}
\begin{definition}[Extreme Point]
  Let $C \subseteq \R^{n}$ be a convex set.
  A point $\vec{x} \in C$ is called an \textit{extreme point} if it cannot be written as a convex combination of two distinct points in $C$.

  That is, for all $\vec{y}, \vec{z} \in C$ and $\delta \in (0, 1)$:
  \[
    \vec{x} = \delta \vec{y} + (1 - \delta) \vec{z} \implies \vec{x} = \vec{y} = \vec{z}
  \]
\end{definition}
\begin{remark}[Intuition]
  A point is extreme if it does not lie on any line segment \textbf{contained within} $C$.

  So if $C$ is a triangle, only its vertices are extreme points, and if $C$ is a circle, then only its boundary points are extreme points.
\end{remark}
\begin{lemma}
  A convex function $f$ on a convex set $S$ has at least one maximiser at an extreme point of $C$.
\end{lemma}
\begin{remark}
  This means that to find the optimal value we need only compare values of $f$ at all the extreme points of $C$.
\end{remark}
\begin{proof}
  Not lectured.
\end{proof}
Although the full proof is not covered in this course, we can get an intuition for why this is true.
Suppose that $\vec{x}$ is a maximiser of $f$ but is not an extreme point.
Thus there is $\vec{y}, \vec{z} \in C$ and $\delta \in (0, 1)$ such that $\vec{x} = \delta \vec{y} + (1 - \delta)\vec{z}$.
Since $f$ is convex:
\begin{align*}
  f(\vec{x}) &\leq \delta f(\vec{y}) + (1 - \delta)f(\vec{z}) \\
             &\leq \delta f(\vec{x}) + (1 - \delta)f(\vec{x}) \text{ since $\vec{x}$ is a maximiser} \\
             &= f(\vec{x})
\end{align*}
Thus we must have $f(\vec{x}) = f(\vec{y}) = f(\vec{z})$ and so $\vec{y}$ and $\vec{z}$ are also optimal.
The remainder of the proof relies on that, if the set is suitably well behaved, we can use this to ``move towards'' an extreme point of the set, showing that it is also optimal.

In a linear program, the constraint set is always some high dimensional polytope.
Such regions always have finitely many extreme points and so if we can find a list of these extreme points we can simply check the value of $-\vec{c}^{\trans} \vec{x}$ at these extreme points in order to find the maximum.
\subsection{Basic Solutions and Basic Feasible Solutions}
Consider a linear program in the standard form (\cref{LPStandard}):
\begin{mini*}
  {\vec{x} \in \R^{n}}{\vec{c}^{\trans} \vec{x}}{}{}
  \addConstraint{A\vec{x}}{= \vec{b}}
  \addConstraint{\vec{x}}{\geq \vec{0}}
\end{mini*}
for a matrix $A$ and vector $\vec{b} \in \R^{m}$ where the matrix $A$ has $m$ rows and $n$ columns given by:
\[
  A = \begin{pmatrix}
  \leftarrow \vec{a}^{\trans}_1 \rightarrow \\
  \leftarrow \vec{a}^{\trans}_2 \rightarrow \\
  \cdots \\
  \leftarrow \vec{a}^{\trans}_m \rightarrow
  \end{pmatrix} = \begin{pmatrix}
  \uparrow & \uparrow &  & \uparrow \\
  \vec{A}_1 & \vec{A}_2 & \cdots & \vec{A}_n \\
  \downarrow & \downarrow &  & \downarrow \\
  \end{pmatrix}
\]
We will make three assumptions:
\begin{enumerate}
  \item The rows of $A$ are linearly independent.
    This can in fact be made without loss in generality as if the rows are linearly dependant then we have redundant constraints and so we can simply remove the redundant rows.
  \item Every subset of $m$ columns of $A$ is linearly independent.
    This cannot be made without loss in generality, but for most real world applications it will usually hold, for example the components of $A$ are measurements of some physical system.
\end{enumerate}
For our final assumption, we will need some additional definitions.
\begin{definition}[Basic Solution]
  We call a point $\vec{x}$ that satisfies $A\vec{x} = \vec{b}$ and has at most $m$ non-zero entries a \textit{basic solution}.
\end{definition}
\begin{definition}[Basic Feasible Solution]
  We call a basic solution $\vec{x}$ that satisfies $\vec{x} \geq \vec{0}$ a \textit{basic feasible solution} or \textit{BFS}.
\end{definition}
Suppose we choose indices $\{B(1), \ldots, B(m)\} \subseteq \{1, \ldots, n\}$ to be the components of $\vec{x}$ that we allow to be non-zero, i.e. $x_i = 0$ for all $i \notin \{B(1), \ldots, B(m)\}$.
\[
  A \vec{x} = \sum_{i = 1}^{n} \vec{A}_i x_i = \sum_{i = 1}^{m} \vec{A}_{B(i)}\vec{x}_{B(i)} = B \vec{x}_{B}
\]
where $B$ is the $m \times m$ square matrix and $\vec{x}_B \in \R^{m}$ given by:
\[
  B = \begin{pmatrix}
  \uparrow & & \uparrow  \\
  \vec{A}_{B(1)} & \cdots & \vec{A}_{B(m)}  \\
  \downarrow &  & \downarrow \\
  \end{pmatrix}
 \text{ and } \vec{x}_{B} = \begin{pmatrix}
  x_{B(1)} \\
  \vdots \\
  x_{B(m)} \\
  \end{pmatrix}
\]
By assumption \textbf{(ii)} columns of $B$ must be linearly independent and so $B$ is invertible.
Thus:
\[
  A\vec{x} = \vec{b} \iff B\vec{x}_B = \vec{b} \iff \vec{x}_{B} = B^{-1} \vec{b}
\]
so by constructing an $\vec{x}$ from $\vec{x}_B$ above by adding zero entries for $i \notin \{B(1), \ldots, B(m)\}$, we obtain a basic solution.
\begin{definition}[Variable Names]
  We call:
  \begin{enumerate}
    \item $\{B(1), \ldots, B(m)\}$ the \textit{basic indices},
    \item $x_{B(1), \ldots, x_{B(m)}}$ the \textit{basic variables},
    \item $B$ the \textit{basis matrix},
    \item $\vec{A}_{B(1)}, \ldots, \vec{A}_{B(m)}$ the \textit{basic columns}.
  \end{enumerate}
\end{definition}
Since there are $n$ columns of $A$, there are $\binom{n}{m}$ ways for us to pick the basis matrix $B$.
For each such choice, we obtain a basic solution $\vec{x}_{B} = B^{-1} \vec{b}$.
So if $\vec{x}_B = B^{-1}\vec{b} \geq 0$, since the other components of $\vec{x}$ must be 0, then we have a basic \textbf{feasible} solution.

We can now state our third assumption:
\begin{enumerate}
  \setcounter{enumi}{2}
  \item Every basic feasible solution $\vec{x}$ is \textit{non-degenerate}.
    That is, it has \textbf{exactly} $m$ non-zero entries and so all of the basic variables must be non-zero.
    Algebraically:
    \[
      B^{-1} \vec{b} \geq 0 \implies B^{-1} \vec{b} > 0
    \]
    This also cannot be made without loss in generality but is required to ensure that algorithms later in this chapter do not get stuck.
\end{enumerate}
\begin{theorem}
  A vector $\vec{x}$ is a basic feasible solution if and only if it is an extreme point of the set:
  \[
    \mathcal{X}(\vec{b}) = \{\vec{x} \in \R^{n}: A \vec{x} = \vec{b},\ \vec{x} \geq 0\}
  \]
\end{theorem}
\begin{remark}[Rudimentary Algorithm for Solving a LP]
  This theorem means that we can solve an LP by just checking each of the basic feasible solutions, that is: \label{rudimentaryAlgorithm}
  \begin{itemize}
    \item List all $\binom{n}{m}$ basic solutions by generating all possible basis matrices and finding their corresponding feasible solutions $B^{-1}\vec{b}$.
    \item Filter these basic solutions to just the basic feasible solutions (i.e. checking if it is non-negative)
    \item Evaluate $\vec{c}^{\trans} \vec{x}$ for each of these basic feasible solutions and choose the best one.
  \end{itemize}
  This algorithm works, however it becomes slow for problems in high dimensions due to the rapid growth of $\binom{n}{m}$ since we need to check the value of $\vec{c}^{\trans} \vec{x}$ at \textbf{every basic feasible solution} before we can determine the optimal value.
\end{remark}
\begin{proof}
  \begin{proofdirection}{Assume $\vec{x}$ is a basic feasible solution}
    Suppose that $\vec{x}$ can be written as:
    \[
      \vec{x} = (1 - \delta)\vec{y} + \delta \vec{z}
    \]
    for some $\vec{y}, \vec{z} \in \mathcal{X}(\vec{b})$ and $\delta \in (0, 1)$.

    If $x_i = 0$, then $x_i = 0 = (1 - \delta)y_i + \delta z_i$ but we also have that $y_i, z_i \geq 0$ as $\vec{y}, \vec{z} \in \mathcal{X}(\vec{b})$ and so $y_i = z_i = 0$.
    Thus:
    \[
      i \notin \{B(1), \ldots, B(m)\} \implies x_i = 0 \implies y_i = z_i = 0
    \]
    We can then define $\vec{y}_B$ and $\vec{z}_B$ to contain the allowed non-zero entries of $\vec{y}$ and $\vec{z}$ as we did for $\vec{x}$ earlier.
    Since $\vec{y}, \vec{z} \in \mathcal{X}(\vec{b})$, we have $A\vec{z} = A\vec{y} = \vec{b}$.
    Thus by the argument as before, $\vec{y}_B = \vec{z}_B = B^{-1} \vec{b} = \vec{x}_B$.
    This means that the zero entries and the basic variables of $\vec{x}, \vec{y}, \vec{z}$ agree and so $\vec{x} = \vec{y} = \vec{z}$, i.e. $\vec{x}$ is an extreme point of $\mathcal{X}(\vec{b})$.
  \end{proofdirection}
  It is easier if we instead show the contrapositive of the reverse direction.
  \begin{proofdirection}{Assume $\vec{x}$ is not a basic feasible solution}
    We wish to show that $\vec{x}$ is not an extreme point of $\mathcal{X}(\vec{b})$.
    If $\vec{x} \notin \mathcal{X}(\vec{b})$, then obviously it is not an extreme point  of $\mathcal{X}(\vec{b})$ and so we assume that $\vec{x} \in \mathcal{X}(\vec{b})$.

    Since $\vec{x}$ \textbf{is not} a basic feasible solution, it must have $r > m$ positive entries which we will label as $x_{i_1} > 0, x_{i_2} > 0, \ldots, x_{i_r} > 0$.
    Because $\vec{A}_j \in \R^{m}$ and $r > m$, the set of columns $\{\vec{A}_{i_1}, \ldots, \vec{A}_{i_r}\}$ must be linearly dependent.
    By definition this means that $\exists w_{i_1}, \ldots, w_{i_r} \in \R$ not all 0 such that:
    \[
      w_{i_1} \vec{A}_{i_1} + \cdots + w_{i_r} \vec{A}_{i_r} = \vec{0}
    \]
    Construct a vector $\vec{w} \in \R^{n}$ with zero entries everywhere except indices $\{i_1, \ldots, i_r\}$ where the entries are given by $w_{i_j}$.

    Now observe that $A\vec{w} = \vec{0}$ because:
    \begin{align*}
      A \vec{w} &= \vec{A}_0 w_0 + \cdots + \vec{A}_{i_1} w_{i_1} + \cdots + \vec{A}_{i_r} w_{i_r} + \cdots + \vec{A}_{n} w_{n} \\
                &= \vec{A}_{i_1} w_{i_1} + \cdots + \vec{A}_{i_r} w_{i_r} \text{ as the other entries of $\vec{w}$ are zero} \\
                &= \vec{0} \text{ by construction}
    \end{align*}
    Now consider two new points $\vec{y} = \vec{x} + \varepsilon \vec{w}$ and $\vec{z} = \vec{x} - \varepsilon \vec{w}$ for some $\varepsilon > 0$.
    Since $\vec{x} \in \mathcal{X}(\vec{b})$ and $A \vec{w} = \vec{0}$, we have $A\vec{y} = A\vec{z} = \vec{b}$.
    We also have that $\vec{x} \geq 0$ and note that because $\vec{w}$ has a zero component in every zero component of $\vec{x}$, adding $\varepsilon \vec{w}$ only affects the positive entries of $\vec{x}$.
    Thus we can pick $\varepsilon$ to be sufficiently small so that $\vec{y}, \vec{z} \geq \vec{0}$ and thus $\vec{y}, \vec{z} \in \mathcal{X}(\vec{b})$.
    But we can write $\vec{x} = \frac{1}{2}\vec{y} + \frac{1}{2}\vec{z}$ and so $\vec{x}$ is not an extreme point of $\mathcal{X}(\vec{b})$ as desired.
  \end{proofdirection}
\end{proof}
\subsection{Towards the Simplex Algorithm}
In our current rudimentary algorithm from \cref{rudimentaryAlgorithm}, we need to check the value of $\vec{c}^{\trans} \vec{x}$ at \textbf{every} basic feasible solution before we can determine which one is the minimum.
However, for the \textit{simplex algorithm}, we would like a simple way to check if a basic feasible solution is optimal so that we can just test basic feasible solutions \textbf{until} we find one which is optimal instead of testing all of them.

Recall from \cref{stdDual} that, if the primal problem is a standard LP:
\begin{mini*}
  {\vec{x} \in \R^{n}}{\vec{c}^{\trans} \vec{x}}{}{}
  \addConstraint{A\vec{x}}{= \vec{b}}
  \addConstraint{\vec{x}}{\geq \vec{0}}
\end{mini*}
then its dual is:
\begin{maxi*}
  {\vec{\lambda} \in \R^{m}}{\vec{b}^{\trans} \vec{\lambda}}{}{}
  \addConstraint{A^{\trans} \vec{\lambda}}{\leq \vec{c}}
\end{maxi*}
So we have primal feasibility if and only if $A\vec{x} = \vec{b}$ and $\vec{x} \geq \vec{0}$ and dual feasibility if and only if $A^{\trans} \vec{\lambda} \leq \vec{c}$.
From \cref{optimalityConds}, given primal and dual feasible $\vec{x}$ and $\vec{\lambda}$, the solution is optimal if and only if:
\[
  \lambda_i (\vec{a}^{\trans}_i \vec{x} - b_i) = 0\ \forall i \text{ and } (c_j - \vec{\lambda}^{\trans}\vec{A}_j)x_j = 0\ \forall j
\]
We can now state the optimality conditions for an LP in standard form.
\begin{theorem}
  For an LP in standard form, a primal and dual solution $\vec{x}, \vec{\lambda}$ is optimal if and only if the following hold:
  \begin{enumerate}
    \item \textbf{Primal feasibility --} $A\vec{x} = \vec{b}$, $\vec{x} \geq 0$.
    \item \textbf{Dual feasibility --} $A^{\trans} \vec{\lambda} \leq \vec{c}$
    \item \textbf{Complementary Slackness --} $\vec{x}^{\trans}(\vec{c} - A^{\trans} \vec{\lambda}) = 0$
  \end{enumerate}
\end{theorem}
\begin{proof}
  \begin{proofdirection}{Assume the solution is optimal}
    The solution must be both primal and dual feasible, satisfying \textbf{(i)} and \textbf{(ii)}.
    We then note that:
    \[
      \vec{x}^{\trans}(\vec{c} - A^{\trans} \vec{\lambda}) = \sum_{j = 1}^{n} (c_j - \vec{\lambda}^{\trans} \vec{A}_j)x_j = 0
    \]
    by \cref{optimalityConds} and so \textbf{(iii)} also holds.
  \end{proofdirection}
  \begin{proofdirection}{Assume the conditions hold}
    \textbf{(i)} and \textbf{(ii)} give us primal and dual feasibility.
    From \textbf{(iii)} we have:
    \[
      \vec{x}^{\trans}(\vec{c} - A^{\trans} \vec{\lambda}) = \sum_{j = 1}^{n} (c_j - \vec{\lambda}^{\trans} \vec{A}_j)x_j = 0
    \]
    \textbf{(ii)} also tells us that $c_j - \vec{\lambda}^{\trans} \vec{A}_j \geq 0\ \forall j$ and \textbf{(i)} tells us that $x_j \geq 0\ \forall j$, thus since the sum is zero we must have $(c_j - \vec{\lambda}^{\trans} \vec{A}_j)x_j = 0\ \forall j$.
    Finally from \textbf{(i)} we have $A \vec{x} = \vec{b} \implies \vec{a}^{\trans}_i \vec{x} - b_i = 0\ \forall i$.
    Thus by \cref{optimalityConds}, the solution is optimal.
  \end{proofdirection}
\end{proof}
We first note that for any basic feasible solution with basis matrix $B$, the complementary slackness condition is equivalent to:
\[
  \vec{x}_{B}^{\trans}(\vec{c}_{B} - B^{\trans} \vec{\lambda}) = 0
\]
Since $\vec{x}_{B} > \vec{0}$ by the non-degeneracy assumption we have:
\begin{equation}
  \vec{x}_{B}^{\trans}(\vec{c}_{B} - B^{\trans} \vec{\lambda}) = 0 \iff \vec{c}_B = B^{\trans} \vec{\lambda} \iff \vec{\lambda} = (B^{\trans})^{-1} \vec{c}_B \label{slacknessEquiv}
\end{equation}
\begin{corollary}
  A basic feasible solution is optimal if and only if $A^{\trans} (B^{\trans})^{-1} \vec{c}_B \leq \vec{c}$. \label{BFSCond}
\end{corollary}
\begin{proof}
  Since it is a basic feasible solution, \textbf{(i)} is always satisfied.
  \begin{proofdirection}{Assume the solution is optimal}
    By \cref{slacknessEquiv}, we have $\vec{\lambda} = (B^{\trans})^{-1} \vec{c}_B$ and by \textbf{(ii)} we have $A^{\trans} \vec{\lambda} \leq \vec{c}$ which is equivalent to the above inequality.
  \end{proofdirection}
  \begin{proofdirection}{Assume the inequality is satisfied}
    We define $\vec{\lambda} = (B^{\trans})^{-1} \vec{c}_B$ which satisfies \textbf{(ii)} by assumption and \textbf{(iii)} by \cref{slacknessEquiv} and so the solution is optimal.
  \end{proofdirection}
\end{proof}
This means that we can just check $A^{\trans}(B^{\trans})^{-1}\vec{c}_{B} \leq \vec{c}$ at every basic feasible solution until we find one where the inequality holds.
\begin{definition}[Vector of reduced costs]
  For a basic feasible solution with basis matrix $B$, we define the \textit{vector of reduced costs} $\overline{\vec{c}}$ as:
  \[
    \overline{\vec{c}} = \vec{c} - A^{\trans} (B^{\trans})^{-1} \vec{c}_B
  \]
\end{definition}
So a basic feasible solution is optimal if and only if its vector of reduced costs satisfies $\overline{\vec{c}} \geq \vec{0}$.

To find simple expressions for its components we rewrite:
\[
  \overline{\vec{c}}^{\trans} = \vec{c}^{\trans} - \vec{c}^{\trans}_B B^{-1} A
\]
and so its components are given by:
\begin{equation}
  \overline{c}_j = c_j - \vec{c}^{\trans}_{B} B^{-1} \vec{A}_j \label{reducedCostComponents}
\end{equation}
\section{Simplex Algorithm}
\subsection{Algorithm}
Consider a linear program in standard form:
\begin{mini*}
  {\vec{x} \in \R^{n}}{\vec{c}^{\trans} \vec{x}}{}{}
  \addConstraint{A \vec{x}}{= \vec{b}}
  \addConstraint{\vec{x}}{\geq \vec{0}}
\end{mini*}
We will start at a non-degenerate basic feasible solution $\vec{x}$ with vector of reduced costs $\overline{\vec{c}}$.
\begin{remark}[Notation]
  For notational simplicity, when we are writing out vectors in components form, we assume that the basic indices are given by $B(i) = i$, i.e. we can write $\vec{x}$ as:
  \[
    \vec{x} = (x_{B(1)}, \ldots, x_{B(m)}, 0, \ldots, 0)^{\trans}
  \]
\end{remark}
If $\overline{\vec{c}} \geq \vec{0}$, then $\vec{x}$ is optimal by \cref{BFSCond} and so we stop.
Otherwise, $\vec{x}$ is not optimal and there exists some index $j^{*}$ such that $\overline{c}_{j^{*}} < 0$.
Since we want to keep testing basic feasible solutions, we would like a way to move from this non-optimal basic feasible solution to some other basic feasible solution.

\subsubsection{Choosing a direction}
We start by choosing some index $j \notin \{B(1), \ldots, B(m)\}$ (i.e. $x_j = 0$) to make strictly positive and then perturb on this index to a new position $\vec{x}'$:
\[
  \vec{x}' = (x_{B(1)}, \ldots, x_{B(m)}, 0, \ldots, 0)^{\trans} + t (d_{B(1)}, \ldots, d_{B(m)}, 0, \ldots, 1, \ldots, 0)^{\trans}
\]
where the $1$ is in the $j$-th position, $t > 0$, and $\vec{d}_{B} \in \R^{m}$ to be chosen later.
For this new point to be feasible we need $A\vec{x}' = \vec{b}$:
\begin{align*}
  A\vec{x}' &= \vec{b} \\
  \iff A(\vec{x} + t(d_{B(1)}, \ldots, d_{B(m)}, 0, \ldots, 1, \ldots, 0)^{\trans}) &= \vec{b} \\
  \iff \vec{b} + tA(d_{B(1)}, \ldots, d_{B(m)}, 0, \ldots, 1, \ldots, 0)^{\trans} &= \vec{b} \text{ since $\vec{x}$ is a BFS}\\
  \iff tA(d_{B(1)}, \ldots, d_{B(m)}, 0, \ldots, 1, \ldots, 0)^{\trans} &= \vec{0} \\
  \iff A(\vec{d}_B, 0, \ldots, 1, \ldots, 0)^{\trans} &= \vec{0} \text{ since $t > 0$} \\
  \iff B \vec{d}_{B} + \vec{A}_j &= \vec{0} \\
  \iff \vec{d}_B &= -B^{-1} \vec{A}_j
\end{align*}
Thus we have $A\vec{x}' = \vec{b}$ for this unique choice of $\vec{d}_B$.

\subsubsection{Calculating new cost}
We now consider the cost at $\vec{x}'$:
\begin{align*}
  \vec{c}^{\trans} \vec{x}' &= \vec{c}^{\trans}(\vec{x} + t(\vec{d}_B, 0, \ldots, 1, \ldots, 0)^{\trans}) \\
                            &= \vec{c}^{\trans} \vec{x} + t(\vec{c}^{\trans}_{B} \vec{d}_B + c_j) \\
                            &= \vec{c}^{\trans} \vec{x} + t(c_j - \vec{c}^{\trans}_{B} B^{-1}\vec{A}_j) \\
                            &= \vec{c}^{\trans} \vec{x} + t\overline{c}_j \text{ using \cref{reducedCostComponents}}
\end{align*}
Since we are trying to minimise this and $t > 0$, we want to pick $j$ such that $\overline{c}_j < 0$.
Recall that $j^{*}$ provides such an index and so we take $j = j^{*}$.
This uniquely gives us a direction to move in and tells us by how much we reduce the cost for a given $t$.

\subsubsection{Ensuring basic feasibility}
We now need to pick how big $t$ is.
From the above we want to make $t$ as large as possible in order to reduce the new cost as much as possible.
From how we chose $\vec{d}_B$, we know that regardless of $t$ we will always have $A\vec{x}' = \vec{b}$.
However, for $\vec{x}'$ to be feasible, we also require $\vec{x}' \geq \vec{0}$ and so we want to pick the largest $t$ such that $\vec{x}' \geq \vec{0}$.

By construction, $x'_{j^{*}} > 0$ and $x'_i = 0$ for $j \notin \{B(1), \ldots, B(m), j^{*}\}$.
This means that we need only check how $t$ affects the basic indices:
\[
  x'_{B(i)} = x_{B(i)} + t d_{B(i)}
\]
If $d_{B(i)} \geq 0$, then we do not need to worry about leaving the constrained set in this direction.
But if $d_{B(i)} < 0$, then the $x'_{B(i)}$ could become negative and so we would need to ensure that $t \leq -\frac{x_{B(i)}}{d_{B(i)}}$.

In the case of $\vec{d}_{B} \geq \vec{0}$, then $\vec{x}' \geq 0$ for all $t > 0$ and so the optimal cost is just $-\infty$ by taking $t \to \infty$.

Otherwise, there must be at least one basic index such that $d_{B(i)} < 0$ and so we need to ensure that $t \leq - \frac{x_{B(i)}}{d_{B(i)}}$ for all such basic indices.
We still wish to move as much as possible despite this and so we take:
\[
  t^{*} = \min_{\{i \in \{1, \ldots, m\}: d_{B(i)} < 0\}} -\frac{x_{B(i)}}{d_{B(i)}}
\]
Thus if we define $\vec{d} = (\vec{d}_B, 0, \ldots, 1, \ldots, 0)^{\trans}$ with 1 in the  $j^{*}$ position then $\vec{x}' = \vec{x} + t^{*} \vec{d}$ satisfies $A\vec{x}' = \vec{b}$ and $\vec{x}' \geq \vec{0}$.

We started with $\vec{x}$ having exactly $m$ non-zero entries, we then made the entry $x'_{j^{*}}$ non-zero but then our choice of $t^{*}$ ensures that exactly one of $x'_{B(1)}, \ldots, x'_{B(m)}$ is now zero.
Thus $\vec{x}'$ also has at most $m$ non-zero entries and so is also a non-degenerate basic feasible solution.

\subsubsection{Summary}
So we have devised a method to move from a basic feasible solution to a new basic feasible solution where the reduced cost is strictly less than our original reduced cost.
We simply repeat this until we find a basic feasible solution that has a non-negative reduced costs vector.

\subsection{Using a Tableau}
We can use a tableau to conveniently carry out the simplex algorithm manually.
A \textit{simplex tableau} has four regions, each containing data about the current state of the algorithm:
\begin{center}
\begin{tabular}{|c|c|}
\hline
$-\text{cost}$ & reduced cost vector \\
\hline
current BFS & matrix for directions \\
\hline
\end{tabular}
$\longleftrightarrow$
\begin{tabular}{|c|c|}
\hline
$- \vec{c}^{\trans}_{B} \vec{x}_{B}$ & $\overline{\vec{c}}^{\trans}$ \\
\hline
$\vec{x}_{B}$ & $B^{-1} A$ \\
\hline
\end{tabular}
\end{center}
\begin{remark}[Warning]
  Do not forget to negate the cost otherwise the algorithm will not work!
\end{remark}
On the page, we write the tableau as:
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
$-\vec{c}^{\trans}_{B} \vec{x}_{B}$ & $\overline{c}_1$ & $\overline{c}_2$ & $\cdots$ & $\overline{c}_n$ \\
\hline
$x_{B(1)}$ & $\uparrow$ & $\uparrow$ &  & $\uparrow$ \\
$\vdots$ & $B^{-1}\vec{A}_1$ & $B^{-1} \vec{A}_2$ & $\cdots$ & $B^{-1} \vec{A}_n$ \\
$x_{B(m)}$ & $\downarrow$ & $\downarrow$ &  & $\downarrow$ \\
\hline
\end{tabular}
\end{center}
\begin{remark}[Note]
  It is important to note that a new tableau is created after each BFS checked by the algorithm so it may be useful to use pre-printed tables.
\end{remark}
\subsubsection{Running the algorithm using a tableau}
\begin{enumerate}
  \item If the problem is not already in the standard form, convert it into the standard form as described in \cref{LPEquivForms}.
  \item Find a basic feasible solution to start from.

    It is useful to note that if we have just added slack variables in step \textbf{(i)} and $\vec{b} \geq 0$, we can easily find a basic feasible solution by setting all the slack components to match $\vec{b}$ and all the $\vec{x}$ components to be $0$.
  \item Fill in the initial tableau by calculating and \textbf{negating} the starting cost, computing $B^{-1}A$, and then computing the reduced costs vector with $\overline{c}_j = c_j - \vec{c}^{\trans}_B B^{-1} \vec{A}_j$.

    It is also useful to note down the corresponding basic indices to the left of each row so that the final solution can be recovered.
  \item Repeat the following until $\overline{\vec{c}} \geq \vec{0}$:
    \begin{enumerate}
      \item Pick some pivot column $j$ such that $\overline{c}_j < 0$.
        If there are multiple such columns, pick any of them.

        {\color{gray}(This picks which entry to make non-zero and then $\vec{d}_{B} = B^{-1} \vec{A}_j$), i.e. the matrix part of the $j$-th column}
      \item Calculate $\frac{x_{B(i)}}{[B^{-1} \vec{A}_j]_i}$ for each row $i$ in column $j$ of the matrix section, the minimum of these tells us the pivot row $i$.
        If there are multiple such rows, pick any of them.
        We then call the entry $[B^{-1} A]_{i j}$ the pivot cell.

        {\color{gray}(This finds which $i$ corresponds to $t^{*}$)}
      \item Create a fresh tableau and copy over the noted basic indices but replacing that of the $i$-th row of the matrix with $j$.

        {\color{gray}(This is because after moving to our new point, this basic entry will be 0 and $j$ will replace it as a new basic index)}
      \item Find and perform row operations (scaling rows and adding one row to another) \textbf{on the entire current table} so that the pivot entry is 1 and all other entries in the $j$-th column (including $\overline{c}_j$) are 0.
        We do this by first scaling the pivot row so that the pivot cell is 1 and then adding multiples of the pivot row to the other rows so that their $j$-th column entries are 0.

        Instead of updating the values in the current table, instead write the new entries into the fresh tableau.

        {\color{gray}(This is equivalent to updating all the data in the \textbf{entire} table to be correct for the new BFS found by the algorithm -- the equivalence of this step is not proved within this course)}
    \end{enumerate}
  \item We have now arrived at an optimal BFS.
    Thus, the top left entry of the table is the negation of the optimal cost and the full optimal solution $\vec{x}$ can be recovered using the leftmost column of the table and the noted basic indices by setting all non-basic indices to 0.
\end{enumerate}
