\chapter{The Second Variation}
\section{Defining the Second Variation}
The \textit{second variation} of a functional is analogous to the second derivative of a function.
\begin{remark}[Recap]
  Recall from \cref{classifyingStatPts} that for a function $f$:
  \[
    f(\vec{x} + \varepsilon \vec{\xi}) - f(\vec{x}) = \varepsilon \vec{\xi} \cdot \nabla f(\vec{x}) + \frac{1}{2} \varepsilon^2 \xi_i \xi_j H_{i j}(\vec{x}) + O(\varepsilon^3)
  \]
  It follows that the point $\vec{x} = \vec{a}$ is a local minimum if $H_{i j}(\vec{a})$ is \textit{positive definite}, i.e. $\xi_i\xi_j H_{i j}(\vec{a}) > 0\ \forall \vec{\xi} \neq \vec{0}$, or equivalently, all eigenvalues of $H_{i j}(\vec{a})$ are positive.
\end{remark}
We want to find something analogous to the above but instead for functionals.
Consider the general functional:
\[
  F[y] = \int_{\alpha}^{\beta} f(y, y', x) \d{x}
\]
where $y \in \mathscr{C}$ defined by:
\[
  \mathscr{C} = \{(y: [\alpha, \beta] \to \R) : y(\alpha) = a, y(\beta) = b, \text{$y$ is twice continuously differentiable}\}
\]
Consider a variation $y + \varepsilon \xi \in \mathscr{C}$.
Since $y + \varepsilon x \in \mathscr{C}$, we must have
\begin{equation}
  y(\alpha) + \xi(\alpha) = y(\beta) + \xi(\beta) = 0 \implies \xi(\alpha) = \xi(\beta) = 0 \label{xiBdd}
\end{equation}

To calculate the variation $\delta F$, we first Taylor expand $f$ for small $\varepsilon$ to second order:
\begin{align*}
  f(y + \varepsilon \xi, y' + \varepsilon \xi', x) &= f(y, y', x) + \varepsilon \xi \pderiv{f}{y} + \varepsilon \xi' \pderiv{f}{y'} \\
                                                   &\quad + \frac{1}{2} \varepsilon^2 \left[\xi^2 \pderiv[2]{f}{y} + 2 \xi \xi' \frac{\partial^2 f}{\partial y \partial y'} + {\xi'}^2 \pderiv[2]{f}{{y'}}\right] + O(\varepsilon^3) \\
                                                   &= f(y, y', x) + \varepsilon \xi\left[\pderiv{f}{y} - \deriv{}{x} \pderiv{f}{y'}\right] + \varepsilon \left[\xi' \pderiv{f}{y'} + \xi \deriv{}{x}\pderiv{f}{y'}\right] \\
                                                   &\quad + \frac{1}{2} \varepsilon^2 \left[\xi^2 \pderiv[2]{f}{y} + 2 \xi \xi' \frac{\partial^2 f}{\partial y \partial y'} + {\xi'}^2 \pderiv[2]{f}{{y'}}\right] + O(\varepsilon^3) \\
                                                   &= f(y, y', x) + \varepsilon \xi \frac{\delta F[y]}{\delta y(x)} + \varepsilon \deriv{}{x}\left[\xi \pderiv{f}{y'}\right] \\
                                                   &\quad + \frac{1}{2} \varepsilon^2 \left[\xi^2 \pderiv[2]{f}{y} + 2 \xi \xi' \frac{\partial^2 f}{\partial y \partial y'} + {\xi'}^2 \pderiv[2]{f}{{y'}}\right] + O(\varepsilon^3)
\end{align*}
Upon substituting this back in to the functional, we obtain:
\begin{equation}
  F[y + \varepsilon \xi] - F[y] = \varepsilon \int_{\alpha}^{\beta} \xi(x) \frac{\delta F[y]}{\delta y(x)} \d{x} + \varepsilon \cancelto{0}{\eval{\xi \pderiv{f}{y'}}{\alpha}{\beta}} + \varepsilon^2 \delta^2 F[y, \xi] + O(\varepsilon^3) \label{2ndVariationDiff}
\end{equation}
where we define $\delta^2 F[y, \xi]$ as:
\[
  \delta^2 F[y, \xi] \equiv \frac{1}{2} \int_{\alpha}^{\beta} \left[\xi^2 \pderiv[2]{f}{y} + 2\xi \xi' \frac{\partial^2 f}{\partial y \partial y'} + {\xi'}^2 \pderiv[2]{f}{{y'}}\right] \d{x}
\]
We can integrate the middle term of this by parts:
\begin{align*}
  \int_{\alpha}^{\beta} 2\xi\xi' \frac{\partial^2 f}{\partial y \partial y'} \d{x} &= \int_{\alpha}^{\beta} \deriv{}{x}(\xi^2) \frac{\partial^2 f}{\partial y \partial y'} \d{x} \\
                                                                                   &= \cancelto{0}{\eval{\xi^2 \frac{\partial^2 f}{\partial y \partial y'}}{\alpha}{\beta}} - \int_{\alpha}^{\beta} \xi^2 \deriv{}{x} \frac{\partial^2 f}{\partial y \partial y'} \d{x}
\end{align*}
where the boundary term again disappears by \cref{xiBdd}.
Thus we can write the second variation as:
\begin{equation}
  \delta^2 F[y, \xi] = \frac{1}{2}\int_{\alpha}^{\beta} \left[\xi^2 \left(\pderiv[2]{f}{y} - \deriv{}{x} \frac{\partial^2 f}{\partial y \partial y'}\right) + {\xi'}^2 \pderiv[2]{f}{{y'}}\right] \d{x} \label{2ndVariation}
\end{equation}
\subsection{Global Minima of Functionals}
\begin{remark}[Recap]
  Recall from \cref{2ndOrderConvexity} that if $H(\vec{x})$ is positive definite for all $\vec{x}$, then $f(\vec{x})$ is convex and so any stationary point must be a global minimum by \cref{globalMin}.
\end{remark}
We can formulate an analogous statement to the above for functionals:
\begin{theorem}
  If $\delta^2 F[y, \xi] \geq 0$ for all $y \in \mathscr{C}$ and allowable $\xi(x)$, then $y = y_0$ is a global minimum of $F$ within $\mathscr{C}$ if $y_0 \in \mathscr{C}$ satisfies the Euler-Lagrange equation. \label{globalMinFunct}
\end{theorem}
\begin{remark}[Notes]
  \begin{itemize}
    \item When we refer to ``allowable'' $\xi(x)$, we mean that it is twice continuously differentiable and obeys the boundary conditions in \cref{xiBdd}.
    \item For a function to be convex, we require that its domain is also convex.
      Notice that the class of functions $\mathscr{C}$ is indeed convex as if $y_1, y_2 \in \mathscr{C}$ then so is $ty_1 + (1 - t)y_2 \in \mathscr{C}$ for all $t \in [0, 1]$ as it is clearly twice continuously differentiable and satisfies the boundary conditions for $\mathscr{C}$.
  \end{itemize}
\end{remark}
\begin{proof}
  Provided without proof.
\end{proof}
\begin{example}[Geodesics Revisted]
  Recall from \cref{geodesicSec} that we used the functional $L[y]$ for the length of the curve:
  \[
    L[y] = \int_{\alpha}^{\beta} \sqrt{1 + {y'}^2} \d{x}
  \]
  To find the second variation of this we need to calculate a few derivatives of the integrand $f(y, y', x) = \sqrt{1 + (y')^2}$.
  We first notice that $f$ has no explicit $y$ dependence and so:
  \[
    \pderiv[2]{f}{y} = 0 \text{ and }  \frac{\partial^2 f}{\partial y \partial y'} = \frac{\partial^2 f}{\partial y' \partial y} = 0
  \]
  We can also find:
  \[
    \pderiv[2]{f}{{y'}} = \left(1 + {y'}^2\right)^{-\frac{3}{2}}
  \]
  Therefore the second variation is given by:
  \[
    \delta^2 F[y, \xi] = \frac{1}{2} \int_{\alpha}^{\beta} {\xi'}^2 (1 + (y')^2)^{-\frac{3}{2}} \d{x} \geq 0\ \forall y, \xi
  \]
  Thus any solution of the Euler-Lagrange equation for this function (which we saw was a straight line) is indeed a \textbf{global minimum} of the distance between two points.
\end{example}
\section{Classifying Optimal Solutions}
\subsection{Using the Second Variation}
The requirement $\delta^2 F[y, \xi] \geq 0$ for all $y \in \mathscr{C}$ is quite a strict requirement.
However we can use the second variation to figure out if a function $y_0$ that extremises $F$ (i.e. satisfies its Euler-Lagrange equation) is a local minimum or a local maximum.
\begin{proposition}
  If $y_0 \in \mathscr{C}$ satisfies the Euler-Lagrange equation for $F$, then: \label{classifyingFunct}
  \begin{itemize}
    \item If $\delta^2 F[y_0, \xi] > 0$ for all allowed $\xi(x)$ that are not identically 0, then $y_0$ is a local minimum.
    \item Conversely, we can conclude that $y_0$ is not a local minimum if we can find any $\xi(x)$ such that $\delta^2 F[y_0, \xi] < 0$.
    \item If $\delta^2 F[y_0, \xi] < 0$ for all allowed $\xi(x)$ that are not identically 0, then $y_0$ is a local maximum.
    \item Conversely, we can conclude that $y_0$ is not a local maximum if we can find any $\xi(x)$ such that $\delta^2 F[y_0, \xi] > 0$.
\end{itemize}
\end{proposition}
\begin{remark}
  This is an analogue of using the definiteness/eigenvalues of the Hessian to determine the nature of stationary points for functions.
\end{remark}
\begin{proof}
  Using the expression in \cref{2ndVariationDiff}.
\end{proof}
\begin{example}[Geodesics on a Circle]
  In Example Sheet 2 Q5, you will show that a geodesic between two points $A, B$ on a circle is a segment of the unique great circle that passes through $A, B$.

  For this unique great circle, there is two segments we could choose.
  Upon calculating the second second variation at the shorter of the two paths, we see that it has $\delta^2 F > 0\ \forall \xi \centernot\equiv 0$ and so it must be a local minimum.

  The longer of the two paths can be shown to be non-minimal as we can find $\xi$ such that $\delta^2 F < 0$.

  In the case where $A$ and $B$ are antipodal points, then $\delta^2 F \geq 0\ \forall \xi$ which is not covered by our criteria, despite still being a local minimum.
\end{example}
\subsection{Legendre Condition}
\begin{proposition}[Lengendre Condition]
  Given any $y_0 \in \mathscr{C}$ satisfying the Euler-Lagrange equations, a \textbf{necessary condition} for $y_0$ to be a local minimum is: \label{legendreCond}
\[
    \at{\pderiv[2]{f}{{y'}}}{y = y_0} \geq 0\ \forall x \in [\alpha, \beta]
  \]
\end{proposition}
\begin{proof}
  Suppose, for contradiction, that $\at{\pderiv[2]{f}{{y'}}}{y = y_0} < 0$ at some $x_0 \in [\alpha, \beta]$.
  Recall the expression for $\delta^2 F$ from \cref{2ndVariation}:
  \[
    \delta^2 F[y, \xi] = \frac{1}{2}\int_{\alpha}^{\beta} \left[\xi^2 \left(\pderiv[2]{f}{y} - \deriv{}{x} \frac{\partial^2 f}{\partial y \partial y'}\right) + {\xi'}^2 \pderiv[2]{f}{{y'}}\right] \d{x}
  \]
  Since we want to make this negative for some $\xi$, we can then construct a function $\xi(x)$ such that $\xi$ is non-zero only near $x_0$ with with large $\xi'$ but small $\xi$ value so that the integrand is negative near $x_0$.
  Since $\xi$ is zero otherwise, $\delta^2 F[y_0, \xi] < 0$ and so by \cref{classifyingFunct} $y_0$ cannot be a local minimum.
  This is a contradiction and so the result follows.
\end{proof}
\begin{example}
  Consider the functional:
  \[
    F[y] = \int_{-1}^{1} x\sqrt{1 + {y'}^2} \d{x}
  \]
  Since $f(y, y', x) = x \sqrt{1 + {y'}^2}$, have:
  \[
    \pderiv[2]{f}{{y'}} = x(1 + {y'}^2)^{-\frac{3}{2}}
  \]
  which violates the Legendre condition any choice of $x \in [-1, 0)$.
  Thus any solution of the Euler-Lagrange equations is \textbf{not} a local minimum.
\end{example}
Let $y_0 \in \mathscr{C}$ satisfy the Euler-Lagrange equation and write:
\begin{equation}
  \delta^2 F[y_0, \xi] = \frac{1}{2} \int_{\alpha}^{\beta} [\rho(x) {\xi'}^2 + \sigma(x)\xi^2] \d{x} \label{2ndVar}
\end{equation}
where we define:
\[
  \rho = \pderiv[2]{f}{{y'}} \text{ and } \sigma = \pderiv[2]{f}{y} - \deriv{}{x} \frac{\partial^2 f}{\partial y \partial y'}
\]
With these variables, the Legendre condition (\cref{legendreCond}) can be rewritten as:
\[
  \rho \geq 0\ \forall x \in [\alpha, \beta] \text{ is a \textbf{necessary condition} for $y_0$ to be a local minimum.}
\]
for $y_0 \in \mathscr{C}$ satisfying the Euler-Lagrange equations.

The converse of this does not hold but we can still formulate a sufficient condition.
\begin{proposition}
  A \textbf{sufficient condition} for a solution of the Euler-Lagrange equation $y_0$ to be local minimum is: \label{minSuff}
  \[
    \rho > 0 \text{ and } \sigma \geq 0\ \forall x \in (\alpha, \beta)
  \]
\end{proposition}
\begin{proof}
  From \cref{2ndVar} we see that these inequality clearly tell us that $\delta^2 F \geq 0\ \forall \xi$.
  To apply \cref{classifyingFunct}, we need that $\delta^2 F \geq 0\ \forall \xi \centernot\equiv 0$.

  Suppose that we have $\delta^2 F = 0$ for some $\xi$.
  Since $\rho > 0$, we must then have $\xi' = 0$ on $[\alpha , \beta]$, and thus $\xi$ is constant on $\alpha, \beta$.
  But $\xi \in \mathscr{C}$, which means that $\xi(\alpha) = \xi(\beta) = 0$ and so $\xi$ is identically 0.

  Thus $\xi \centernot\equiv 0 \implies \delta^2 F[y_0, \xi] > 0$ and so we can use \cref{classifyingFunct} to conclude.
\end{proof}
\begin{example}[Brachistochrone Revisited]
  Revisiting \cref{exBrachistochrone}, recall that we were aiming to minimise the functional:
  \[
    T[y] = \int_{0}^{x_B} f \d{x} \text{ where } f= \sqrt{\frac{1 + {y'}^2}{- y}}
  \]
  For this functional to make sense, we need $y$ to be negative and so we impose this additional restriction on $\mathscr{C}$.

  Now assume that $y$ satisfies the Euler-Lagrange equations (which we saw means it is a cycloid).
  So that we can use this fact, it is beneficial to try and write $\rho$ and $\sigma$ in terms of $\pderiv{f}{y}$ and $\deriv{}{x}\pderiv{f}{y'}$ as these appear in the Euler-Lagrange equations.

  Calculating derivatives of $f$ we have:
  \[
    \pderiv{f}{{y'}} = \frac{y'}{\sqrt{-y(1 + {y'}^2)}} \text{ and } \pderiv{f}{y} = - \frac{f}{2y}
  \]
  and so:
  \[
    \rho = \pderiv[2]{f}{{y'}} = \frac{1}{\sqrt{(-y)(1 + {y'}^2)^3}} > 0
  \]
  so the first condition is satisfied.

  To find $\sigma$ we need to calculate:
  \[
    \frac{\partial^2 f}{\partial y \partial y'} = - \frac{1}{2y} \pderiv{f}{y'} \text{ and } \pderiv[2]{f}{y} = \frac{3f}{4y^2}
  \]
  and so:
  \begin{align*}
    \sigma &= \pderiv[2]{f}{y} - \deriv{}{x}\left(-\frac{1}{2y} \pderiv{f}{{y'}}\right) \\
           &= \pderiv[2]{f}{y} - \frac{y'}{2y^2} \pderiv{f}{{y'}} + \frac{1}{2y} \deriv{}{x} \pderiv{f}{y'} \\
           &= \pderiv[2]{f}{y} - \frac{y'}{2y^2} \pderiv{f}{{y'}} + \frac{1}{2y} \pderiv{f}{y} \text{ using the Euler-Lagrange equation} \\
           &= \frac{3f}{4y^2} - \frac{{y'}^2}{2y^2\sqrt{(-y)(1 + {y'}^2)}} - \frac{f}{4y^2} \text{ by substituting in} \\
           &= \frac{1}{2\sqrt{(-y)(1 + {y'}^2)}} > 0
  \end{align*}
  Thus $\rho > 0$ and $\sigma > 0$ so by \cref{minSuff}, $y$ (a cycloid) is a local minimum of $T$.
\end{example}
\subsection{Using a Sturm-Liouville Problem}
We can also integrate \cref{2ndVar} by parts to yield:
\begin{align}
  \delta^2 F[y_0, \xi] &= \frac{1}{2} \int_{\alpha}^{\beta} [\rho \xi' \cdot \xi' + \sigma \xi^2] \d{x} \nonumber \\
                       &= \frac{1}{2} \cancelto{0}{\eval{\rho \xi \xi'}{\alpha}{\beta}}  + \frac{1}{2} \int_{\alpha}^{\beta} \left[-\xi\deriv{}{x}(\rho \xi') + \sigma \xi^2\right] \d{x} \text{ as $\xi(\alpha) = \xi(\beta) = 0$} \nonumber \\
                       &= \frac{1}{2}\int_{\alpha}^{\beta} \xi \mathcal{L}\xi(x) \d{x} \label{2ndVarSLP}
\end{align}
where $\mathcal{L}$ is the differential operator defined as:
\begin{equation}
  \mathcal{L}\xi \equiv -\deriv{}{x}\left(\rho \deriv{\xi}{x}\right) + \sigma \xi \label{LDef}
\end{equation}
We can use this form of $\delta^2 F[y_0, \xi]$ to formulate alternative conditions for $y_0$ to be a local minimum.
\begin{proposition}
  Consider the Sturm-Liouville problem: \label{SLPClassification}
  \[
    \mathcal{L} \eta = \lambda w(x) \eta \text{ and } \eta(\alpha) = \eta(\beta) = 0
  \]
  for some \textit{weight function} where $w(x) > 0$ on $(\alpha, \beta)$.

  We then have that:
  \begin{itemize}
    \item If $y_0$ is a local minimum, then all of the eigenvalues of the above are non-negative.
    \item If all of the eigenvalues of the above are positive, then $y_0$ is a local minimum
  \end{itemize}
\end{proposition}
\begin{remark}[Notes]
  \begin{enumerate}
    \item We saw another example of a Sturm-Liouville problem in \cref{SLPSec}.
    \item The weight function $w$ is not fixed and so we can pick a convenient one provided that is is non-negative.
    \item Knowing that all of the eigenvalues are \textbf{non-negative} does not help us as the above does not tell us anything about the nature of $y_0$.
  \end{enumerate}
\end{remark}
\begin{proof}
  \begin{proofdirection}{Assume $y_0$ is a local minimum}
    Suppose, for contradiction, that there is an eigenfunction $\eta$ of the above Sturm-Liouville problem with negative eigenvalue $\lambda$.
    Using \cref{2ndVarSLP}, we have:
    \begin{align*}
      \delta^2 F[y_0, \eta] &= \frac{1}{2} \int_{\alpha}^{\beta} \eta \mathcal{L} \eta \d{x} \\ \
                            &= \frac{\lambda}{2} \int_{\alpha}^{\beta} w \eta^2 \d{x} < 0
    \end{align*}
    and so $y_0$ cannot be a local minimum by \cref{classifyingFunct}, which is a contradiction.
    Thus all eigenvalues must be non-negative.
  \end{proofdirection}
  \begin{proofdirection}{Assume all eigenvalues are positive}
    Let $\lambda_n$ be the $n$-th eigenvalue and $y_n$ the $n$-th eigenfunction of the above Sturm-Liouville problem.
    From IB Methods, we know that the eigenfunctions $\{y_n\}$ form a complete space for the set of allowed $\eta$.
    That is, we can write:
    \[
      \eta(x) = \sum_{n} a_n y_n(x)
    \]
    for some coefficients $\{a_n\}$.
    Furthermore, we can choose $\{y_n\}$ to be orthogonal, that is:
    \[
      \int_{\alpha}^{\beta} w y_m y_n \d{x} = 0 \text{ for } m \neq n
    \]

    Subsisting our new expression for $\eta$ into \cref{2ndVarSLP}, we obtain:
    \begin{align*}
      \delta^2 F[y_0, \eta] &= \frac{1}{2} \int_{\alpha}^{\beta} \left[\sum_{m} a_m y_m\right] \mathcal{L} \left[\sum_{n} a_n y_n\right] \d{x} \\
                            &= \frac{1}{2} \int_{\alpha}^{\beta} \left[\sum_{m} a_m y_m\right] \left[\sum_{n} \lambda_n w y_n\right] \d{x} \\
                            &= \frac{1}{2} \sum_{m, n} \left[a_m a_n \lambda_n \int_{\alpha}^{\beta} w y_m y_n \d{x}\right]
    \end{align*}
    Using orthogonality, the terms of this are non-zero if and only if $m = n$ and so:
    \[
      \delta^2 F[y_0, \eta] = \frac{1}{2} \sum_{m} \left[\lambda_m a^2_m \int_{\alpha}^{\beta} w y^2_m \d{x}\right]
    \]
    Considering the summand here:
    \[
      \underbrace{\lambda_m}_{> 0} \underbrace{a^2_m}_{\geq 0} \underbrace{\int_{\alpha}^{\beta} w y^2_m \d{x}}_{> 0} \geq 0
    \]
    We know that $\int_{\alpha}^{\beta} w y^2_m \d{x} > 0$ as $w > 0$ and $y_m$ is not identically zero as it is an eigenfunction.

    Thus $\delta^2 F[y_0, \eta] \geq 0$ with equality if and only if $a_m = 0\ \forall m$ and so $\delta^2 F > 0$ for $\eta \centernot\equiv 0$.
    So $y_0$ is a local minimum by \cref{classifyingFunct}.
  \end{proofdirection}
\end{proof}
\begin{example}
  Consider the functional: \label{2ndVarExample}
  \[
    F[y] = \frac{1}{2} \int_{0}^{a} ({y'}^2 - y^2) \d{x}
  \]
  where we restrict $y$ to have the fixed endpoints $y(0) = y_0$ and $y(a) = y_1$ for $a \neq 0$.

  The Euler-Lagrange equation for this functional is:
  \[
    - y - \deriv{}{x} y' = 0 \iff y'' + y = 0
  \]
  which we can easily solve to yield a solution $y_0$.

  To find the second variation we can calculate:
  \begin{align*}
    \rho &= \frac{1}{2}\pderiv[2]{}{{y'}} ({y'}^2 - y^2) = 1 \\
    \text{ and } \sigma &= \frac{1}{2}\pderiv[2]{}{y}({y'}^2 - y^2) - \frac{1}{2}\deriv{}{x}\frac{\partial^2}{\partial y \partial y'}({y'}^2 - y^2) = -1
  \end{align*}
  and so using \cref{2ndVar} we have:
  \[
    \delta^2 F[y_0, \xi] = \frac{1}{2} \int_{0}^{a} ({\xi'}^2 - \xi^2) \d{x}
  \]
  Using \cref{LDef}, $\mathcal{L} \eta$ is given by:
  \[
    \mathcal{L} \eta = -\deriv{}{x}\left(1 \cdot \deriv{\eta}{x}\right) + (-1) \eta = -\deriv[2]{\eta}{x} - \eta
  \]
  Thus the associated Sturm-Liouville problem is:
  \[
    -\deriv[2]{\eta}{x} - \eta = \lambda w \eta \text{ and } \eta(0) = \eta(a) = 0
  \]
  If we choose $w(x) \equiv 1$, then:
  \[
    \eta'' + (\lambda + 1)\eta = 0 \implies \eta = A \sin(x\sqrt{\lambda + 1}) + B \cos(x\sqrt{\lambda + 1})
  \]
  Applying the boundary conditions, $\eta(0) = 0 \implies B = 0$ and:
  \[
    \eta(a) = 0 \implies A \sin(a \sqrt{\lambda + 1}) = 0
  \]
  If $A = 0$, then $\eta \equiv 0$ and so it cannot be an eigenfunction.
  Otherwise, $\sin(a \sqrt{\lambda + 1}) = 0 \iff a \sqrt{\lambda + 1} = n\pi$ for some $n \in \Z$.
  We discard $n = 0$ as this again leads to $\eta \equiv 0$.

  Thus the eigenfunctions are exactly:
  \[
    \eta_{n}(x) = A \sin \left(\frac{n \pi x}{a}\right)
  \]
  for $A \in \R$ and $n \in \N$ (as $n < 0$ can be absorbed by changing the sign of $A$)

  To find the eigenvalues:
  \[
    a \sqrt{\lambda_n + 1} = n \pi \iff \lambda_n = \frac{n^2 \pi^2}{a^2} - 1
  \]
  for $n \in \N$.
  We can then check when all of the eigenvalues are positive:
  \[
    \lambda_n > 0\ \forall n \in \N \iff a < \pi
  \]
  So by \cref{SLPClassification}, if $a < \pi$, then $y_0$ is a local minimum of $F$.

  To use the other direction of \cref{SLPClassification}, we can check when there is a negative eigenvalue:
  \[
    \exists n \in \N \text{ s.t. } \lambda_n < 0 \iff a > \pi
  \]
  So if $a > \pi$, then $y_0$ is not a local minimum of $F$.

  In the case of $a = \pi$, we cannot use \cref{SLPClassification} to determine the nature of $y_0$ as we only have $\lambda_n \geq 0\ \forall n \in \N$.
\end{example}
\subsection{Jacobi Condition}
Let $y_0 \in \mathscr{C}$ satisfy the Euler-Lagrange equation.
As in \cref{xiBdd}, for a variation $y \to y + \xi$, we must have $\xi(\alpha) = \xi(\beta) = 0$.
Consider the following integral for any differentiable function $\varphi$:
\begin{align*}
  \int_{\alpha }^{\beta} \deriv{}{x}(\varphi \xi^2) \d{x} &= \eval{\varphi \xi^2}{\alpha}{\beta} = 0 \\
                                                &= \int_{\alpha}^{\beta} (2 \varphi \xi \xi' + \varphi' \xi^2) \d{x}
\end{align*}
Since this is zero, we can freely add it to the expression for $\delta^2 F[y_0, \xi]$ from \cref{2ndVar}:
\[
  \delta^2 F[y_0, \xi] = \frac{1}{2} \int_{\alpha}^{\beta} (\rho {\xi'}^2 + 2 \varphi \xi \xi' + (\sigma + \varphi')\xi^2) \d{x}
\]
Assuming that $\rho > 0$ for $\alpha \leq x \leq \beta$, we can complete the square inside the integral as:
\[
  \delta^2 F[y_0, \xi] = \frac{1}{2} \int_{\alpha}^{\beta} \left[\rho\left(\xi' + \frac{\varphi}{\rho}\xi\right)^2 + \left(\sigma + \varphi' - \frac{\varphi^2}{\rho}\right)\xi^2\right] \d{x}
\]
The second term in this integrand looks quite complicated and the first term is clearly non-negative and it would be convenient for us to choose a $\varphi$ which satisfies the \textit{Ricatti equation}:
\begin{equation}
  \varphi' + \sigma = \frac{\varphi^2}{\rho} \label{ricatti}
\end{equation}
If we choose such a $\varphi$, then $\delta^2 F[y_0, \xi] \geq 0$ with equality if and only if $\xi' + \frac{\varphi}{\rho}\xi = 0\ \forall x \in [\alpha, \beta]$ i.e. if and only if $\xi$ is of the form:
\[
  \xi(x) = C \exp\left[-\int_{\alpha}^{x} \frac{\varphi(s)}{\rho(s)} \d{s}\right]
\]
However, $\xi(\alpha) = 0 \implies C = 0$ and so $\delta^2 F[y_0, \xi] = 0 \iff \xi(x) \equiv 0$.
Thus $\delta^2 F[y_0, \xi] > 0$ for all non-trivial allowed $\xi$, and so $y_0$ is a local minimum by \cref{classifyingFunct}.

\begin{theorem}[Jacobi Condition]
  If there exists a solution to \cref{ricatti} on $[\alpha, \beta]$, then $y_0$ is a local minimum.
\end{theorem}
\begin{proof}
  See above.
\end{proof}
We see that \cref{ricatti} is a first order nonlinear ODE, however we can convert to a linear second order ODE by performing the substitution $\varphi = - \rho \frac{u'}{u}$ for $u \neq 0$.
This transforms \cref{ricatti} into:
\begin{align*}
  -\left(\rho \frac{u'}{u}\right)' + \sigma &= \rho \left(\frac{u'}{u}\right)^2 \\
   \iff - \frac{(\rho u')'}{u} + \cancel{\rho \frac{{u'}^2}{u^2}} + \sigma &= \cancel{\rho \frac{{u'}^2}{u^2}} \\
   \iff -(\rho u')' + \sigma u &= 0
\end{align*}
This is called the \textit{Jacobi accessory equation}.
Thus if there exists a solution to this with $u \neq 0$ on $[\alpha, \beta]$, then we obtain a solution $\varphi$ of \cref{ricatti} and so $y_0$ is a local minimum.
\begin{remark}
  Since the Jacobi accessory equation is a linear ODE, it always has a solution but not necessarily one with $u \neq 0$ on $[\alpha, \beta]$.
\end{remark}
\begin{example}
  Consider again the functional from \cref{2ndVarExample}.
  Recall that we had $\rho = 1$ and $\sigma = -1$ and so the Jacobi accessory equation is:
  \[
    -u'' - u = 0 \implies u = A \sin (x - x_0)
  \]
  There is a solution to this that is non-zero on $[0, a]$ if and only if $a < \pi$.

  So if $a < \pi$, then a solution of the Euler-Lagrange equation is a local minimum.
  This is as we found with the Sturm-Liouville approach.
  However, unlike the Sturm-Liouville approach, this method does not let us prove that the solution is not a local minimum for $a > \pi$.
\end{example}
\begin{remark}[Advice]
  The Sturm-Liouville approach is usually the best choice as it stronger and uses machinery discussed in IB Methods.
  However, it is good to be aware of this alternative approach using the Jacobi condition.
\end{remark}
