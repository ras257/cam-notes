\chapter{Principle of Least Action}
\section{Configuration Space}
Consider a system of $N$ particles in 3D with positions $\vec{x}_1, \ldots, \vec{x}_n$.
We can consider the positions of the particles at a single point as a $3N$ dimensional point in \textit{configuration space}:
\[
  (\vec{x}_1, \ldots, \vec{x}_N)
\]
It is sometimes also useful to use non-Cartesian coordinates for the configuration space:
\[
  \vec{q} = (\vec{q}_1, \ldots, \vec{q}_n)
\]
\section{The Lagrangian}
The \textit{principle of least action} gives us a simple way to determine the equations of motion from the kinetic energy $T$ and potential energy $V$ in terms of $\vec{q}(t)$ and $\dot{\vec{q}}(t)$.
\begin{definition}[Lagrangian]
  We define the \textit{Lagrangian} as:
  \[
    L(\vec{q}, \dot{\vec{q}}, t) = T - V
  \]
  for kinetic energy $T$ and potential energy $V$.
\end{definition}
\begin{definition}[Action]
  The \textit{action} of a path $\vec{q}(t)$ in configuration space between $\vec{q}_A$ and time $t_A$, and $\vec{q}_B$ at time $t_B$ is given by the functional:
  \[
    I[\vec{q}] = \int_{t_A}^{t_B} L(\vec{q}(t), \dot{\vec{q}}(t), t) \d{t}
  \]
\end{definition}
\begin{law}[Principle of Least Action]
  The \textit{principle of least action} tells us that the actual path from $\vec{q}_A$ to $\vec{q}_B$ of the system extremises the action.
\end{law}
Since $\vec{q}_A$ and $\vec{q}_B$ are fixed, the boundary term in $\delta I$ vanishes and so we can utilise the Euler-Lagrange equations.
As $\vec{q}$ has $3N$ components, we will obtain $3N$ Euler-Lagrange equations:
\[
  \pderiv{L}{q_i} - \deriv{}{t} \pderiv{L}{\dot{q}_i} = 0 \text{ for } i \in \{1, \ldots, 3N\}
\]
\begin{example}
  If we take $N = 1$ and $\vec{q} = \vec{x}$ (i.e. we work in Cartesian coordinates), we know that $T = \frac{1}{2} m \dot{\vec{x}} \cdot \dot{\vec{x}}$ for fixed $m$ and assume that we can write $V = V(\vec{x}, t)$. \label{n1LG}

  The Lagrangian is then:
  \[
    L = \frac{1}{2} m \dot{\vec{x}} \cdot \dot{\vec{x}} - V(\vec{x}, t)
  \]
  and so the Euler-Lagrange equations tells us that:
  \[
    -\pderiv{V}{x_i} - \deriv{}{t}(m \dot{x}_i) = 0 \ \forall i \in \{1, 2, 3\} \iff m \ddot{\vec{x}} = -\nabla V
  \]
  This is Newton's 2nd Law for a conservative force $\vec{F} = - \nabla V$.
\end{example}
\begin{proposition}
If $L$ has no explicit $t$ dependence, then for any solution of the Euler-Lagrange equations there exists a first integral: \label{LnoT}
  \[
    L - \sum_{i = 1}^{3N}  \dot{q}_i \pderiv{L}{\dot{q}_i} = - E
  \]
  for some constant $E$.
\end{proposition}
\begin{proof}
  We first consider the full derivative $\deriv{L}{t}$ which is not necessarily 0 because of the implicit $t$ dependence:
  \begin{align*}
    \deriv{L}{t} &= \deriv{}{t} L(\vec{q}(t), \dot{\vec{q}}(t), t) \\
                 &= \pderiv{L}{t} + \sum_{i = 1}^{3N} \left(\pderiv{L}{q_i} \dot{q}_i + \pderiv{L}{\dot{q_i}} \ddot{q}_i\right) \text{ by multivariate chain rule}\\
                 &= \pderiv{L}{t} + \sum_{i = 1}^{3N} \dot{q}_i \cancelto{0}{\left(\pderiv{L}{q_i} - \deriv{}{t} \pderiv{L}{\dot{q_i}}\right)} + \sum_{i = 1}^{3N} \left(\ddot{q}_i \pderiv{L}{\dot{q_i}} + \dot{q}_i \deriv{}{t} \pderiv{L}{\dot{q_i}}\right) \\
                 &= \pderiv{L}{t} + \deriv{}{t}\left(\sum_{i = 1}^{3N} \dot{q}_i \pderiv{L}{\dot{q_i}}\right)
  \end{align*}
  Rearranging we have:
  \begin{equation}
    \deriv{}{t}\left(L - \sum_{i = 1}^{3N} \dot{q}_i \pderiv{L}{\dot{q}_i}\right) = \pderiv{L}{t} \label{derivPderiv}
  \end{equation}
  So if $L$ has no explicit $t$ dependence, then $\pderiv{L}{t} = 0$ and so:
  \[
     L - \sum_{i = 1}^{3N} \dot{q}_i \pderiv{L}{\dot{q}_i} = -E
  \]
  for some constant $E$.
\end{proof}
\begin{example}
  Continuing from \cref{n1LG}, if we further assume that $V$ has no $t$ dependence, i.e. $V = V(\vec{x})$, then the constant $E$ is:
  \begin{align*}
    -E &= \frac{1}{2} m \dot{\vec{x}} \cdot \dot{\vec{x}} - V - \sum_{i = 1}^{3} \dot{x}_i m \dot{x}_i \\
       &= -\frac{1}{2} m \dot{\vec{x}} \cdot \dot{\vec{x}} - V(\vec{x}) \\
       &= - (T + V)
  \end{align*}
  So $E = T + V$ is the \textbf{conserved} total energy.
\end{example}
\begin{remark}
  Because we can recover the equations of motion from the Lagrangian, we think of it as \textit{encoding} the laws of physics.
  So we can think of $L$ having no $t$ dependence as being the same as the statement that the laws of physics are time translation invariant.
\end{remark}
\begin{example}[Polar Coordinates]
  Consider a particle in 2D dimensions with position $\vec{q} = (r, \theta)$ in plane polar coordinates.
  In polar coordinates, the kinetic energy $T$ is given by: \label{polarEx}
  \[
    T = \frac{1}{2}m(\dot{r}^2 + r^2 \dot{\theta}^2)
  \]
  If we assume that the force is radial, i.e. $V = V(r)$, then the Lagrangian is:
  \[
    L = \frac{1}{2}m(\dot{r}^2 + r^2 \dot{\theta}^2) - V(r)
  \]
  Using the Euler-Lagrange equation for $\theta(t)$:
  \[
    \pderiv{L}{\theta} - \deriv{}{t} \pderiv{L}{\dot{\theta}} = 0
  \]
  We see that $\pderiv{L}{\theta} = 0$ so we have a first integral:
  \[
    \pderiv{L}{\dot{\theta}} = mr^2 \dot{\theta} = \text{constant}
  \]
  which is the conservation of angular momentum.
  Since $m$ is constant, we call the conserved angular momentum per unit mass $h = r^2 \dot{\theta}$.

  We could try and use the Euler-Lagrange equation for $r(t)$, however there is explicit $r$ dependence and so we would end up with a second order ODE.
  However, since $\pderiv{L}{t} = 0$, we know from \cref{LnoT} that there is a conserved energy:
  \[
    E = \dot{r} \pderiv{L}{\dot{r}} + \dot{\theta} \pderiv{L}{\dot{\theta}} - L = \frac{1}{2}m (\dot{r}^2 + r^2 \dot{\theta}^2) + V = T + V
  \]
  Solving to eliminate $\dot{\theta}$, we obtain:
  \[
    \frac{1}{2} m\dot{r}^2 + V_{\text{eff}}(r) = E
  \]
  where:
  \[
    V_{\text{eff}}(r) = \frac{mh^2}{2r^2} + V(r)
  \]
  We can then go on to analyse this using methods as discussed in IA Dynamics \& Relativity.
\end{example}
\section{The Hamiltonian}
\subsection{Definition}
\begin{definition}
  Assuming that the Lagrangian $L(\vec{q}, \dot{\vec{q}}, t)$ is a convex function of $\dot{\vec{q}}$, we then define the \textit{Hamiltonian} as the Legendre transform of $L$ with respect to $\dot{\vec{q}}$:
  \begin{align*}
    H(\vec{q}, \vec{p}, t) &= \sup_{\dot{\vec{q}}} [\vec{p} \cdot \dot{\vec{q}} - L(\vec{q}, \dot{\vec{q}}, t)] \\
                           &= \at{[\vec{p} \cdot \dot{\vec{q}} - L(\vec{q}, \dot{\vec{q}}, t)]}{\dot{\vec{q}} = \dot{\vec{q}}(\vec{q}, \vec{p}, t)}
  \end{align*}
  which follows from \cref{legDiff} where $\dot{\vec{q}}(\vec{q}, \vec{p}, t)$ is a solution to $\nabla_{\dot{\vec{q}}} L(\vec{q}, \dot{\vec{q}}, t) = \vec{p}$.
  That is, $\dot{\vec{q}}(\vec{q}, \vec{p}, t)$ extremises $\vec{p} \cdot \dot{\vec{q}} - L(\vec{q}, \dot{\vec{q}}, t)$ for a given $\vec{q}$, $\vec{p}$ and $t$ which by convexity must exist and also be the global maximum.
\end{definition}
The components of $\vec{p}$ satisfy:
\begin{equation}
  p_i = \pderiv{L}{\dot{q}_i}(\vec{q}, \dot{\vec{q}}(\vec{q}, \vec{p}, t), t) \label{pComp}
\end{equation}
which we can use to determine the components of $\dot{\vec{q}}(\vec{q}, \vec{p}, t)$ for a given $\vec{q}$, $\vec{p}$, t.
Substituting this expression into $H$, we have:
\[
  H = \sum_{i = 1}^{3N} \dot{q}_i \pderiv{L}{\dot{q}_i} - L
\]
From \cref{LnoT}, we know that if $L$ has no explicit $t$ dependence (i.e. $\pderiv{L}{t} = 0$), then the above is exactly the conserved energy $E$ and so in this case the Hamiltonian is also conserved.
\subsection{Examples}
\begin{example}[Cartesian Coordinates]
  Considering \cref{n1LG} again, the Lagrangian was: \label{cartHamil}
  \[
    L = \frac{1}{2} m \dot{\vec{x}} \cdot \dot{\vec{x}} - V(\vec{x}, t)
  \]
  From \cref{pComp}, we have:
  \[
    p_i = \pderiv{L}{\dot{x}_i} = m \dot{x}_i \implies \vec{p} = m \dot{\vec{x}}
  \]
  which is the momentum and $\dot{\vec{x}}(\vec{x}, \vec{p}, t) = \frac{\vec{p}}{m}$.
  Thus the Hamiltonian is:
  \begin{align*}
    H &= p \cdot \frac{\vec{p}}{m} - \left(\frac{1}{2}m \frac{\vec{p}}{m} \cdot \frac{\vec{p}}{m} - V\right) \\
      &= \frac{\vec{p} \cdot \vec{p}}{2m} + V
  \end{align*}
  which is the total energy.
\end{example}
\begin{example}[Polar Coordinates]
  Considering \cref{polarEx} again where $\vec{q} = (r, \theta)$ under a central force depending only on $r$, the Lagrangian was:
  \[
    L = \frac{1}{2} m (\dot{r}^2 + r^2 \dot{\theta}^2) - V(r)
  \]
  By \cref{pComp}, for a given $\vec{p}, \vec{q}, t$, we have that $\dot{\vec{q}}(\vec{q}, \vec{p}, t) = (\dot{r}, \dot{\theta})$ in the Hamiltonian must satisfy:
  \[
    p_r = \pderiv{L}{\dot{r}} = m\dot{r} \text{ and } p_{\theta} = \pderiv{L}{\dot{\theta}} = mr^2 \dot{\theta}
  \]
  So the Hamiltonian is given by:
  \begin{align*}
    H &= p_r \dot{r} + p_{\theta} \dot{\theta} - L \\
      &= p_r \left(\frac{p_r}{m}\right) + p_{\theta} \left(\frac{p_{\theta}}{mr^2}\right) - \frac{1}{2} m \left[\left(\frac{p_r}{m}\right)^2 + r^2 \left(\frac{p_{\theta}}{mr^2}\right)^2\right] + V(r) \\
      &= \frac{p^2_r}{2m} + \frac{p^2_{\theta}}{2mr^2} + V(r)
  \end{align*}
\end{example}
In \cref{cartHamil}, we saw that $\vec{p}$ was the momentum of the particle, however in a general set of coordinates, we call $\vec{p}$ the \textit{conjugate momentum} to $\vec{q}$.

\begin{remark}
  The main thing to remember when working with the Hamiltonian is that we want our final expression to \textbf{not depend on} $\dot{\vec{q}}$ and so our first step should be to find expressions for $q_i$ using \cref{pComp}.
\end{remark}
\subsection{Hamiltonian Equations of Motion}
We can also write the equations of motion in terms of $H$.
We first consider the derivative with respect to the components of $\vec{p}$, using the multivariate chain rule:
\begin{align*}
  \at{\pderiv{H}{p_i}}{\vec{q}} &= \at{\pderiv{}{p_i}\left[\sum_{j = 1}^{3N} p_j \dot{q}_j - L(\vec{q}, \dot{\vec{q}}(\vec{q}, \vec{p}, t), t)\right]}{\vec{q}, t} \\
                                &= \dot{q}_i + \sum_{j = 1}^{3N}  p_j \pderiv{\dot{q}_j}{p_i} - \sum_{j = 1}^{3N} \pderiv{L}{\dot{q}_j} \pderiv{\dot{q}_j}{p_i} \text{ as $\dot{\vec{q}}$ is a function of $\vec{p}$}
\end{align*}
Similarly for the derivative with respect to the components of $\vec{q}$:
\begin{align*}
  \at{\pderiv{H}{q_i}}{\vec{p}} &= \at{\pderiv{}{q_i}\left[\sum_{j = 1}^{3N} p_j \dot{q}_j - L(\vec{q}, \dot{\vec{q}}(\vec{q}, \vec{p}, t), t)\right]}{\vec{p}, t}\\
                                &= \sum_{j = 1}^{3N} p_j \pderiv{\dot{q}_j}{q_i} - \sum_{j = 1}^{3N} \pderiv{L}{q_j} \underbrace{\pderiv{q_j}{q_i}}_{\delta_{ij}}  - \sum_{j = 1}^{3N} \pderiv{L}{\dot{q}_j} \pderiv{\dot{q}_j}{q_i} \\
                                &= - \pderiv{L}{q_i} + \sum_{j = 1}^{3N} \pderiv{\dot{q}_j}{q_i}\cancelto{0}{\left[p_j - \pderiv{L}{\dot{q}_j}\right]} \text{ using \cref{pComp}} \\
                                &= - \deriv{}{t} \pderiv{L}{\dot{q}_i} \text{ using the Euler-Lagrange equations} \\
                                &= - \dot{p}_i \text{ using \cref{pComp} again}
\end{align*}
Thus the Euler Lagrange equations imply \textit{Hamilton's Equations}:
\begin{equation}
  \dot{q}_i = \at{\pderiv{H}{p_i}}{\vec{q}, t} \text{ and } \dot{p}_i = -\at{\pderiv{H}{q_i}}{\vec{p}, t} \label{hamiltonsEqs}
\end{equation}
The Euler-Lagrange equations are second order ODEs in a $3N$ dimensional \textit{configuration space} $\vec{q}$, whereas Hamilton's equations are first order ODEs in a $6N$ dimensional \textit{phase space} with coordinates $(\vec{q}, \vec{p})$.

We can in fact also obtain these equations as the Euler-Lagrange equations for the following functional:
\[
  I[\vec{q}, \vec{p}] = \int_{t_A}^{t_B} (\dot{\vec{q}} \cdot \vec{p} - H(\vec{q}, \vec{p}, t)) \d{t}
\]
where we fix $\vec{q}(t_A)$ and $\vec{q}(t_B)$ so that the Euler-Lagrange equations are valid.
\subsection{Time Dependence}
The full time derivative of the Hamiltonian is:
\begin{align*}
  \deriv{H}{t} &= \deriv{}{t}H(\vec{q}(t), \vec{p}(t), t) \\
               &= \sum_{i = 1}^{3N} \left(\pderiv{H}{q_i}\dot{q}_i + \pderiv{H}{p_i}\dot{p}_i\right) + \pderiv{H}{t} \\
               &= \sum_{i = 1}^{3N} (-\dot{p}_i \dot{q}_i + \dot{q}_i \dot{p}_i) + \pderiv{H}{t} \text{ using \cref{hamiltonsEqs}} \\
               &= \pderiv{H}{t}
\end{align*}
Thus if the Hamiltonian has \textbf{no explicit time dependence}, then it has \textbf{no implicit time dependence} and so is conserved.
\section{Symmetry and Noether's Theorem}
We want to explore the following idea:
\begin{center}
  ``Continuous symmetries of the action are in correspondence with the 1st integrals of the Euler-Lagrange equations.''
\end{center}
\subsection{Symmetries of the Action}
Consider a smooth map $\vec{Q}: \R^{6N + 1} \to \R^{3N}$.
Given a curve $\vec{q}(t)$, we define the curve $\vec{q}^{*}$ as:
\[
  \vec{q}^{*}(t) = \vec{Q}(\vec{q}(t), \dot{\vec{q}}(t), t)
\]
We can now define what we mean by the ``symmetry of the action'':
\begin{definition}[Symmetry of the Action]
  We say that $\vec{Q}$  is a \textit{symmetry of the action} if for all curves $\vec{q}(t)$ and all $t_A, t_B$ we have: \label{symAct}
  \[
    I[\vec{q}^{*}] = \int_{t_A}^{t_B} L(\vec{q}^{*}(t), \dot{\vec{q}}^{*}(t), t) \d{t} = \int_{t_A}^{t_B} L(\vec{q}(t), \dot{\vec{q}}(t), t) \d{t} = I[\vec{q}]
  \]
  That is, ``applying'' $\vec{Q}$ does not change the action.
\end{definition}
By ``\textbf{continuous} symmetry'', are referring to a \textit{1-parameter family} of such maps, that is, we have a collection of maps characterised in a continuously by the parameter $s$:
\[
  \vec{Q}(\vec{q}, \dot{\vec{q}}, t, s)
\]
where we insist that $\vec{Q}(\vec{q}, \dot{\vec{q}}, t, 0) = \vec{q}$, i.e. $s = 0$ corresponds to the identity map.
We now define $q^{*}$ to include this parameter $s$, that is:
\begin{equation}
  \vec{q}^{*}(t, s) = \vec{Q}(\vec{q}(t), \dot{\vec{q}}(t), t, s) \label{smallS}
\end{equation}
Considering the behaviour of this for small $s$, using a Taylor expansion:
\begin{align*}
  \vec{q}^{*}(t, s) &= \vec{q}^{*}(t, 0) + s \at{\pderiv{\vec{q}^{*}}{s}}{s = 0} + O(s^2) \\
                    &= \vec{q}(t) + s \at{\pderiv{\vec{Q}}{s}}{s = 0} + O(s^2)
\end{align*}
\begin{theorem}[Noether's Theorem -- Verson 1]
  If there exists such a 1-parameter family of symmetries of the action then any solution of the Euler-Lagrange equations has a conserved quantity (i.e. a first integral): \label{noetherV1}
  \[
    \sum_{i = 1}^{3N} \pderiv{L}{\dot{q}_i} \at{\pderiv{Q_i}{s}}{s = 0} = \text{constant}
  \]
\end{theorem}
\begin{remark}
  Note that there is no $s$ dependence in the above, only dependence on $\vec{q}, \dot{\vec{q}}, t$.
\end{remark}
\begin{proof}
  Since $\vec{Q}$ is always a symmetry of the action, we have from \cref{symAct} that:
  \[
    \int_{t_A}^{t_B} [L(\vec{q}^{*}(t, s), \dot{\vec{q}}^{*}(t, s), t) - L(\vec{q}(t), \dot{\vec{q}}(t), t)] \d{t} = 0\ \forall s
  \]
  Substituting in our expression for small $s$ from \cref{smallS} and using summation convention for brevity, we have:
  \begin{align*}
    &\int_{t_A}^{t_B} \left[L\left(\vec{q}(t) + s \at{\pderiv{\vec{Q}}{s}}{s = 0} + \cdots, \dot{\vec{q}}(t) + s \deriv{}{t}\left[\at{\pderiv{\vec{Q}}{s}}{s = 0}\right] + \cdots, t\right) -L(\vec{q}(t), \dot{\vec{q}}(t), t)\right] \d{t} \\
    &= \int_{t_A}^{t_B} \left[s \at{\pderiv{Q_i}{s}}{s=0} \pderiv{L}{q_i} + s \deriv{}{t}\left[\at{\pderiv{Q_i}{s}}{s = 0}\right] \pderiv{L}{\dot{q}_i} + \cdots\right] \d{t} \text{ by Taylor expanding} \\
    &=\int_{t_A}^{t_B} \left[s \at{\pderiv{Q_i}{s}}{s = 0} \underbrace{\left(\pderiv{L}{q_i} - \deriv{}{t}\pderiv{L}{\dot{q}_i}\right)}_{=0} + s \left[\at{\pderiv{Q_i}{s}}{s=0}\deriv{}{t}\pderiv{L}{\dot{q}_i} + \deriv{}{t}\at{\pderiv{Q_i}{s}}{s=0}\pderiv{L}{\dot{q}_i}\right] + \cdots\right] \d{t} \\
    &=\int_{t_A}^{t_B} \left[s \deriv{}{t}\left[\pderiv{L}{\dot{q}_i} \at{\pderiv{Q_i}{s}}{s = 0}\right] + \cdots\right] \d{t} \\
    &= \eval{s \pderiv{L}{\dot{q}_i} \at{\pderiv{Q_i}{s}}{s = 0}}{t_A}{t_B} + O(s^2)
  \end{align*}
  We know this whole integral vanishes for all $s$, so if we fix some sufficiently small $s$, then $\pderiv{L}{\dot{q}_i} \at{\pderiv{Q_i}{s}}{s = 0}$ is the same at both $t_A$ and $t_{B}$.
  But in our definition of a symmetry of the action (\cref{symAct}), this has to hold for arbitrary $t_A$ and $t_B$ and so $\pderiv{L}{\dot{q}_i} \at{\pderiv{Q_i}{s}}{s = 0}$ takes the same value at any time, i.e. it is conserved.
\end{proof}
\begin{example}[Space Translational Symmetry in One Coordinate]
  Suppose that $L$ does not depend on a particular component of $\vec{q}$, say $q_1$.
  Consider the 1-parameter family of maps defined by:
  \[
    \vec{Q}(\vec{q}, s) = (q_1 + s, q_2, q_3, \ldots, q_{3N})
  \]
  that is, the components of $\vec{q}^{*}$ are given by:
  \begin{align*}
    q^{*}_1 &= q_1 + s \text{ and } q^{*}_i = q_i\ \forall i \in \{2, \ldots, 3N\} \\
    \dot{q}^{*}_i &= \dot{q}_i\ \forall i \in \{1, 2, \ldots, 3N\}
  \end{align*}
  Note that this satisfies $\vec{Q}(\vec{q}, 0) = \vec{q}$ as required for \cref{noetherV1}.

  This must be a symmetry of the action since we assumed that $L$ is independent of $q_1$ and so the action must be conserved.
  Thus by \cref{noetherV1}, we have a conserved quantity:
  \[
    \sum_{i = 1}^{3N} \pderiv{L}{\dot{q}_i} \at{\pderiv{Q_i}{s}}{s = 0} = \pderiv{L}{\dot{q}_1} \underbrace{\at{\pderiv{Q_1}{s}}{s = 0}}_{=1} = \pderiv{L}{\dot{q_1}} = p_1
  \]
  using \cref{pComp}.

  So if the Lagrangian does not depend on a particular component of $\vec{q}$, then its conjugate momentum is a conserved quantity.
  In this case, since $L$ does not depend on $q_1$, its conjugate momentum $p_1$ is conserved.
\end{example}
\begin{example}[Translational Symmetry in Euclian Space]
  Suppose that we have $N$ particles using Cartesian coordinates in $\R^3$, i.e. $\vec{q} = (\vec{x}_1, \ldots, \vec{x}_N)$, and that $L$ depends on $\vec{q}$ only by the difference in position $\vec{x}_i - \vec{x}_j$ of the particles.
  Physically, this encodes the idea that the laws of physics are invariant under spatial transformations.

  Consider the 1-parameter family of maps defined by:
  \[
    \vec{Q}(\vec{q}, s) = (\vec{x}_1 + s\vec{a}, \vec{x}_2 + s\vec{a}, \ldots, \vec{x}_N + s\vec{a})
  \]
  for some constant $\vec{a} \in \R^3$.
  Note that this satisfies $\vec{Q}(\vec{q}, 0) = \vec{q}$ as required for \cref{noetherV1}.

  We see that $\vec{x}^{*}_i - \vec{x}^{*}_j = \vec{x}_i - \vec{x}_j$ and so $L(\vec{q}^{*}, \dot{\vec{q}}^{*}, t) = L(\vec{q}, \dot{\vec{q}}, t)$ since $L$ only depends on the differences which are preserved under $\vec{Q}$.
  Thus $\vec{Q}$ is a symmetry of the action and so by \cref{noetherV1}, we have a conserved quantity:
  \begin{align*}
    \sum_{i = 1}^{3N} \pderiv{L}{\dot{q_i}} \at{\pderiv{Q_i}{s}}{s = 0} &= \sum_{i = 1}^{N} \left[\pderiv{L}{(\dot{\vec{x}}_i)_1} a_1 + \pderiv{L}{(\dot{\vec{x}}_i)_2} a_2 + \pderiv{L}{(\dot{\vec{x}}_i)_3} a_3\right] \\
                                                                        &= \sum_{i = 1}^{N} \left[(\vec{p}_i)_1 a_1 + (\vec{p}_i)_2 a_2 + (\vec{p}_i)_3 a_3\right] \text{ using \cref{pComp}} \\
                                                                        &= \vec{a} \cdot \sum_{i = 1}^{N} \vec{p}_i = \vec{a} \cdot \vec{P}
  \end{align*}
  where $\vec{P}$ is the total momentum of the system.

  Since $\vec{a} \cdot \vec{P}$ is conserved for any arbitrary $\vec{a}$, we conclude that $\vec{P}$ is conserved.
  Thus the invariance of physical laws under spatial transformations implies the conservation of momentum.
\end{example}
\begin{example}[Rotational Symmetries of Cartesian Coordinates]
  Consider again a system of $N$ particles in 3D Cartesian space, i.e. $\vec{q} = (\vec{x}_1, \ldots, \vec{x}_N)$.
  A rotation $R$ maps:
  \[
    \vec{q} \mapsto \vec{q}_{R} = (R\vec{x}_1, \ldots, R\vec{x}_N)
  \]
  Suppose that $L$ is invariant under notations, that is:
  \[
    L(\vec{q}_R, \dot{\vec{q}}_R, t) = L(\vec{q}, \dot{\vec{q}}, t) \text{ for all rotations $R$}
  \]
  Physically, this encodes the idea that the laws of physics are invariant under rotations.
  Consider the 1-parameter family of maps defined by:
  \begin{equation}
    \vec{Q}(\vec{q}, s) = \vec{q}_{R(s)} \tag{$\dag$} \label{eqnQRot}
  \end{equation}
  where $R(s)$ is a rotation through angle $s$ about some axis $\uvec{n}$.
  Note that this satisfies $\vec{Q}(\vec{q}, 0) = \vec{q}$ as required for \cref{noetherV1}.

  When applying \cref{noetherV1}, we only care about the behaviour $\vec{Q}$ for infinitesimal $s$.
  Recall from IA Vectors and Matrices that:
  \[
    R(s)\vec{x} = (\cos s)\vec{x} + (1 - \cos s)(\uvec{n} \cdot \vec{x})\uvec{n} + (\sin s)(\uvec{n} \times \vec{x})
  \]
  and so for infinitesimal $s$, we have:
  \[
    R(s)\vec{x} = \vec{x} + s \vec{n} \times \vec{x}
  \]
  We can then use this to find $\at{\pderiv{Q}{s}}{s = 0}$:
  \[
    \at{\pderiv{Q}{s}}{s = 0} = (\uvec{n} \times \vec{x}_1, \ldots, \uvec{n} \times \vec{x}_N)
  \]
  Because of \cref{eqnQRot}, we can apply \cref{noetherV1} and so have a conserved quantity:
  \begin{align*}
    \sum_{i = 1}^{3N} \pderiv{L}{\dot{q_i}} \at{\pderiv{Q_i}{s}}{s = 0} &= \sum_{i = 1}^{N} \left[\pderiv{L}{(\dot{\vec{x}}_i)_1} [\uvec{n} \times \vec{x}_i]_1 + \pderiv{L}{(\dot{\vec{x}}_i)_2} [\uvec{n} \times \vec{x}_i]_2 + \pderiv{L}{(\dot{\vec{x}}_i)_3} [\uvec{n} \times \vec{x}_i]_3\right] \\
                                                                        &= \sum_{i = 1}^{N} \vec{p}_i \cdot (\uvec{n} \times \vec{x}_i) \text{ using \cref{pComp}} \\
                                                                        &= \uvec{n} \cdot \sum_{i = 1}^{N} \vec{x}_i \cdot \vec{p}_i \text{ by cycling the triple product} \\
                                                                        &= \uvec{n} \cdot \vec{L}
  \end{align*}
  where $\vec{L}$ is the total angular momentum of the system.
  Since the axis $\uvec{n}$ was arbitrary, we conclude that $\vec{L}$ is conserved.
  Thus the invariance of physical laws under rotation implies the conservation of angular momentum.
\end{example}
\subsection{Symmetries with a Boundary Term}
We can generalise the definition of a 1-parameter family of symmetries of the action to allow a boundary term:
\[
  \int_{t_A}^{t_B} L(\vec{q}^{*}, \dot{\vec{q}}^{*}, t) \d{t} = \int_{t_A}^{t_B} L(\vec{q}, \dot{\vec{q}}, t) \d{t} + \eval{sK(\vec{q}, \dot{\vec{q}}, t)}{t_A}{t_B}
\]
for small $s$ and some function $K(\vec{q}, \dot{\vec{q}}, t)$.

In our proof for \cref{noetherV1}, we showed that:
\[
  \eval{s \pderiv{L}{\dot{q}_i} \at{\pderiv{Q_i}{s}}{s = 0}}{t_A}{t_B} = \int_{t_A}^{t_B} [L(\vec{q}^{*}, \dot{\vec{q}}^{*}, t) - L(\vec{q}, \dot{\vec{q}}, t)] \d{x}
\]
Now, instead of being zero, this is now our new boundary term and so:
\[
  \eval{s\left(\pderiv{L}{\dot{q}_i} \at{\pderiv{Q_i}{s}}{s = 0} - K(\vec{q}, \dot{\vec{q}}, t)\right)}{t_A}{t_B} = 0
\]
Thus by the same argument as before we conclude that we have a different conserved quantity:
\begin{equation}
  \sum_{i = 1}^{3N} \pderiv{L}{\dot{q}_i} \at{\pderiv{Q_i}{s}}{s = 0} - K(\vec{q}, \dot{\vec{q}}, t) \label{boundConserved}
\end{equation}
where we have now explicitly written out the sum instead of using summation convention.
\begin{example}[Galilean Boosts]
  Consider again a system of $N$ particles in 3D Cartesian space, i.e. $\vec{q} = (\vec{x}_1, \ldots, \vec{x}_N)$.
  Suppose that they are interacting through some potential $V$ and so the Lagrangian is given by:
  \[
    L = \sum_{i = 1}^{N} \frac{1}{2} m_i \dot{\vec{x}}_i \cdot \dot{\vec{x}}_i - V(\vec{x}_1, \ldots, \vec{x}_N)
  \]
  where we specify that \textbf{the potential} $V$ depends only on the separations $\vec{x}_i - \vec{x}_j$ of the particles.

  From IA Dynamics and Relativity, we know a Galilean boost is of the form $\vec{x}^{*} = \vec{x} - \vec{v}t$.
  Define the 1-parameter family of Galilean boosts with $\vec{v} = \uvec{n} s$ as:
  \[
    Q(\vec{q}, s) = (\vec{x} - \uvec{n} st, \vec{x}_2 - \uvec{n} st, \ldots, \vec{x}_{N} - \uvec{n} st)
  \]
  Note that this satisfies $\vec{Q}(\vec{q}, 0) = \vec{q}$ as required for \cref{noetherV1}.
  The components of $\vec{x}^{*}$ and $\dot{\vec{x}}^{*}$ are:
  \[
    \vec{x}^{*}_i = \vec{x}_i - \uvec{n}st \text{ and } \dot{\vec{x}}^{*}_i = \dot{\vec{x}}_i - \uvec{n} s
  \]
  Since $\vec{x}^{*}_i - \vec{x}^{*}_j = \vec{x}_i - \vec{x}_j$, the potential is unchanged under such boosts, that is:
  \[
    V(\vec{x}_1, \ldots, \vec{x}_N) = V(\vec{x}^{*}_{1}, \ldots, \vec{x}^{*}_{N})
  \]
  To check if this is a symmetry of the action we need to find the transformed Lagrangian:
  \begin{align*}
    L(\vec{x}^{*}, \dot{\vec{x}}^{*}, t) &= \sum_{i = 1}^{N} \frac{1}{2} m_i (\dot{\vec{x}}_i - \uvec{n} s) \cdot (\dot{\vec{x}}_i - \uvec{n} s)  - V(\vec{x}^{*}_1, \ldots, \vec{x}^{*}_N) \\
                                         &= \sum_{i = 1}^{N} \frac{1}{2} m_i \dot{\vec{x}}_i \cdot \dot{\vec{x}}_i - V(\vec{x}_1, \ldots, \vec{x}_N) - s \uvec{n} \cdot \sum_{i = 1}^{N} m_i \dot{\vec{x}}_i + O(s^2) \\
                                         &= L(\vec{x}, \dot{\vec{x}}, t) - \deriv{}{t}\left(s \uvec{n} \cdot \sum_{i = 1}^{N} m_i \vec{x}_i\right)
  \end{align*}
  Integrating both sides from $t_{A}$ to $t_{B}$ we have:
  \[
    \int_{t_A}^{t_B} L(\vec{q}^{*}, \dot{\vec{q}}^{*}, t) \d{t} = \int_{t_A}^{t_B} L(\vec{q}, \dot{\vec{q}}, t) \d{t} + \eval{s \uvec{n} \cdot \sum_{i = 1}^{N} m_i \vec{x}_i}{t_A}{t_B}
  \]
  So $\vec{Q}$ is a 1-parameter symmetry of the action but with boundary term:
  \[
    K = - \uvec{n} \cdot \sum_{i = 1}^{N} m_i \vec{x}_i = - \uvec{n} \cdot (M \vec{R})
  \]
  where $M$ is the total mass and $\vec{R}$ is the centre of mass.
  Thus by \cref{boundConserved}, we have a conserved quantity:
  \begin{align*}
    \sum_{i = 1}^{3N} \pderiv{L}{\dot{q}_i} \at{\pderiv{Q_i}{s}}{s = 0} - K &= \sum_{i = 1}^{N} \vec{p}_i \cdot (- \uvec{n} t) +  \uvec{n} \cdot (M\vec{R}) \\
                                                                            &= \uvec{n} \cdot (M\vec{R} - \vec{P} t)
  \end{align*}
  Since $\uvec{n}$ was arbitrary, the quantity $M\vec{R} - \vec{P}t$ is conserved which is equivalent to:
  \[
    \vec{0} = \deriv{}{t}(M\vec{R} - \vec{P} t) = M \dot{\vec{R}} - \vec{P} \iff \vec{P} = M \dot{\vec{R}}
  \]
  which is what we saw for systems of particles in IA Dynamics and Relativity.
\end{example}
\subsection{Symmetries of Time Transformations}
A different type of symmetry of the action involves transforming only the time coordinate.
The transformed time is given by $t^{*} = T(t)$ for an invertible function $T$.
The transformed path $\vec{q}^{*}$ is therefore parameterised by $t^{*}$ and so:
\[
  \vec{q}^{*}(t^{*}) = \vec{q}(t) = \vec{q}(T^{-1}(t^{*}))
\]
\begin{definition}[Time Transformation Symmetries of the Action]
  We say that the time transformation $t^{*} = T(t)$ is a \textit{symmetry of the action} if for all curves $\vec{q}(t)$ and all $t_A, t_B$ we have: \label{timeSym}
  \begin{equation}
    \int_{t^{*}_A}^{t^{*}_B} L\left(\vec{q}^{*}(t^{*}), \deriv{\vec{q}^{*}}{{t^{*}}}(t^{*}), t^{*}\right) \d{t^{*}} = \int_{t_{A}}^{t_{B}} L(\vec{q}(t), \dot{\vec{q}}(t), t) \d{t} \label{timeSymEqn}
  \end{equation}
\end{definition}
Similarly to before, we want to consider a 1-parameter family $T(t, s)$ of time transformation symmetries of the action parameterised by $s$.
Again, we insist that $T(t, 0) = t$.
We can also Taylor expand for small $s$ as we did before:
\begin{align}
  t^{*} &= T(t, s) \nonumber \\
        &= T(t, 0) + s \at{\pderiv{T}{s}}{s = 0} + O(s^2) \nonumber \\
        &= t + s \at{\pderiv{T}{s}}{s = 0} + O(s^2) \label{tStar}
\end{align}
\begin{theorem}[Noether's Theorem -- Version 2]
  If there exists a 1-parameter family of time transformation symmetries of the action as defined in \cref{timeSym}, then any solution of the Euler-Lagrange equation has a conserved quantity: \label{noetherV2}
  \[
    \left(L - \sum_{i = 1}^{3N} \dot{q}_i \pderiv{L}{\dot{q_i}}\right)\at{\pderiv{T}{s}}{s = 0} = \text{constant}
  \]
\end{theorem}
\begin{proof}
  To change variables from $t^{*}$ to $t$ in the action functional, we need $\deriv{t^{*}}{t}$.
  Using \cref{tStar}, we have:
  \begin{equation}
    \deriv{{t^{*}}}{t} = 1 + s \deriv{}{t}\at{\pderiv{T}{s}}{s = 0} + O(s^2) \tag{$\ast$} \label{tStarDeriv}
  \end{equation}
  Let $I^{*}$ and $I$ be the left hand sides and right hand sides of \cref{timeSymEqn} respectively, that is:
  \[
    I^{*} = \int_{t^{*}_A}^{t^{*}_B} L\left(\vec{q}^{*}(t^{*}), \deriv{\vec{q}^{*}}{{t^{*}}}(t^{*}), t^{*}\right) \d{t^{*}} \text{ and } I = \int_{t_{A}}^{t_{B}} L(\vec{q}(t), \dot{\vec{q}}(t), t) \d{t}
  \]
  We change variables in $I^{*}$ from $t^{*}$ to $t$:
  \[
    I^{*} = \int_{t_A}^{t_B} L\left(\vec{q}(t), \deriv{t}{{t^{*}}} \deriv{}{t}\vec{q}(t), T(t, s)\right) \deriv{t^{*}}{t} \d{t}
  \]
  We can find $\deriv{t}{{t^{*}}}$ by expanding the reciprocal of \cref{tStarDeriv}:
  \[
    \deriv{t}{{t^{*}}} = \left(\deriv{t^{*}}{t}\right)^{-1} = 1 - s \deriv{}{t} \at{\pderiv{T}{s}}{s = 0} + O(s^2)
  \]
  Using this and the expansion for $t^{*} = T(t, s)$ in \cref{tStar}, we have:
  \begin{align*}
    I^{*} &= \int_{t_A}^{t_B} L\left(\vec{q}(t), \dot{\vec{q}}(t) - s \deriv{}{t}\at{\pderiv{T}{s}}{s = 0} \dot{\vec{q}}(t) + \cdots, t + s \at{\pderiv{T}{s}}{s = 0} + \cdots\right) \\
          &\quad\qquad\left[1 + s \deriv{}{t}\at{\pderiv{T}{s}}{s = 0} + \cdots\right] \d{t} \\
  \end{align*}
  We then Taylor expand $L$ in its second and third arguments keeping only terms up to first order in $s$:
  \begin{align*}
   I^{*} &= \int_{t_A}^{t_B} \left[L(\vec{q}(t), \dot{\vec{q}}(t), t) - s \deriv{}{t}\at{\pderiv{T}{s}}{s = 0} \dot{q}_i \pderiv{L}{\dot{q}_i} + s \at{\pderiv{T}{s}}{s=0} \pderiv{L}{t} + O(s^2) \right] \\
         &\hspace{1.15cm}\left[1 + s \deriv{}{t}\at{\pderiv{T}{s}}{s = 0} + \cdots\right] \d{t} \\
         &= \int_{t_A}^{t_B} \left[L(\vec{q}(t), \dot{\vec{q}}(t), t) - s \deriv{}{t}\at{\pderiv{T}{s}}{s = 0} \dot{q}_i \pderiv{L}{\dot{q}_i} + s \at{\pderiv{T}{s}}{s=0} \pderiv{L}{t} \right. \\
         &\left.\hspace{1.15cm}+ sL \deriv{}{t}\at{\pderiv{T}{s}}{s = 0} + O(s^2)\right] \d{t} \\
         &= I + \int_{t_{A}}^{t_{B}} \left[s \deriv{}{t} \at{\pderiv{T}{s}}{s = 0}\left(L -\dot{q}_i \pderiv{L}{\dot{q}_i}\right) + s \at{\pderiv{T}{s}}{s = 0}\pderiv{L}{t} + O(s^2)\right] \d{t}
  \end{align*}
  The time derivative acting on $\at{\pderiv{T}{s}}{s = 0}$ here is quite awkward so we integrate the first term of this integral by parts to move it over:
  \begin{align*}
    &\int_{t_{A}}^{t_{B}} \left[s \deriv{}{t} \at{\pderiv{T}{s}}{s = 0}\left(L - \dot{q}_i \pderiv{L}{\dot{q}_i}\right)\right] \d{t} \\
    &= \eval{s \at{\pderiv{T}{s}}{s = 0}\left(L - \dot{q}_i \pderiv{L}{\dot{q}_i}\right)}{t_A}{t_B} - \int_{t_A}^{t_B} s \at{\pderiv{T}{s}}{s = 0}\deriv{}{t}\left(L - \dot{q}_i \pderiv{L}{\dot{q}_i}\right) \d{t}
  \end{align*}
  Regrouping terms we have:
  \[
    I^{*} = I + \eval{s \at{\pderiv{T}{s}}{s = 0}\left(L - \dot{q}_i \pderiv{L}{\dot{q}_i}\right)}{t_A}{t_B} + \int_{t_A}^{t_B} s \at{\pderiv{T}{s}}{s = 0}\left[\pderiv{L}{t} - \deriv{}{t}\left(L - \dot{q}_i \pderiv{L}{\dot{q}_i}\right)\right] \d{t}
  \]
  Recalling the proof for \cref{LnoT}, we arrived at \cref{derivPderiv} tells us that the term in square brackets in the above integral vanishes and so:
  \[
    I^{*} - I = \eval{s \at{\pderiv{T}{s}}{s = 0}\left(L - \dot{q}_i \pderiv{L}{\dot{q}_i}\right)}{t_A}{t_B}
  \]
  Since $t^{*} = T(t)$ is a symmetry of the action, $I^{*} - I = 0\ \forall t_A, t_B$ so by the same argument as at the end of the proof for \cref{noetherV1}, the quantity:
  \[
    \at{\pderiv{T}{s}}{s = 0}\left(L - \dot{q}_i \pderiv{L}{\dot{q}_i}\right)
  \]
  is independent of $t$ and so we are done.
\end{proof}
\begin{remark}
  We can also carry a spatial (as in \cref{noetherV1}) symmetry and a time symmetry (as above) simultaneously, yielding the conserved quantity:
  \[
    \sum_{i = 1}^{3N} \pderiv{L}{\dot{q}_i} \at{\pderiv{Q_i}{s}}{s = 0} + \left(L - \sum_{i = 1}^{3N} \dot{q}_i \pderiv{L}{\dot{q_i}}\right)\at{\pderiv{T}{s}}{s = 0} = \text{constant}
  \]
  for any solution of the Euler-Lagrange equations.
\end{remark}
\begin{example}
  Suppose that $L$ has no explicit $t$ dependence, i.e $L(\vec{q}, \dot{\vec{q}})$.
  Physically, this encodes the idea that the laws of physics are invariant under time translations.

  Consider the family of time translations defined by:
  \[
    T(t, s) = t + s \implies t^{*} = t + s
  \]
  Note that this satisfies $T(t, 0) = t$ as required for \cref{noetherV2}.
  So then:
  \[
    \vec{q}^{*}(t^{*}) = \vec{q}(t) = \vec{q}(t^{*} - s) \implies \deriv{q^{*}}{{t^{*}}} = \dot{\vec{q}}(t^{*} - s)
  \]
  We then calculate:
  \begin{align*}
    \int_{t^{*}_A}^{t^{*}_B} L\left(\vec{q}^{*}(t^{*}), \deriv{\vec{q}^{*}}{{t^{*}}}(t^{*})\right) \d{t^{*}} &= \int_{t_A + s}^{t_B + s} L(\vec{q}(t^{*} - s), \dot{\vec{q}}(t^{*} - s)) \d{t^{*}} \\
                                                                                                  &= \int_{t_A}^{t_B} L(\vec{q}(t), \dot{\vec{q}}(t)) \d{t}
  \end{align*}
  and so $t^{*} = T(t, s)$ is a symmetry of the action.

  Thus by \cref{noetherV2}, we have a conserved quantity:
  \[
    \underbrace{\at{\pderiv{T}{s}}{s = 0}}_{= 1}\left(L - \dot{q}_i \pderiv{L}{\dot{q}_i}\right) = L - \dot{q}_i \pderiv{L}{\dot{q}_i}
  \]
  Since $L$ has no explicit $t$ dependence, by \cref{LnoT}, the above is just the negation of the conserved energy.
  This means that invariance of the laws of physics under time translations implies the conservation of energy.
\end{example}
